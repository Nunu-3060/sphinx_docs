 10"""

 32def rotate_hue(color: str, degrees: float) -> str:
 33    """色相を degrees 度回転させた色を '#rrggbb' 形式で返す。
 34
 35    HLS の明るさ (L) と彩度 (S) は元の色のまま保つ。
 36    ただし、知覚される明るさは色相によって変わる。
 37    """
 38    r, g, b = (int(color[i:i + 2], 16) / 255 for i in (1, 3, 5))
 39    h, lightness, s = colorsys.rgb_to_hls(r, g, b)
 40    h = (h + degrees / 360) % 1.0
 41    r, g, b = colorsys.hls_to_rgb(h, lightness, s)
 42    return "#{:02x}{:02x}{:02x}".format(
 43        round(r * 255), round(g * 255), round(b * 255))
 44
 45
 46def make_schemes(base: str) -> dict[str, list[str]]:
 47    """配色パターンの名前と色のリストを返す。"""
 48    return {
 49        "補色": [base, rotate_hue(base, 180)],
 50        "類似色": [rotate_hue(base, -30), base, rotate_hue(base, 30)],
 51        "トライアド": [base, rotate_hue(base, 120), rotate_hue(base, 240)],
 52        "スプリット補色": [base, rotate_hue(base, 150),
 53                    rotate_hue(base, 210)],
 54    }
 55

 57def draw_schemes(schemes: dict[str, list[str]], path: Path) -> None:
 58    """配色パターンごとに色見本を 1 行ずつ描画して保存する。"""
 59    row_step = 1.3  # 色見本の高さ 0.8 + 色コードの表示領域
 60    fig, ax = plt.subplots(figsize=(7, 1.1 * len(schemes)))
 61    for row, (name, colors) in enumerate(schemes.items()):
 62        y = (len(schemes) - 1 - row) * row_step
 63        ax.text(-0.1, y + 0.4, name, ha="right", va="center", fontsize=12)
 64        for col, color in enumerate(colors):
 65            ax.add_patch(Rectangle((col * 1.6, y), 1.5, 0.8, color=color))
 66            ax.text(col * 1.6 + 0.75, y - 0.1, color, ha="center",
 67                    va="top", fontsize=9)
 68    ax.set_xlim(-2.2, 5.0)
 69    ax.set_ylim(-0.5, len(schemes) * row_step)
 70    ax.axis("off")
 71    fig.savefig(path, dpi=100, bbox_inches="tight", facecolor="white")
 72    plt.close(fig)

 90    ax.set_xlim(0, 100)
 91    ax.set_ylim(-1.5, 0.5)
 92    ax.axis("off")

 47def hex_to_rgb(code: str) -> RGB:
 48    """'#rrggbb' 形式の文字列を (R, G, B) に変換する。"""
 49    code = code.lstrip("#")
 50    return (int(code[0:2], 16), int(code[2:4], 16), int(code[4:6], 16))

 53def srgb_to_linear(value: int) -> float:
 54    """sRGB の 0〜255 の値を、ガンマ補正を外した 0〜1 の値に変換する。"""
 55    c = value / 255
 56    if c <= 0.04045:
 57        return c / 12.92
 58    return float(((c + 0.055) / 1.055) ** 2.4)
 59
 60
 61def relative_luminance(rgb: RGB) -> float:
 62    """相対輝度 (黒 = 0, 白 = 1) を返す。"""
 63    r, g, b = (srgb_to_linear(v) for v in rgb)
 64    return 0.2126 * r + 0.7152 * g + 0.0722 * b
 65
 66
 67def contrast_ratio(color1: RGB, color2: RGB) -> float:
 68    """2 色のコントラスト比 (1〜21) を返す。引数の順序は問わない。"""
 69    l1 = relative_luminance(color1)
 70    l2 = relative_luminance(color2)
 71    lighter, darker = max(l1, l2), min(l1, l2)
 72    return (lighter + 0.05) / (darker + 0.05)
 73
 74
 75def judge(ratio: float) -> str:
 76    """WCAG 2.2 の達成基準 (通常の大きさの文字) に対する判定を返す。"""
 77    if ratio >= 7.0:
 78        return "AAA"
 79    if ratio >= 4.5:
 80        return "AA"
 81    if ratio >= 3.0:
 82        return "大きな文字のみ AA"
 83    return "不適合"
 84

 86def draw_samples(pairs: list[tuple[str, str]]) -> Image.Image:
 87    """色の組み合わせごとに、見本の文字とコントラスト比を描画する。"""
 88    row_h, width = 64, 720
 89    image = Image.new("RGB", (width, row_h * len(pairs)), "white")
 90    draw = ImageDraw.Draw(image)
 91    sample_font = load_font(22)
 92    info_font = load_font(18)
 93
 94    for i, (fg, bg) in enumerate(pairs):
 95        top = i * row_h
 96        ratio = contrast_ratio(hex_to_rgb(fg), hex_to_rgb(bg))
 97        draw.rectangle((0, top, 360, top + row_h - 1), fill=bg)
 98        draw.text((20, top + row_h // 2), "読みやすさの見本 Aa",
 99                  font=sample_font, fill=fg, anchor="lm")
100        info = f"{fg} / {bg}  {ratio:5.2f} : 1  {judge(ratio)}"
101        draw.text((380, top + row_h // 2), info, font=info_font,
102                  fill="#222222", anchor="lm")
103    return image

 37def spread_labels(values: list[float], min_gap: float) -> list[float]:
 38    """ラベルの y 座標が min_gap 以上離れるように調整した値を返す。

 42    """
 43    order = sorted(range(len(values)), key=lambda i: values[i])
 44    adjusted = list(values)
 45    for prev, curr in zip(order, order[1:]):
 46        adjusted[curr] = max(adjusted[curr], adjusted[prev] + min_gap)
 47    return adjusted

 43    """ラベルの y 座標が min_gap 以上離れるように調整した値を返す。
 44
 45    値の小さい順に見ていき、直前のラベルと近すぎる場合は上へずらす。
 46    戻り値の順序は引数と同じ。
 47    """
 48    order = sorted(range(len(values)), key=lambda i: values[i])
 49    adjusted = list(values)
 50    for prev, curr in zip(order, order[1:]):
 51        adjusted[curr] = max(adjusted[curr], adjusted[prev] + min_gap)
 52    return adjusted

 55def lightness(srgb: FloatArray) -> FloatArray:
 56    """(N, 3) の sRGB (0〜1) 配列から CIE L* (0〜100) を計算する。"""
 57    linear = np.where(srgb <= 0.04045, srgb / 12.92,
 58                      ((srgb + 0.055) / 1.055) ** 2.4)
 59    y = linear @ RGB_TO_XYZ[1]  # L* の計算には Y (輝度) だけを使う
 60    epsilon = (6 / 29) ** 3
 61    f = np.where(y > epsilon, np.cbrt(y), y / (3 * (6 / 29) ** 2) + 4 / 29)
 62    result: FloatArray = 116 * f - 16
 63    return result

 47class IconPen:
 48    """グリッド単位の座標をピクセルに変換して描画する。"""
 49
 50    def __init__(self, draw: ImageDraw.ImageDraw, scale: float,
 51                 offset: Point = (0, 0)) -> None:
 52        self.draw = draw
 53        self.scale = scale  # 1 グリッド単位あたりのピクセル数
 54        self.offset = offset
 55        self.width = max(1, round(STROKE * scale))
 56
 57    def px(self, point: Point) -> Point:
 58        """グリッド座標をピクセル座標に変換する。"""
 59        return (self.offset[0] + point[0] * self.scale,
 60                self.offset[1] + point[1] * self.scale)
 61
 62    def lines(self, points: list[Point]) -> None:
 63        """折れ線を描く。角は丸める。"""
 64        self.draw.line([self.px(p) for p in points], fill=ICON_COLOR,
 65                       width=self.width, joint="curve")
 66
 67    def rect(self, box: tuple[float, float, float, float],
 68             radius: float = 0) -> None:
 69        """四角形の輪郭を描く。radius は角の丸みの半径 (グリッド単位)。"""
 70        left, top = self.px((box[0], box[1]))
 71        right, bottom = self.px((box[2], box[3]))
 72        self.draw.rounded_rectangle((left, top, right, bottom),
 73                                    radius=radius * self.scale,
 74                                    outline=ICON_COLOR, width=self.width)
 75
 76    def circle(self, center: Point, radius: float) -> None:
 77        """円の輪郭を描く。"""
 78        cx, cy = center
 79        self.ellipse((cx - radius, cy - radius, cx + radius, cy + radius))
 80
 81    def ellipse(self, box: tuple[float, float, float, float]) -> None:
 82        """楕円の輪郭を描く。box はグリッド座標の (左, 上, 右, 下)。"""
 83        left, top = self.px((box[0], box[1]))
 84        right, bottom = self.px((box[2], box[3]))
 85        self.draw.ellipse((left, top, right, bottom), outline=ICON_COLOR,
 86                          width=self.width)
 87
 88    def arc(self, box: tuple[float, float, float, float], start: float,
 89            end: float) -> None:
 90        """楕円の弧を描く。角度は 3 時の方向を 0 度とし、時計回り。"""
 91        left, top = self.px((box[0], box[1]))
 92        right, bottom = self.px((box[2], box[3]))
 93        self.draw.arc((left, top, right, bottom), start, end,

 92def draw_kpi_card(ax: Axes, kpi: Kpi) -> None:
 93    """指標カードを 1 枚描く。数値を最も大きく表示する。"""
 94    ax.axis("off")
 95    ax.add_patch(FancyBboxPatch((0, 0), 1, 1, boxstyle="round,pad=0,"
 96                                "rounding_size=0.06", facecolor=SURFACE,
 97                                edgecolor="none", transform=ax.transAxes))
 98    ax.text(0.08, 0.78, kpi.name, fontsize=11, color=SUBTEXT,
 99            transform=ax.transAxes)
100    ax.text(0.08, 0.40, kpi.value, fontsize=18, fontweight="bold",
101            color=TEXT, transform=ax.transAxes)
102    ax.text(0.08, 0.14, kpi.change, fontsize=10,
103            color=GOOD if kpi.is_good else SUBTEXT, transform=ax.transAxes)

106def draw_after(path: Path) -> None:
107    """改善後: 結論 → 指標 → 根拠の順に、上から下へ読める構成。"""
108    fig = plt.figure(figsize=(10, 6))
109    grid = fig.add_gridspec(3, 4, height_ratios=[0.5, 0.9, 2.6],
110                            left=0.06, right=0.97, top=0.95, bottom=0.08,
111                            hspace=0.45, wspace=0.25)
112
113    ax_head = fig.add_subplot(grid[0, :])
114    ax_head.axis("off")
115    ax_head.text(0, 0.75, "2026 年 9 月 月次報告", fontsize=11,
116                 color=SUBTEXT, transform=ax_head.transAxes)
117    ax_head.text(0, 0.05, "売上は前月比 8 % 増。東日本が全体の 3 分の 1 を占める",
118                 fontsize=22, fontweight="bold", color=TEXT,
119                 transform=ax_head.transAxes)
120
121    for i, kpi in enumerate(KPIS):
122        draw_kpi_card(fig.add_subplot(grid[1, i]), kpi)
123
124    # 構成比を比べる場合も、円グラフより並べ替えた棒グラフの方が読みやすい
125    ax_region = fig.add_subplot(grid[2, :2])
126    region = REGION_SALES.sort_values()
127    colors = [PRIMARY if name == "東日本" else MUTED for name in region.index]
128    ax_region.barh(region.index, region, color=colors, height=0.6)
129    for y, value in enumerate(region):
130        ax_region.text(value + 0.05, y, f"{value:.1f}", va="center",
131                       fontsize=10, color=SUBTEXT)
132    ax_region.set_title("地域別売上 (千万円)", loc="left", fontsize=12)
133
134    ax_month = fig.add_subplot(grid[2, 2:])
135    x = range(len(MONTHLY_SALES))
136    ax_month.plot(x, MONTHLY_SALES, color=PRIMARY, linewidth=2.5)
137    ax_month.scatter([x[-1]], [MONTHLY_SALES.iloc[-1]], color=PRIMARY,
138                     zorder=3)
139    ax_month.annotate(f"{MONTHLY_SALES.iloc[-1]:.2f}", (x[-1],
140                      MONTHLY_SALES.iloc[-1]), xytext=(-8, 8),
141                      textcoords="offset points", ha="right",
142                      color=PRIMARY, fontweight="bold")
143    ax_month.set_xticks(list(x)[::2], MONTHLY_SALES.index[::2])
144    ax_month.set_ylim(0.9, 1.3)
145    ax_month.set_title("月別売上 (億円)", loc="left", fontsize=12)
146    ax_month.grid(axis="y", color="#e5e5e5")
147
148    for ax in (ax_region, ax_month):
149        for side in ("top", "right"):
150            ax.spines[side].set_visible(False)
151        ax.tick_params(colors=SUBTEXT, length=0)
152    ax_region.spines["bottom"].set_visible(False)
153    ax_region.set_xticks([])
154    fig.savefig(path, dpi=100, facecolor="white")
155    plt.close(fig)
