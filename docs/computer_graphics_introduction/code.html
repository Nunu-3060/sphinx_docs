cg_utils.py

全ての章で使用する共通のモジュールである。ベクトルの正規化、sRGB のガンマ補正、画像の保存と拡大、図を横に並べる関数などを定義している。

examples/cg_utils.py

  1"""全ての章で使う共通の関数と型をまとめたモジュール。
  2
  3本資料のサンプルコードでは、画像を NumPy 配列で表す。
  4
  5* 画像は形状 (高さ, 幅, 3) の配列とし、末尾の軸を (R, G, B) とする。
  6* 色の値は 0.0 から 1.0 の実数で扱い、保存するときだけ 0 から 255 の
  7  整数に変換する。
  8* 照明の計算は光の強さに比例する線形な値で行い、保存する直前に
  9  sRGB の値に変換する。
 10
 11主な内容は次のとおり。
 12
 13* ベクトルの正規化
 14* sRGB のガンマ補正(線形な値と sRGB の値の相互変換)
 15* 画像の保存、拡大、横に並べた図の作成
 16* 図の保存先ディレクトリの取得
 17"""
 18
 19import sys
 20from pathlib import Path
 21from typing import Sequence
 22
 23import numpy as np
 24from numpy.typing import NDArray
 25from PIL import Image
 26
 27FloatArray = NDArray[np.float64]
 28IntArray = NDArray[np.int64]
 29BoolArray = NDArray[np.bool_]
 30
 31# 図の背景や、図を並べるときの余白の色(白)。
 32WHITE: FloatArray = np.array([1.0, 1.0, 1.0])
 33
 34
 35# ---------------------------------------------------------------------------
 36# ベクトル
 37# ---------------------------------------------------------------------------
 38
 39def normalize(v: FloatArray) -> FloatArray:
 40    """末尾の軸をベクトルとみなし、長さを 1 にしたベクトルを返す。
 41
 42    長さが 0 のベクトルは、0 除算を避けるためにそのまま返す。
 43    """
 44    length = np.linalg.norm(v, axis=-1, keepdims=True)
 45    return np.asarray(v / np.where(length > 0.0, length, 1.0),
 46                      dtype=np.float64)
 47
 48
 49# ---------------------------------------------------------------------------
 50# ガンマ補正
 51# ---------------------------------------------------------------------------
 52
 53def srgb_to_linear(c: FloatArray) -> FloatArray:
 54    """sRGB の値(0.0 から 1.0)を、光の強さに比例する線形な値に変換する。"""
 55    c = np.asarray(c, dtype=np.float64)
 56    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)
 57
 58
 59def linear_to_srgb(c: FloatArray) -> FloatArray:
 60    """線形な値を sRGB の値に変換する。範囲外の値は 0.0 から 1.0 に収める。"""
 61    c = np.clip(np.asarray(c, dtype=np.float64), 0.0, 1.0)
 62    return np.where(c <= 0.0031308, c * 12.92,
 63                    1.055 * c ** (1.0 / 2.4) - 0.055)
 64
 65
 66# ---------------------------------------------------------------------------
 67# 画像
 68# ---------------------------------------------------------------------------
 69
 70def to_pil(image: FloatArray) -> Image.Image:
 71    """0.0 から 1.0 の値を持つ (高さ, 幅, 3) の配列を Pillow の画像にする。"""
 72    data = np.clip(image, 0.0, 1.0) * 255.0 + 0.5
 73    return Image.fromarray(data.astype(np.uint8))
 74
 75
 76def from_pil(image: Image.Image) -> FloatArray:
 77    """Pillow の画像を 0.0 から 1.0 の値を持つ (高さ, 幅, 3) の配列にする。"""
 78    return np.asarray(image.convert("RGB"), dtype=np.float64) / 255.0
 79
 80
 81def enlarge(image: FloatArray, scale: int) -> FloatArray:
 82    """画素を scale × scale の正方形に引き伸ばして画像を拡大する。
 83
 84    画素の様子が見えるように、補間はせずに同じ値を繰り返す。
 85    """
 86    return np.repeat(np.repeat(image, scale, axis=0), scale, axis=1)
 87
 88
 89def hstack(images: Sequence[FloatArray], gap: int = 16) -> FloatArray:
 90    """高さの同じ画像を、白い余白を挟んで横に並べた画像を返す。"""
 91    height = images[0].shape[0]
 92    spacer = np.ones((height, gap, 3))
 93    parts: list[FloatArray] = []
 94    for i, image in enumerate(images):
 95        if i > 0:
 96            parts.append(spacer)
 97        parts.append(image)
 98    return np.concatenate(parts, axis=1)
 99
100
101def save_image(image: FloatArray, path: Path) -> None:
102    """画像を PNG 形式で保存する。"""
103    to_pil(image).save(path)
104    print(f"saved: {path}")
105
106
107def output_dir() -> Path:
108    """図の保存先ディレクトリを返す。
109
110    コマンドライン引数で指定されていればそのディレクトリを、なければ
111    カレントディレクトリの下の ``figures`` を使う。
112    """
113    path = Path(sys.argv[1]) if len(sys.argv) > 1 else Path("figures")
114    path.mkdir(parents=True, exist_ok=True)
115    return path

image_basics.py

デジタル画像で使用する。解像度の比較、R、G、B のチャンネル、ガンマ補正とグラデーション、アルファ値による合成の図を作る。

examples/image_basics.py

  1"""「デジタル画像」の章の図を作るスクリプト。
  2
  3次の図を作る。
  4
  5* resolution.png: 同じ円を異なる解像度で表した画像
  6* channels.png: カラー画像と、その R、G、B のチャンネル
  7* gamma_blend.png: sRGB の値と線形な値で補間したグラデーション
  8* alpha_over.png: アルファ値を使った半透明の合成
  9"""
 10
 11import numpy as np
 12
 13from cg_utils import (BoolArray, FloatArray, WHITE, enlarge, hstack,
 14                      linear_to_srgb, output_dir, save_image,
 15                      srgb_to_linear)
 16
 17
 18def pixel_centers(width: int, height: int) -> tuple[FloatArray, FloatArray]:
 19    """各画素の中心の座標 (x, y) を返す。
 20
 21    画素 (i, j) の中心は (i + 0.5, j + 0.5) とする。y 軸は下向きである。
 22    """
 23    xs, ys = np.meshgrid(np.arange(width) + 0.5, np.arange(height) + 0.5)
 24    return xs, ys
 25
 26
 27def disk_mask(size: int, cx: float, cy: float, radius: float) -> BoolArray:
 28    """画素の中心が円の内側にあれば True となる配列を返す。
 29
 30    座標は 0.0 から 1.0 に正規化した値で指定する。
 31    """
 32    xs, ys = pixel_centers(size, size)
 33    return np.asarray((xs / size - cx) ** 2 + (ys / size - cy) ** 2
 34                      <= radius ** 2)
 35
 36
 37def add_grid(image: FloatArray, scale: int) -> FloatArray:
 38    """enlarge() で拡大した画像に、画素の境界を示す灰色の線を引く。"""
 39    result = image.copy()
 40    result[::scale, :, :] = 0.75
 41    result[:, ::scale, :] = 0.75
 42    return result
 43
 44
 45def resolution_figure() -> FloatArray:
 46    """同じ円を 8、16、64 画素四方で表した画像を、同じ大きさに拡大して並べる。"""
 47    display = 256
 48    panels = []
 49    for size in (8, 16, 64):
 50        mask = disk_mask(size, 0.5, 0.5, 0.38)
 51        image = np.where(mask[..., None], np.array([0.15, 0.35, 0.7]), WHITE)
 52        scale = display // size
 53        image = enlarge(image, scale)
 54        if scale >= 8:
 55            image = add_grid(image, scale)
 56        panels.append(image)
 57    return hstack(panels)
 58
 59
 60def test_image(width: int, height: int) -> FloatArray:
 61    """横方向に色相、縦方向に明るさが変わるカラー画像を作る。"""
 62    xs, ys = pixel_centers(width, height)
 63    hue = xs / width * 6.0
 64    # 色相環を 6 つの区間に分け、R、G、B それぞれの値を折れ線で作る。
 65    r = np.clip(np.abs(hue - 3.0) - 1.0, 0.0, 1.0)
 66    g = np.clip(2.0 - np.abs(hue - 2.0), 0.0, 1.0)
 67    b = np.clip(2.0 - np.abs(hue - 4.0), 0.0, 1.0)
 68    brightness = 1.0 - ys / height
 69    return np.stack([r, g, b], axis=-1) * brightness[..., None]
 70
 71
 72def channels_figure() -> FloatArray:
 73    """カラー画像と、その R、G、B のチャンネルを灰色で表した画像を並べる。"""
 74    image = test_image(192, 128)
 75    panels = [image]
 76    for channel in range(3):
 77        gray = np.repeat(image[..., channel:channel + 1], 3, axis=-1)
 78        panels.append(gray)
 79    return hstack(panels)
 80
 81
 82def lerp(a: FloatArray, b: FloatArray, t: FloatArray) -> FloatArray:
 83    """a と b を t : (1 - t) に内分した値を返す(線形補間)。"""
 84    return np.asarray(a * (1.0 - t) + b * t, dtype=np.float64)
 85
 86
 87def gamma_blend_figure() -> FloatArray:
 88    """赤から緑へのグラデーションを、sRGB の値と線形な値で補間して比べる。
 89
 90    上の帯は sRGB の値をそのまま補間したもの、下の帯は線形な値に変換して
 91    から補間し、sRGB の値に戻したものである。
 92    """
 93    width, height = 512, 64
 94    red = np.array([1.0, 0.0, 0.0])
 95    green = np.array([0.0, 1.0, 0.0])
 96    t = (np.arange(width) + 0.5) / width
 97    t = t[None, :, None]
 98
 99    srgb_band = lerp(red, green, t)
100    linear_band = linear_to_srgb(
101        lerp(srgb_to_linear(red), srgb_to_linear(green), t))
102
103    srgb_band = np.repeat(srgb_band, height, axis=0)
104    linear_band = np.repeat(linear_band, height, axis=0)
105    gap = np.ones((12, width, 3))
106    return np.concatenate([srgb_band, gap, linear_band], axis=0)
107
108
109def over(fg: FloatArray, alpha: FloatArray, bg: FloatArray) -> FloatArray:
110    """前景の色 fg を、アルファ値 alpha で背景 bg の上に重ねる。
111
112    Porter と Duff の over 演算子(前景のアルファ値が乗算済みでない場合)。
113    色は線形な値で渡す。
114    """
115    return np.asarray(fg * alpha + bg * (1.0 - alpha), dtype=np.float64)
116
117
118def checkerboard(width: int, height: int, cell: int) -> FloatArray:
119    """白と灰色の市松模様の画像を作る(線形な値)。"""
120    xs, ys = np.meshgrid(np.arange(width) // cell, np.arange(height) // cell)
121    light = (xs + ys) % 2 == 0
122    value = np.where(light, 0.9, 0.55)
123    return np.asarray(srgb_to_linear(np.repeat(value[..., None], 3, axis=-1)))
124
125
126def alpha_figure() -> FloatArray:
127    """市松模様の背景に、アルファ値 0.6 の赤、緑、青の円を順に重ねる。"""
128    size = 256
129    image = checkerboard(size, size, 16)
130    circles = [
131        ((0.38, 0.38), np.array([0.9, 0.1, 0.1])),
132        ((0.62, 0.38), np.array([0.1, 0.75, 0.2])),
133        ((0.5, 0.6), np.array([0.1, 0.25, 0.9])),
134    ]
135    for (cx, cy), color in circles:
136        mask = disk_mask(size, cx, cy, 0.25)
137        alpha = np.where(mask, 0.6, 0.0)[..., None]
138        image = over(srgb_to_linear(color), alpha, image)
139    return linear_to_srgb(image)
140
141
142def main() -> None:
143    out = output_dir()
144    save_image(resolution_figure(), out / "resolution.png")
145    save_image(channels_figure(), out / "channels.png")
146    save_image(gamma_blend_figure(), out / "gamma_blend.png")
147    save_image(alpha_figure(), out / "alpha_over.png")
148
149
150if __name__ == "__main__":
151    main()

raster2d.py

2 次元の図形のラスタライズで使用する。ブレゼンハムのアルゴリズムによる線分、エッジ関数による三角形の塗りつぶし、重心座標による色の補間、スーパーサンプリングの図を作る。

examples/raster2d.py

  1"""「2 次元の図形のラスタライズ」の章の図を作るスクリプト。
  2
  3考え方が分かりやすいように、画素を 1 つずつ調べる素朴な書き方をしている。
  4小さな画像にしか使わないので、速さは重視していない。
  5
  6次の図を作る。
  7
  8* line.png: ブレゼンハムのアルゴリズムで描いた線分
  9* triangle.png: エッジ関数で塗りつぶした三角形
 10* barycentric.png: 重心座標で頂点の色を補間した三角形
 11* aliasing.png: 画素ごとに 1 点で判定した図形と、16 点で判定した図形
 12"""
 13
 14import numpy as np
 15from PIL import Image, ImageDraw
 16
 17from cg_utils import (FloatArray, enlarge, from_pil, hstack, output_dir,
 18                      save_image, to_pil)
 19
 20Point = tuple[float, float]
 21
 22# 拡大図で、画素の境界の線と塗りつぶした画素に使う色。
 23GRID_COLOR = (190, 190, 190)
 24FILL_COLOR = np.array([0.15, 0.35, 0.7])
 25OUTLINE_COLOR = (220, 40, 40)
 26
 27
 28# ---------------------------------------------------------------------------
 29# 線分
 30# ---------------------------------------------------------------------------
 31
 32def draw_line(x0: int, y0: int, x1: int, y1: int) -> list[tuple[int, int]]:
 33    """ブレゼンハムのアルゴリズムで、線分上の画素の座標を求める。
 34
 35    整数の加算と比較だけで、全ての向きの線分を扱える形にしている。
 36    """
 37    pixels = []
 38    dx = abs(x1 - x0)
 39    dy = -abs(y1 - y0)
 40    sx = 1 if x0 < x1 else -1
 41    sy = 1 if y0 < y1 else -1
 42    err = dx + dy  # 理想的な線分からのずれ(を整数倍したもの)
 43    x, y = x0, y0
 44    while True:
 45        pixels.append((x, y))
 46        if x == x1 and y == y1:
 47            break
 48        e2 = 2 * err
 49        if e2 >= dy:  # x 方向に進む
 50            err += dy
 51            x += sx
 52        if e2 <= dx:  # y 方向に進む
 53            err += dx
 54            y += sy
 55    return pixels
 56
 57
 58# ---------------------------------------------------------------------------
 59# 三角形
 60# ---------------------------------------------------------------------------
 61
 62def edge_function(a: Point, b: Point, p: Point) -> float:
 63    """辺 a→b に対して、点 p がどちら側にあるかを表す値を返す。
 64
 65    ベクトル (b - a) と (p - a) の外積の z 成分であり、値の絶対値は
 66    3 点 a、b、p が作る三角形の面積の 2 倍に等しい。
 67    """
 68    return (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0])
 69
 70
 71def barycentric(v0: Point, v1: Point, v2: Point,
 72                p: Point) -> tuple[float, float, float] | None:
 73    """点 p の重心座標 (w0, w1, w2) を返す。p が三角形の外にあれば None。"""
 74    area = edge_function(v0, v1, v2)
 75    if area == 0.0:
 76        return None  # 面積が 0 の三角形は描かない
 77    w0 = edge_function(v1, v2, p) / area
 78    w1 = edge_function(v2, v0, p) / area
 79    w2 = edge_function(v0, v1, p) / area
 80    if w0 < 0.0 or w1 < 0.0 or w2 < 0.0:
 81        return None
 82    return w0, w1, w2
 83
 84
 85def rasterize_triangle(
 86        v0: Point, v1: Point, v2: Point, width: int, height: int
 87) -> list[tuple[int, int, tuple[float, float, float]]]:
 88    """三角形の内側にある画素と、その中心の重心座標を求める。
 89
 90    三角形を囲む長方形(バウンディングボックス)の画素だけを調べる。
 91    """
 92    xs = [v0[0], v1[0], v2[0]]
 93    ys = [v0[1], v1[1], v2[1]]
 94    x_min = max(int(np.floor(min(xs))), 0)
 95    x_max = min(int(np.ceil(max(xs))), width - 1)
 96    y_min = max(int(np.floor(min(ys))), 0)
 97    y_max = min(int(np.ceil(max(ys))), height - 1)
 98
 99    result = []
100    for y in range(y_min, y_max + 1):
101        for x in range(x_min, x_max + 1):
102            center = (x + 0.5, y + 0.5)
103            weights = barycentric(v0, v1, v2, center)
104            if weights is not None:
105                result.append((x, y, weights))
106    return result
107
108
109# ---------------------------------------------------------------------------
110# 図の作成
111# ---------------------------------------------------------------------------
112
113def grid_image(mask: FloatArray, scale: int) -> Image.Image:
114    """塗りつぶす画素を表す配列を拡大し、画素の境界の線を引いた画像を返す。"""
115    height, width = mask.shape
116    image = np.where(mask[..., None] > 0.0, FILL_COLOR, 1.0)
117    pil = to_pil(enlarge(image, scale))
118    draw = ImageDraw.Draw(pil)
119    for i in range(width + 1):
120        draw.line([(i * scale, 0), (i * scale, height * scale)],
121                  fill=GRID_COLOR)
122    for j in range(height + 1):
123        draw.line([(0, j * scale), (width * scale, j * scale)],
124                  fill=GRID_COLOR)
125    return pil
126
127
128def line_figure() -> FloatArray:
129    """ブレゼンハムのアルゴリズムで描いた線分と、元の線分を重ねて示す。"""
130    width, height, scale = 24, 14, 20
131    mask = np.zeros((height, width))
132    segments = [(1, 12, 22, 2), (2, 2, 9, 12)]
133    for x0, y0, x1, y1 in segments:
134        for x, y in draw_line(x0, y0, x1, y1):
135            mask[y, x] = 1.0
136    pil = grid_image(mask, scale)
137    draw = ImageDraw.Draw(pil)
138    for x0, y0, x1, y1 in segments:
139        # 画素 (x, y) の中心は拡大図の ((x + 0.5) * scale, ...) にある。
140        start = ((x0 + 0.5) * scale, (y0 + 0.5) * scale)
141        end = ((x1 + 0.5) * scale, (y1 + 0.5) * scale)
142        draw.line([start, end], fill=OUTLINE_COLOR, width=2)
143    return from_pil(pil)
144
145
146def triangle_figure() -> FloatArray:
147    """エッジ関数で塗りつぶした三角形と、元の三角形の輪郭を重ねて示す。"""
148    width, height, scale = 24, 14, 20
149    v0, v1, v2 = (2.3, 12.4), (21.6, 9.2), (8.7, 1.4)
150    mask = np.zeros((height, width))
151    for x, y, _ in rasterize_triangle(v0, v1, v2, width, height):
152        mask[y, x] = 1.0
153    pil = grid_image(mask, scale)
154    draw = ImageDraw.Draw(pil)
155    corners = [(x * scale, y * scale) for x, y in (v0, v1, v2)]
156    draw.polygon(corners, outline=OUTLINE_COLOR, width=2)
157    # 画素の中心に小さな点を打ち、判定に使う位置を示す。
158    for j in range(height):
159        for i in range(width):
160            cx, cy = (i + 0.5) * scale, (j + 0.5) * scale
161            draw.ellipse([cx - 1.5, cy - 1.5, cx + 1.5, cy + 1.5],
162                         fill=(90, 90, 90))
163    return from_pil(pil)

166def barycentric_figure() -> FloatArray:
167    """頂点に赤、緑、青を与え、重心座標で内部の色を補間した三角形を描く。"""
168    width, height = 320, 256
169    v0, v1, v2 = (30.0, 236.0), (290.0, 236.0), (160.0, 20.0)

 99def look_at(eye: FloatArray, target: FloatArray,
100            up: FloatArray) -> FloatArray:
101    """ビュー変換の行列。ワールド座標をカメラ座標に変換する。
102
103    カメラを eye に置き、target の方向に向ける。up はカメラの上の向きの
104    目安である。カメラ座標では、カメラは原点にあって -z 方向を向き、
105    上が +y、右が +x になる。
106    """
107    forward = normalize(target - eye)          # カメラの前方
108    right = normalize(np.cross(forward, up))   # カメラの右
109    true_up = np.cross(right, forward)         # カメラの上
110    rotation = np.eye(4)
111    rotation[0, :3] = right
112    rotation[1, :3] = true_up
113    rotation[2, :3] = -forward

117def perspective(fovy_degrees: float, aspect: float, near: float,
118                far: float) -> FloatArray:
119    """透視投影の行列。
120
121    fovy_degrees は縦方向の視野角、aspect は画面の幅 / 高さである。
122    カメラから near から far までの範囲が、正規化デバイス座標の
123    z = -1 から 1 に対応する。
124    """
125    s = 1.0 / np.tan(np.radians(fovy_degrees) / 2.0)
126    return np.array([
127        [s / aspect, 0.0, 0.0, 0.0],
128        [0.0, s, 0.0, 0.0],
129        [0.0, 0.0, (far + near) / (near - far),
130         2.0 * far * near / (near - far)],
131        [0.0, 0.0, -1.0, 0.0],
132    ])

 51def make_cube() -> Mesh:
 52    """1 辺の長さが 1 の立方体を作る。
 53
 54    面ごとに法線が異なるので、各面に専用の 4 つの頂点を持たせる
 55    (頂点は全部で 24 個)。
 56    """
 57    positions: list[FloatArray] = []
 58    normals: list[FloatArray] = []
 59    uvs: list[tuple[int, int]] = []
 60    faces: list[tuple[int, int, int]] = []
 61    # 各面の法線と、面の上で u、v が増える向き。
 62    sides = [
 63        ((1, 0, 0), (0, 0, -1), (0, 1, 0)),
 64        ((-1, 0, 0), (0, 0, 1), (0, 1, 0)),
 65        ((0, 1, 0), (1, 0, 0), (0, 0, -1)),
 66        ((0, -1, 0), (1, 0, 0), (0, 0, 1)),
 67        ((0, 0, 1), (1, 0, 0), (0, 1, 0)),
 68        ((0, 0, -1), (-1, 0, 0), (0, 1, 0)),
 69    ]
 70    for normal, u_dir, v_dir in sides:
 71        n, u, v = np.array(normal), np.array(u_dir), np.array(v_dir)
 72        base = len(positions)
 73        for su, sv in ((0, 0), (1, 0), (1, 1), (0, 1)):
 74            positions.append(0.5 * n + (su - 0.5) * u + (sv - 0.5) * v)
 75            normals.append(n)
 76            uvs.append((su, sv))
 77        faces += [(base, base + 1, base + 2), (base, base + 2, base + 3)]
 78    return Mesh(np.array(positions, dtype=np.float64),
 79                np.array(normals, dtype=np.float64),
 80                np.array(uvs, dtype=np.float64),
 81                np.array(faces, dtype=np.int64))

 84def make_sphere(n_lon: int = 48, n_lat: int = 24) -> Mesh:
 85    """半径 1 の球を、経線 n_lon 本、緯線 n_lat 本で区切って作る(UV 球)。"""
 86    lon = np.linspace(0.0, 2.0 * np.pi, n_lon + 1)
 87    lat = np.linspace(0.0, np.pi, n_lat + 1)
 88    lon_grid, lat_grid = np.meshgrid(lon, lat)
 89    x = np.sin(lat_grid) * np.cos(lon_grid)
 90    y = np.cos(lat_grid)
 91    z = -np.sin(lat_grid) * np.sin(lon_grid)
 92    positions = np.stack([x, y, z], axis=-1).reshape(-1, 3)
 93    u = lon_grid / (2.0 * np.pi)
 94    v = 1.0 - lat_grid / np.pi
 95    uvs = np.stack([u, v], axis=-1).reshape(-1, 2)
 96
 97    faces = []
 98    row = n_lon + 1
 99    for j in range(n_lat):
100        for i in range(n_lon):
101            a = j * row + i          # 左上
102            b = a + row              # 左下
103            # 極に接する四角形は三角形に縮退するので、1 つだけ作る。
104            if j > 0:
105                faces.append((a, b + 1, a + 1))
106            if j < n_lat - 1:
107                faces.append((a, b, b + 1))
108    # 球の中心は原点なので、位置ベクトルがそのまま法線になる。
109    return Mesh(positions, positions.copy(), uvs,
110                np.array(faces, dtype=np.int64))

128def flatten(mesh: Mesh) -> Mesh:
129    """頂点を三角形ごとに複製し、法線を面の法線に置き換えたメッシュを返す。
130
131    フラットシェーディングで使う。
132    """
133    positions = mesh.positions[mesh.faces].reshape(-1, 3)
134    p0, p1, p2 = (mesh.positions[mesh.faces[:, k]] for k in range(3))
135    face_normals = np.cross(p1 - p0, p2 - p0)
136    face_normals /= np.linalg.norm(face_normals, axis=-1, keepdims=True)
137    normals = np.repeat(face_normals, 3, axis=0)
138    uvs = mesh.uvs[mesh.faces].reshape(-1, 2)
139    faces = np.arange(len(positions), dtype=np.int64).reshape(-1, 3)
140    return Mesh(positions, normals, uvs, faces)

158        self.height = height

162        }

170def edge_function(ax: float, ay: float, bx: float, by: float,
171                  px: FloatArray, py: FloatArray) -> FloatArray:
172    """raster2d.py の edge_function() を、多数の点 (px, py) に対して計算する。"""
173    return (bx - ax) * (py - ay) - (by - ay) * (px - ax)

176def draw_mesh(fb: FrameBuffer, mvp: FloatArray, mesh: Mesh,
177              attributes: dict[str, FloatArray], cull_back: bool = True,
178              perspective_correct: bool = True) -> None:
179    """メッシュを描き、深度と属性をフレームバッファーに書き込む。
180
181    mvp はモデル、ビュー、投影の変換を掛け合わせた行列である。
182    attributes には頂点ごとの属性を、fb の attributes と同じ名前で渡す。
183    """
184    clip = to_clip(mvp, mesh.positions)
185    w = clip[:, 3]
186    # 透視除算とビューポート変換(y 軸は下向き)。
187    sx = (clip[:, 0] / w + 1.0) * 0.5 * fb.width
188    sy = (1.0 - clip[:, 1] / w) * 0.5 * fb.height
189    sz = clip[:, 2] / w
190
191    for i0, i1, i2 in mesh.faces:
192        # 簡略化のため、カメラの近くの面(ニアクリップ面)より手前に頂点が
193        # ある三角形は、切り取らずに丸ごと捨てる。
194        if min(w[i0], w[i1], w[i2]) <= 0.0 or \
195                min(sz[i0], sz[i1], sz[i2]) < -1.0:
196            continue
197        area = edge_function(sx[i0], sy[i0], sx[i1], sy[i1],
198                             np.array(sx[i2]), np.array(sy[i2]))
199        if area == 0.0:
200            continue
201        # 画面の y 軸は下向きなので、表を向いた(反時計回りの)三角形は
202        # 画面上では時計回りになり、area が負になる。
203        if cull_back and area > 0.0:
204            continue
205
206        # 三角形を囲む長方形の画素の中心の座標を作る。
207        x_min = max(int(np.floor(min(sx[i0], sx[i1], sx[i2]))), 0)
208        x_max = min(int(np.ceil(max(sx[i0], sx[i1], sx[i2]))), fb.width)
209        y_min = max(int(np.floor(min(sy[i0], sy[i1], sy[i2]))), 0)
210        y_max = min(int(np.ceil(max(sy[i0], sy[i1], sy[i2]))), fb.height)
211        if x_min >= x_max or y_min >= y_max:
212            continue
213        px, py = np.meshgrid(np.arange(x_min, x_max) + 0.5,
214                             np.arange(y_min, y_max) + 0.5)
215
216        # 重心座標を求め、三角形の内側の画素を選ぶ。
217        b0 = edge_function(sx[i1], sy[i1], sx[i2], sy[i2], px, py) / area
218        b1 = edge_function(sx[i2], sy[i2], sx[i0], sy[i0], px, py) / area
219        b2 = 1.0 - b0 - b1
220        inside = (b0 >= 0.0) & (b1 >= 0.0) & (b2 >= 0.0)
221
222        # 深度テスト: すでに書かれている深度より手前の画素だけを残す。
223        z = b0 * sz[i0] + b1 * sz[i1] + b2 * sz[i2]
224        region = (slice(y_min, y_max), slice(x_min, x_max))
225        passed = inside & (z < fb.depth[region])
226        if not passed.any():
227            continue
228        fb.depth[region][passed] = z[passed]
229
230        # 属性の補間に使う重み。透視補正では各頂点の 1 / w を掛けて
231        # 補間し、最後に重みの合計で割る。
232        if perspective_correct:
233            q0, q1, q2 = b0 / w[i0], b1 / w[i1], b2 / w[i2]
234            total = q0 + q1 + q2
235            q0, q1, q2 = q0 / total, q1 / total, q2 / total
236        else:
237            q0, q1, q2 = b0, b1, b2
238        for name, values in attributes.items():
239            value = (q0[..., None] * values[i0] + q1[..., None] * values[i1]
240                     + q2[..., None] * values[i2])
241            fb.attributes[name][region][passed] = value[passed]

244def transform_normals(model: FloatArray, normals: FloatArray) -> FloatArray:
245    """モデル変換に合わせて法線を変換する。
246
247    法線には、モデル変換の左上 3 × 3 の逆行列の転置を掛ける。
248    """
249    normal_matrix = np.linalg.inv(model[:3, :3]).T
250    result = (normal_matrix @ normals.T).T
251    return np.asarray(result / np.linalg.norm(result, axis=-1, keepdims=True))

 39def dot(a: FloatArray, b: FloatArray) -> FloatArray:
 40    """末尾の軸をベクトルとみなした内積。"""
 41    return np.asarray(np.sum(a * b, axis=-1, keepdims=True))

 44def lambert(normal: FloatArray, light_dir: FloatArray) -> FloatArray:
 45    """ランバートの拡散反射の係数 max(N・L, 0) を返す。
 46
 47    normal と light_dir(表面から光源へ向かう向き)は長さ 1 とする。
 48    """
 49    return np.maximum(dot(normal, light_dir), 0.0)

 52def blinn_phong(normal: FloatArray, light_dir: FloatArray,
 53                view_dir: FloatArray, shininess: float) -> FloatArray:
 54    """ブリン-フォンの鏡面反射の係数 max(N・H, 0) ** shininess を返す。
 55
 56    view_dir は表面から視点へ向かう長さ 1 のベクトル、H は light_dir と
 57    view_dir の中間の向き(ハーフベクトル)である。光が裏から当たる場合は
 58    0 とする。
 59    """
 60    half = normalize(light_dir + view_dir)
 61    facing = dot(normal, light_dir) > 0.0
 62    return np.where(facing, np.maximum(dot(normal, half), 0.0) ** shininess,
 63                    0.0)

 66def shade(position: FloatArray, normal: FloatArray, eye: FloatArray,
 67          light_dir: FloatArray, material: Material,
 68          terms: tuple[str, ...] = ("ambient", "diffuse", "specular")
 69          ) -> FloatArray:
 70    """平行光源で照らした表面の色(線形な値)を求める。
 71
 72    terms で、合計に含める成分を選べる。光源の色は白とする。
 73    """
 74    base = srgb_to_linear(np.array(material.color))
 75    n = normalize(normal)
 76    v = normalize(eye - position)
 77    color = np.zeros(position.shape[:-1] + (3,))
 78    if "ambient" in terms:
 79        color = color + material.ambient * base
 80    if "diffuse" in terms:
 81        color = color + lambert(n, light_dir) * base
 82    if "specular" in terms:
 83        color = color + material.specular * blinn_phong(
 84            n, light_dir, v, material.shininess)
 85    return color

 29def texel_coords(texture: FloatArray,
 30                 uv: FloatArray) -> tuple[FloatArray, FloatArray]:
 31    """テクスチャ座標を、テクセル(テクスチャの画素)単位の座標に変換する。
 32
 33    テクセル (i, j) の中心が (i, j) になるように 0.5 をずらす。画像の行は
 34    上から数えるので、v の向きを反転する。
 35    """
 36    height, width = texture.shape[:2]
 37    x = uv[..., 0] * width - 0.5
 38    y = (1.0 - uv[..., 1]) * height - 0.5
 39    return x, y

 42def fetch(texture: FloatArray, x: IntArray, y: IntArray,
 43          wrap: str = "repeat") -> FloatArray:
 44    """整数のテクセル座標の色を返す。
 45
 46    wrap が "repeat" なら範囲外の座標は繰り返し、"clamp" なら最も近い端の
 47    テクセルの色を使う。
 48    """
 49    height, width = texture.shape[:2]
 50    if wrap == "clamp":
 51        return np.asarray(texture[np.clip(y, 0, height - 1),
 52                                  np.clip(x, 0, width - 1)])
 53    return np.asarray(texture[y % height, x % width])

 56def sample_nearest(texture: FloatArray, uv: FloatArray,
 57                   wrap: str = "repeat") -> FloatArray:
 58    """最も近いテクセルの色を返す(最近傍補間)。"""
 59    x, y = texel_coords(texture, uv)
 60    return fetch(texture, np.rint(x).astype(np.int64),
 61                 np.rint(y).astype(np.int64), wrap)

 64def sample_bilinear(texture: FloatArray, uv: FloatArray,
 65                    wrap: str = "repeat") -> FloatArray:
 66    """周りの 4 つのテクセルの色を、距離に応じて混ぜる(バイリニア補間)。"""
 67    x, y = texel_coords(texture, uv)
 68    x0 = np.floor(x).astype(np.int64)
 69    y0 = np.floor(y).astype(np.int64)
 70    fx = (x - x0)[..., None]
 71    fy = (y - y0)[..., None]
 72    top = (fetch(texture, x0, y0, wrap) * (1 - fx)
 73           + fetch(texture, x0 + 1, y0, wrap) * fx)
 74    bottom = (fetch(texture, x0, y0 + 1, wrap) * (1 - fx)
 75              + fetch(texture, x0 + 1, y0 + 1, wrap) * fx)
 76    return np.asarray(top * (1 - fy) + bottom * fy)

 79def build_mipmaps(texture: FloatArray) -> list[FloatArray]:
 80    """縦横を半分ずつに縮めたテクスチャの列(ミップマップ)を作る。
 81
 82    幅と高さは 2 のべき乗とする。縮小は 2 × 2 のテクセルの平均で行い、
 83    1 × 1 になるまで繰り返す。
 84    """
 85    levels = [texture]
 86    while levels[-1].shape[0] > 1 and levels[-1].shape[1] > 1:
 87        t = levels[-1]
 88        smaller = (t[0::2, 0::2] + t[1::2, 0::2]
 89                   + t[0::2, 1::2] + t[1::2, 1::2]) / 4.0
 90        levels.append(smaller)
 91    return levels

 94def mip_level(uv: FloatArray, texture_size: int) -> FloatArray:
 95    """画素ごとに、使うべきミップマップの段(詳細度、LOD)を求める。
 96
 97    隣の画素とのテクスチャ座標の差から、1 画素が何テクセル分に当たるかを
 98    求め、その 2 を底とする対数を段とする。GPU も、隣り合う 2 × 2 の画素の
 99    差から同じように段を決めている。
100    """
101    du_dy, du_dx = np.gradient(uv[..., 0] * texture_size)
102    dv_dy, dv_dx = np.gradient(uv[..., 1] * texture_size)
103    footprint = np.maximum(np.hypot(du_dx, dv_dx), np.hypot(du_dy, dv_dy))
104    return np.asarray(np.log2(np.maximum(footprint, 1e-8)))

107def sample_trilinear(mipmaps: list[FloatArray], uv: FloatArray,
108                     level: FloatArray) -> FloatArray:
109    """隣り合う 2 つの段をバイリニア補間で読み、段の間でも補間する。"""
110    level = np.clip(level, 0.0, len(mipmaps) - 1)
111    lower = np.floor(level).astype(np.int64)
112    upper = np.minimum(lower + 1, len(mipmaps) - 1)
113    t = (level - lower)[..., None]
114    result = np.zeros(uv.shape[:-1] + (3,))
115    for k, texture in enumerate(mipmaps):
116        use_lower = lower == k
117        use_upper = upper == k
118        if not (use_lower.any() or use_upper.any()):
119            continue
120        color = sample_bilinear(texture, uv)
121        result += np.where(use_lower[..., None], color * (1 - t), 0.0)
122        result += np.where(use_upper[..., None], color * t, 0.0)
123    return result

 37@dataclass
 38class Sphere:
 39    """球。"""
 40
 41    center: FloatArray
 42    radius: float
 43    surface: Surface
 44
 45    def intersect(self, origins: FloatArray, dirs: FloatArray) -> FloatArray:
 46        """各レイが球と最初に交わる距離 t を返す。交わらなければ無限大。
 47
 48        レイ上の点 o + t d を球の式 |p - c|^2 = r^2 に代入した、t の 2 次
 49        方程式を解く。d は長さ 1 なので、t^2 の係数は 1 になる。
 50        """
 51        oc = origins - self.center
 52        b = np.sum(oc * dirs, axis=-1)
 53        c = np.sum(oc * oc, axis=-1) - self.radius ** 2
 54        disc = b * b - c
 55        sqrt_disc = np.sqrt(np.maximum(disc, 0.0))
 56        t_near = -b - sqrt_disc
 57        t_far = -b + sqrt_disc
 58        t = np.where(t_near > EPSILON, t_near, t_far)
 59        return np.where((disc >= 0.0) & (t > EPSILON), t, np.inf)
 60
 61    def normal_at(self, points: FloatArray) -> FloatArray:
 62        """球の表面の点 points での法線を返す。"""
 63        return normalize(points - self.center)

 67class Plane:
 68    """点 point を通り、法線が normal の無限に広い平面。"""
 69
 70    point: FloatArray
 71    normal: FloatArray
 72    surface: Surface
 73
 74    def intersect(self, origins: FloatArray, dirs: FloatArray) -> FloatArray:
 75        """各レイが平面と交わる距離 t を返す。交わらなければ無限大。"""
 76        denom = dirs @ self.normal
 77        with np.errstate(divide="ignore", invalid="ignore"):
 78            t = ((self.point - origins) @ self.normal) / denom
 79        return np.where((np.abs(denom) > 1e-9) & (t > EPSILON), t, np.inf)

 86@dataclass
 87class Triangle:
 88    """三角形。頂点は外側から見て反時計回りに並べる。"""
 89
 90    v0: FloatArray
 91    v1: FloatArray
 92    v2: FloatArray
 93    surface: Surface
 94    normal: FloatArray = field(init=False)
 95
 96    def __post_init__(self) -> None:
 97        self.normal = normalize(np.cross(self.v1 - self.v0,
 98                                         self.v2 - self.v0))
 99
100    def intersect(self, origins: FloatArray, dirs: FloatArray) -> FloatArray:
101        """メラー-トランボアの方法で、三角形と交わる距離 t を返す。
102
103        交点を o + t d = v0 + u (v1 - v0) + v (v2 - v0) と表し、t、u、v の
104        連立 1 次方程式をクラメルの公式で解く。u >= 0、v >= 0、u + v <= 1
105        なら交点は三角形の内側にある。
106        """
107        e1 = self.v1 - self.v0
108        e2 = self.v2 - self.v0
109        p = np.cross(dirs, e2)
110        det = p @ e1
111        with np.errstate(divide="ignore", invalid="ignore"):
112            inv_det = 1.0 / det
113            s = origins - self.v0
114            u = np.sum(s * p, axis=-1) * inv_det
115            q = np.cross(s, e1)
116            v = np.sum(dirs * q, axis=-1) * inv_det
117            t = (q @ e2) * inv_det
118        hit = ((np.abs(det) > 1e-9) & (u >= 0.0) & (v >= 0.0)
119               & (u + v <= 1.0) & (t > EPSILON))
120        return np.where(hit, t, np.inf)
121
122    def normal_at(self, points: FloatArray) -> FloatArray:
123        """三角形の法線を、点の数だけ並べて返す。"""
124        return np.asarray(np.broadcast_to(self.normal, points.shape))

144def camera_rays(width: int, height: int, eye: FloatArray,
145                target: FloatArray, fovy_degrees: float,
146                jitter: FloatArray | None = None
147                ) -> tuple[FloatArray, FloatArray]:
148    """カメラから各画素の中心を通るレイの始点と向きを作る。
149
150    jitter に (高さ × 幅, 2) の 0 から 1 の値を渡すと、画素の中心ではなく
151    画素の中のその位置を通るレイを作る。
152    """
153    forward = normalize(target - eye)
154    right = normalize(np.cross(forward, np.array([0.0, 1.0, 0.0])))
155    up = np.cross(right, forward)
156    half_h = np.tan(np.radians(fovy_degrees) / 2.0)
157    half_w = half_h * width / height
158
159    ys, xs = np.mgrid[0:height, 0:width]
160    offset = np.full((height * width, 2), 0.5) if jitter is None else jitter
161    px = xs.ravel() + offset[:, 0]
162    py = ys.ravel() + offset[:, 1]
163    # 画素の座標を、カメラの前方 1 の距離にあるスクリーン上の座標に変換する。
164    sx = (2.0 * px / width - 1.0) * half_w
165    sy = (1.0 - 2.0 * py / height) * half_h
166    dirs = normalize(forward + sx[:, None] * right + sy[:, None] * up)
167    origins = np.broadcast_to(eye, dirs.shape).copy()
168    return origins, dirs

171def closest_hit(scene: list[Shape], origins: FloatArray,
172                dirs: FloatArray) -> tuple[FloatArray, FloatArray]:
173    """各レイが最初に当たる物体の番号と距離を返す。
174
175    どの物体にも当たらないレイは、番号が -1、距離が無限大になる。
176    """
177    distances = np.stack([shape.intersect(origins, dirs) for shape in scene])

210def trace(scene: list[Shape], origins: FloatArray, dirs: FloatArray,
211          depth: int, shadows: bool) -> FloatArray:
212    """レイの色(線形な値)を求める。
213
214    鏡面反射する物体に当たったレイは、depth が 0 になるまで反射の向きに
215    レイを飛ばし直し、その色を混ぜる(再帰的なレイトレーシング)。
216    """
217    color = sky(dirs)
218    index, t = closest_hit(scene, origins, dirs)
219    for k, shape in enumerate(scene):
220        hit = index == k
221        if not hit.any():
222            continue
223        o, d = origins[hit], dirs[hit]
224        points = o + t[hit, None] * d
225        normals = shape.normal_at(points)
226        albedo = base_color(shape, points)
227
228        # 影: 光源の方向がさえぎられていれば、直接光を当てない。
229        light = np.ones((len(points), 1))
230        if shadows:
231            light[in_shadow(scene, points, LIGHT_DIR)] = 0.0
232        diffuse = lambert(normals, LIGHT_DIR)
233        specular = blinn_phong(normals, LIGHT_DIR, -d, 64.0)
234        local = (albedo * (0.12 + 0.88 * diffuse * light)
235                 + 0.4 * specular * light)
236
237        r = shape.surface.reflectivity
238        if r > 0.0 and depth > 0:
239            reflected = trace(scene, points, reflect(d, normals),
240                              depth - 1, shadows)
241            local = (1.0 - r) * local + r * reflected
242        color[hit] = local
243    return color

283def cosine_sample(normals: FloatArray, rng: np.random.Generator) -> FloatArray:
284    """法線のまわりの半球から、cos に比例する確率で向きを選ぶ。
285
286    単位円板上に一様に点を取り、半球に持ち上げる方法を使う。
287    """
288    n = len(normals)
289    r = np.sqrt(rng.random(n))
290    phi = 2.0 * np.pi * rng.random(n)
291    x, y = r * np.cos(phi), r * np.sin(phi)
292    z = np.sqrt(np.maximum(0.0, 1.0 - x * x - y * y))
293    # 法線を z 軸とする正規直交基底を作り、(x, y, z) を変換する。
294    helper = np.where(np.abs(normals[:, :1]) > 0.9,
295                      np.array([0.0, 1.0, 0.0]), np.array([1.0, 0.0, 0.0]))
296    tangent = normalize(np.cross(helper, normals))
297    bitangent = np.cross(normals, tangent)
298    return np.asarray(x[:, None] * tangent + y[:, None] * bitangent
299                      + z[:, None] * normals)

311def path_trace(scene: list[Shape], origins: FloatArray, dirs: FloatArray,
312               rng: np.random.Generator, bounces: int = 4) -> FloatArray:
313    """拡散反射する物体だけの場面で、空を光源としてレイの色を推定する。
314
315    物体に当たるたびに、反射する向きを乱数で 1 つ選んでレイを続ける。
316    cos に比例する確率で向きを選ぶと、ランバート反射では「これまでの
317    反射率の積(throughput)」を掛けていくだけで済む。
318    """
319    color = np.zeros_like(dirs)
320    throughput = np.ones_like(dirs)
321    alive = np.arange(len(dirs))
322    for _ in range(bounces + 1):
323        index, t = closest_hit(scene, origins, dirs)
324        missed = index < 0
325        # 空に抜けたレイは、空の明るさを持ち帰って終わる。
326        color[alive[missed]] = throughput[missed] * dome(dirs[missed])
327        keep = ~missed
328        alive, index, t = alive[keep], index[keep], t[keep]
329        origins, dirs = origins[keep], dirs[keep]
330        throughput = throughput[keep]
331        if len(alive) == 0:
332            break
333        points = origins + t[:, None] * dirs
334        normals = np.zeros_like(points)
335        albedo = np.zeros_like(points)
336        for k, shape in enumerate(scene):
337            hit = index == k
338            if hit.any():
339                normals[hit] = shape.normal_at(points[hit])
340                albedo[hit] = base_color(shape, points[hit])
341        # 裏側から当たった場合は、法線を反転して表側として扱う。
342        normals = np.where(dot(normals, dirs) > 0.0, -normals, normals)
343        throughput = throughput * albedo
344        origins = points
345        dirs = cosine_sample(normals, rng)
346    return color

349def diffuse_scene() -> list[Shape]:
350    """パストレーシング用の、拡散反射だけの物体を並べた場面。"""
351    return [

353              Surface((0.8, 0.8, 0.8))),
354        Sphere(np.array([-0.55, 0.5, 0.0]), 0.5, Surface((0.9, 0.2, 0.15))),
355        Sphere(np.array([0.55, 0.5, -0.2]), 0.5, Surface((0.85, 0.85, 0.85))),
356    ]
