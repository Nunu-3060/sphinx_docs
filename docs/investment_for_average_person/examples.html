サンプルコード

本書で紹介した試算を、自分の条件で試すための Python のスクリプトを掲載する。各ファイルは、見出しの下のリンクからダウンロードできる。

試算の結果は、毎年一定の利回りで運用できたと仮定した場合の目安である。実際の運用成果は毎年大きく上下するため、将来の成果を示すものではない。

実行の方法

すべてのスクリプトは Python 3.10 以降の標準ライブラリだけで動作し、追加のパッケージは不要である。ダウンロードしたフォルダで、次のように実行する。

1python accumulation_simulation.py
2python accumulation_simulation.py --monthly 20000 --rate 4 --years 25

--help を付けて実行すると、指定できる条件の一覧が表示される。

cost_comparison.py は accumulation_simulation.py の関数を使うため、2 つのファイルを同じフォルダに置いて実行する。

積立投資の試算

ダウンロード：accumulation_simulation.py

毎月の積立額、コストを引く前の年率リターン、年率コスト、積立期間を指定すると、各年末の元本・残高・運用益を表示する。既定の条件（毎月 3 万円、年 5 %、コスト年 0.1 %、30 年）で実行した結果は次のとおりである。

1毎月 30,000 円、年率 5.0 % (コスト 0.1 %)、30 年間の試算
2   年      元本 (円)      残高 (円)    運用益 (円)
3   5      1,800,000      2,037,564        237,564
4  10      3,600,000      4,625,710      1,025,710
5  15      5,400,000      7,913,214      2,513,214
6  20      7,200,000     12,089,052      4,889,052
7  25      9,000,000     17,393,267      8,393,267
8  30     10,800,000     24,130,764     13,330,764

 1"""毎月の積立投資の結果を試算する。
 2
 3毎月同じ金額を積み立て、一定の利回りで運用できたと仮定したときの、
 4各年末の元本・残高・運用益を表にして表示する。
 5
 6実際の運用成果は毎年大きく上下する。この試算は「平均してこの利回りだった場合」の目安であり、
 7将来の成果を約束するものではない。
 8
 9使い方の例:
10    python accumulation_simulation.py
11    python accumulation_simulation.py --monthly 30000 --rate 4 --years 20 --cost 0.2
12"""
13
14import argparse
15from dataclasses import dataclass
16
17
18@dataclass(frozen=True)
19class YearResult:
20    """ある年の年末時点の状態。"""
21
22    year: int
23    principal: float
24    balance: float
25
26    @property
27    def gain(self) -> float:
28        """運用益 (残高から元本を引いた額)。"""
29        return self.balance - self.principal
30
31
32def simulate(monthly: float, annual_rate: float, years: int, annual_cost: float = 0.0) -> list[YearResult]:
33    """積立投資を試算し、各年末の結果を返す。
34
35    Args:
36        monthly: 毎月の積立額 (円)。
37        annual_rate: コストを引く前の年率リターン (例: 年 5 % なら 0.05)。
38        years: 積立期間 (年)。
39        annual_cost: 信託報酬などの年率コスト (例: 年 0.1 % なら 0.001)。
40
41    Returns:
42        1 年目から years 年目までの年末の結果。
43    """
44    # 年率を月率に直す (月ごとに複利で増えると考える)
45    monthly_rate = (1.0 + annual_rate - annual_cost) ** (1.0 / 12.0) - 1.0
46    balance = 0.0
47    results: list[YearResult] = []
48    for year in range(1, years + 1):
49        for _ in range(12):
50            # 月初に積み立て、その月の運用成果を加える
51            balance = (balance + monthly) * (1.0 + monthly_rate)
52        results.append(YearResult(year, monthly * 12 * year, balance))
53    return results
54
55
56def print_table(results: list[YearResult], step: int) -> None:
57    """結果を表として表示する。step 年ごとと最終年を表示する。"""
58    print(f"{'年':>4} {'元本 (円)':>14} {'残高 (円)':>14} {'運用益 (円)':>14}")
59    for result in results:
60        if result.year % step == 0 or result.year == len(results):
61            print(f"{result.year:>4} {result.principal:>14,.0f} {result.balance:>14,.0f} {result.gain:>14,.0f}")
62
63
64def main() -> None:
65    """コマンドライン引数を読み取り、試算結果を表示する。"""
66    parser = argparse.ArgumentParser(description="毎月の積立投資の結果を試算する。")
67    parser.add_argument("--monthly", type=float, default=30000, help="毎月の積立額 (円)。既定値は 30000")
68    parser.add_argument("--rate", type=float, default=5.0, help="コストを引く前の年率リターン (%%)。既定値は 5")
69    parser.add_argument("--cost", type=float, default=0.1, help="信託報酬などの年率コスト (%%)。既定値は 0.1")
70    parser.add_argument("--years", type=int, default=30, help="積立期間 (年)。既定値は 30")
71    parser.add_argument("--step", type=int, default=5, help="何年ごとに表示するか。既定値は 5")
72    args = parser.parse_args()
73
74    results = simulate(args.monthly, args.rate / 100.0, args.years, args.cost / 100.0)
75    print(f"毎月 {args.monthly:,.0f} 円、年率 {args.rate} % (コスト {args.cost} %)、{args.years} 年間の試算")
76    print_table(results, args.step)
77
78
79if __name__ == "__main__":
80    main()

運用コストの比較

ダウンロード：cost_comparison.py

積立の条件を同じにして、年率コストだけを変えたときの最終残高を比べる。--costs に比べたいコスト（%）を空白で区切って指定する。

 1"""運用コスト (信託報酬など) の違いが最終的な残高に与える影響を比べる。
 2
 3同じ金額を同じ期間積み立て、コストを引く前の利回りも同じだと仮定し、
 4年率コストだけを変えたときの最終残高を比べる。
 5
 6使い方の例:
 7    python cost_comparison.py
 8    python cost_comparison.py --monthly 50000 --years 20 --costs 0.1 0.5 1.0 2.0
 9"""
10
11import argparse
12
13from accumulation_simulation import simulate
14
15
16def compare(monthly: float, annual_rate: float, years: int, costs: list[float]) -> list[tuple[float, float]]:
17    """コストごとの最終残高を計算する。
18
19    Args:
20        monthly: 毎月の積立額 (円)。
21        annual_rate: コストを引く前の年率リターン (例: 年 5 % なら 0.05)。
22        years: 積立期間 (年)。
23        costs: 比べる年率コストのリスト (例: 年 0.1 % なら 0.001)。
24
25    Returns:
26        (年率コスト, 最終残高) の組のリスト。
27    """
28    return [(cost, simulate(monthly, annual_rate, years, cost)[-1].balance) for cost in costs]
29
30
31def main() -> None:
32    """コマンドライン引数を読み取り、比較結果を表示する。"""
33    parser = argparse.ArgumentParser(description="運用コストの違いによる最終残高の差を比べる。")
34    parser.add_argument("--monthly", type=float, default=30000, help="毎月の積立額 (円)。既定値は 30000")
35    parser.add_argument("--rate", type=float, default=5.0, help="コストを引く前の年率リターン (%%)。既定値は 5")
36    parser.add_argument("--years", type=int, default=30, help="積立期間 (年)。既定値は 30")
37    parser.add_argument("--costs", type=float, nargs="+", default=[0.1, 0.8, 1.5],
38                        help="比べる年率コスト (%%)。複数指定できる。既定値は 0.1 0.8 1.5")
39    args = parser.parse_args()
40
41    costs = [cost / 100.0 for cost in args.costs]
42    results = compare(args.monthly, args.rate / 100.0, args.years, costs)
43    lowest = results[0][1] if results else 0.0
44    print(f"毎月 {args.monthly:,.0f} 円、コスト控除前の年率 {args.rate} %、{args.years} 年間の比較")
45    print(f"{'年率コスト':>10} {'最終残高 (円)':>16} {'最初の行との差 (円)':>20}")
46    for cost, balance in results:
47        print(f"{cost * 100:>9.2f}% {balance:>16,.0f} {balance - lowest:>20,.0f}")
48
49
50if __name__ == "__main__":
51    main()

取り崩しの比較

ダウンロード：withdrawal_simulation.py

取り崩し開始時の資産、運用の年率リターン、定額方式で毎年引き出す額、定率方式で毎年引き出す割合を指定すると、2 つの方式の引き出し額と残高を並べて表示する。

 1"""資産を運用しながら取り崩す場合の、定額方式と定率方式を比べる。
 2
 3* 定額方式: 毎年同じ金額を引き出す。受け取る額は安定するが、資産が尽きることがある。
 4* 定率方式: 毎年、その時点の残高の一定割合を引き出す。資産は尽きにくいが、受け取る額は変動する。
 5
 6毎年、年初に引き出し、残りを一定の利回りで 1 年間運用したと仮定する。
 7実際の運用成果は毎年大きく上下するため、この試算は目安である。
 8
 9使い方の例:
10    python withdrawal_simulation.py
11    python withdrawal_simulation.py --assets 15000000 --rate 2 --amount 900000 --ratio 4 --years 35
12"""
13
14import argparse
15from dataclasses import dataclass
16
17
18@dataclass(frozen=True)
19class YearRecord:
20    """ある年の引き出し額と、年末の残高。"""
21
22    year: int
23    withdrawal: float
24    balance: float
25
26
27def fixed_amount(assets: float, annual_rate: float, amount: float, years: int) -> list[YearRecord]:
28    """定額方式で取り崩したときの各年の記録を返す。残高が足りない年は残高すべてを引き出す。"""
29    balance = assets
30    records: list[YearRecord] = []
31    for year in range(1, years + 1):
32        withdrawal = min(amount, balance)
33        balance = (balance - withdrawal) * (1.0 + annual_rate)
34        records.append(YearRecord(year, withdrawal, balance))
35    return records
36
37
38def fixed_ratio(assets: float, annual_rate: float, ratio: float, years: int) -> list[YearRecord]:
39    """定率方式で取り崩したときの各年の記録を返す。"""
40    balance = assets
41    records: list[YearRecord] = []
42    for year in range(1, years + 1):
43        withdrawal = balance * ratio
44        balance = (balance - withdrawal) * (1.0 + annual_rate)
45        records.append(YearRecord(year, withdrawal, balance))
46    return records
47
48
49def main() -> None:
50    """コマンドライン引数を読み取り、2 つの方式の結果を並べて表示する。"""
51    parser = argparse.ArgumentParser(description="定額方式と定率方式の取り崩しを比べる。")
52    parser.add_argument("--assets", type=float, default=20_000_000, help="取り崩し開始時の資産 (円)。既定値は 20000000")
53    parser.add_argument("--rate", type=float, default=3.0, help="運用の年率リターン (%%)。既定値は 3")
54    parser.add_argument("--amount", type=float, default=1_000_000, help="定額方式で毎年引き出す額 (円)。既定値は 1000000")
55    parser.add_argument("--ratio", type=float, default=5.0, help="定率方式で毎年引き出す割合 (%%)。既定値は 5")
56    parser.add_argument("--years", type=int, default=30, help="試算する年数。既定値は 30")
57    parser.add_argument("--step", type=int, default=5, help="何年ごとに表示するか。既定値は 5")
58    args = parser.parse_args()
59
60    rate = args.rate / 100.0
61    amount_records = fixed_amount(args.assets, rate, args.amount, args.years)
62    ratio_records = fixed_ratio(args.assets, rate, args.ratio / 100.0, args.years)
63
64    print(f"資産 {args.assets:,.0f} 円、年率 {args.rate} % で運用しながら取り崩す試算")
65    print(f"定額方式: 毎年 {args.amount:,.0f} 円 / 定率方式: 毎年 残高の {args.ratio} %")
66    print(f"{'年':>4} {'定額 引出 (円)':>14} {'定額 残高 (円)':>14} {'定率 引出 (円)':>14} {'定率 残高 (円)':>14}")
67    for amount, ratio in zip(amount_records, ratio_records):
68        if amount.year % args.step == 0 or amount.year == 1 or amount.year == args.years:
69            print(f"{amount.year:>4} {amount.withdrawal:>14,.0f} {amount.balance:>14,.0f} "
70                  f"{ratio.withdrawal:>14,.0f} {ratio.balance:>14,.0f}")
71
72    depleted = next((record.year for record in amount_records if record.balance <= 0.0), None)
73    if depleted is None:
74        print(f"定額方式では、{args.years} 年後も資産が残る。")
75    else:
76        print(f"定額方式では、{depleted} 年目に資産が尽きる。")
77
78
79if __name__ == "__main__":
80    main()
