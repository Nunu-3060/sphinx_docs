wav_io.py

WAV ファイルの書き出しと音量の調整を行う共通モジュール。

リスト 1 wav_io.py
 1"""WAV ファイルの書き出しを行う共通モジュール.
 2
 3本書のサンプルコードはすべて、浮動小数点数の配列（値の範囲は -1.0 から 1.0）
 4として音を作り、最後にこのモジュールで 16 bit の WAV ファイルに変換する。
 5"""
 6
 7import wave
 8from pathlib import Path
 9
10import numpy as np
11import numpy.typing as npt
12
13# 1 秒あたりのサンプル数（サンプリング周波数）
14SAMPLE_RATE = 44100
15
16# 浮動小数点数の音声データを表す型
17Signal = npt.NDArray[np.float64]
18
19
20def seconds_to_samples(seconds: float) -> int:
21    """秒数をサンプル数に変換する."""
22    return int(round(seconds * SAMPLE_RATE))
23
24
25def silence(seconds: float) -> Signal:
26    """指定した長さの無音を返す."""
27    return np.zeros(seconds_to_samples(seconds), dtype=np.float64)
28
29
30def normalize(signal: Signal, peak: float = 0.9) -> Signal:
31    """最大振幅が peak になるように音量を揃える."""
32    max_amplitude = float(np.max(np.abs(signal))) if signal.size > 0 else 0.0
33    if max_amplitude == 0.0:
34        return signal
35    return signal * (peak / max_amplitude)
36
37
38def write_wav(path: Path, signal: Signal, sample_rate: int = SAMPLE_RATE) -> None:
39    """モノラル 16 bit の WAV ファイルとして書き出す.
40
41    範囲外の値は -1.0 から 1.0 に切り詰める（クリッピング）。
42    """
43    path.parent.mkdir(parents=True, exist_ok=True)
44    clipped = np.clip(signal, -1.0, 1.0)
45    pcm = (clipped * 32767.0).astype("<i2")  # リトルエンディアンの 16 bit 整数
46    with wave.open(str(path), "wb") as wav_file:
47        wav_file.setnchannels(1)  # モノラル
48        wav_file.setsampwidth(2)  # 1 サンプルあたり 2 バイト = 16 bit
49        wav_file.setframerate(sample_rate)
50        wav_file.writeframes(pcm.tobytes())
51
52
53def main() -> None:
54    """動作確認として 440 Hz の正弦波を 1 秒間書き出す."""
55    t = np.arange(SAMPLE_RATE) / SAMPLE_RATE
56    signal = 0.5 * np.sin(2.0 * np.pi * 440.0 * t)
57    write_wav(Path("output") / "sine_440.wav", signal)
58
59
60if __name__ == "__main__":
61    main()

waveforms.py

矩形波・三角波・ノイズの生成、音名と周波数の変換。

リスト 2 waveforms.py
  1"""ファミコン（NES）の音源 APU を手本にした基本波形の生成.
  2
  3矩形波（パルス波）、三角波、ノイズの 3 種類を作る。
  4実機の完全な再現ではなく、音色を決める本質的な特徴だけを取り入れている。
  5"""
  6
  7from pathlib import Path
  8
  9import numpy as np
 10import numpy.typing as npt
 11
 12from wav_io import SAMPLE_RATE, Signal, normalize, seconds_to_samples, write_wav
 13
 14# NTSC 版ファミコンの CPU クロック周波数 [Hz]
 15CPU_CLOCK = 1_789_773.0
 16
 17# 三角波チャンネルが 1 周期の間に出力する 32 段階の値（4 bit: 0 から 15）
 18TRIANGLE_SEQUENCE = np.array(list(range(15, -1, -1)) + list(range(16)), dtype=np.float64)
 19
 20# ノイズチャンネルのタイマー周期（CPU クロック数）。インデックス 0 が最も高い音
 21NOISE_PERIODS = (4, 8, 16, 32, 64, 96, 128, 160, 202, 254, 380, 508, 762, 1016, 2034, 4068)
 22
 23NOTE_OFFSETS = {"C": 0, "D": 2, "E": 4, "F": 5, "G": 7, "A": 9, "B": 11}
 24
 25
 26def note_number(name: str) -> int:
 27    """音名（例: "A4"、"C#5"、"Bb3"）を MIDI ノート番号に変換する.
 28
 29    C4（中央のド）が 60、A4 が 69 になる。
 30    """
 31    letter = name[0].upper()
 32    rest = name[1:]
 33    accidental = 0
 34    while rest and rest[0] in "#b":
 35        accidental += 1 if rest[0] == "#" else -1
 36        rest = rest[1:]
 37    octave = int(rest)
 38    return 12 * (octave + 1) + NOTE_OFFSETS[letter] + accidental
 39
 40
 41def note_frequency(note: int) -> float:
 42    """MIDI ノート番号を周波数 [Hz] に変換する（12 平均律、A4 = 440 Hz）."""
 43    return float(440.0 * 2.0 ** ((note - 69) / 12.0))
 44
 45
 46def pulse_timer_frequency(frequency: float) -> float:
 47    """矩形波チャンネルが実際に出せる周波数のうち、最も近いものを返す.
 48
 49    実機は 11 bit のタイマー値 t を使い、f = CPU_CLOCK / (16 * (t + 1)) の
 50    周波数しか出せない。t が 8 未満の場合は消音されるため、範囲を制限している。
 51    """
 52    timer = int(round(CPU_CLOCK / (16.0 * frequency) - 1.0))
 53    timer = min(max(timer, 8), 2047)
 54    return CPU_CLOCK / (16.0 * (timer + 1))
 55
 56
 57def phase_from_frequency(frequency: float | Signal, num_samples: int) -> Signal:
 58    """周波数（定数または時間変化する配列）から 0 以上 1 未満の位相を作る.
 59
 60    位相はサンプルごとに frequency / SAMPLE_RATE ずつ進む。
 61    周波数を配列で与えれば、音程が滑らかに変化する音を作れる。
 62    """
 63    freq = np.broadcast_to(np.asarray(frequency, dtype=np.float64), (num_samples,))
 64    increments = freq / SAMPLE_RATE
 65    phase = np.cumsum(increments) - increments  # 最初のサンプルの位相を 0 にする
 66    return np.mod(phase, 1.0)
 67
 68
 69def pulse_wave(phase: Signal, duty: float | Signal = 0.5) -> Signal:
 70    """矩形波（パルス波）を作る. duty は 1 周期のうち高い値になっている割合.
 71
 72    実機と同じく 1 と 0 の 2 値を作り、平均値 duty を引いて直流成分を取り除く。
 73    """
 74    return np.where(phase < duty, 1.0, 0.0) - duty
 75
 76
 77def triangle_wave(phase: Signal) -> Signal:
 78    """4 bit（16 段階）の階段状の三角波を作る."""
 79    step = np.floor(phase * 32.0).astype(np.int64)
 80    return TRIANGLE_SEQUENCE[step] / 7.5 - 1.0
 81
 82
 83def sine_wave(phase: Signal) -> Signal:
 84    """比較用の正弦波を作る."""
 85    return np.sin(2.0 * np.pi * phase)
 86
 87
 88def lfsr_sequence(short_mode: bool = False) -> npt.NDArray[np.int64]:
 89    """15 bit の線形帰還シフトレジスタ（LFSR）が出力するビット列を 1 周期分返す.
 90
 91    通常モードでは bit 0 と bit 1、短周期モードでは bit 0 と bit 6 の
 92    排他的論理和を最上位（bit 14）に戻しながら右シフトする。
 93    """
 94    tap = 6 if short_mode else 1
 95    register = 1
 96    initial = register
 97    bits: list[int] = []
 98    while True:
 99        bits.append(register & 1)
100        feedback = (register & 1) ^ ((register >> tap) & 1)
101        register = (register >> 1) | (feedback << 14)
102        if register == initial:
103            break
104    return np.array(bits, dtype=np.int64)
105
106
107def noise_from_rate(clock_rate: float | Signal, num_samples: int, short_mode: bool = False) -> Signal:
108    """LFSR を毎秒 clock_rate 回進めたときのノイズを作る.
109
110    bit 0 が 0 のとき +1、1 のとき -1 を出力する。
111    """
112    rate = np.broadcast_to(np.asarray(clock_rate, dtype=np.float64), (num_samples,))
113    sequence = lfsr_sequence(short_mode)
114    position = np.floor(np.cumsum(rate / SAMPLE_RATE) - rate / SAMPLE_RATE).astype(np.int64)
115    bits = sequence[position % sequence.size]
116    return np.where(bits == 0, 1.0, -1.0)
117
118
119def noise_wave(seconds: float, period_index: int, short_mode: bool = False) -> Signal:
120    """ノイズチャンネルの周期インデックス（0 から 15）を指定してノイズを作る."""
121    clock_rate = CPU_CLOCK / NOISE_PERIODS[period_index]
122    return noise_from_rate(clock_rate, seconds_to_samples(seconds), short_mode)
123
124
125def fade_edges(signal: Signal, seconds: float = 0.005) -> Signal:
126    """音の始まりと終わりを短く滑らかにし、プチッというクリック音を防ぐ."""
127    length = min(seconds_to_samples(seconds), signal.size // 2)
128    if length == 0:
129        return signal
130    ramp = np.linspace(0.0, 1.0, length)
131    result = signal.copy()
132    result[:length] *= ramp
133    result[-length:] *= ramp[::-1]
134    return result
135
136
137def tone(frequency: float, seconds: float, wave: str = "pulse", duty: float = 0.5) -> Signal:
138    """一定の音程の音を作る. wave には "pulse"、"triangle"、"sine" を指定する."""
139    phase = phase_from_frequency(frequency, seconds_to_samples(seconds))
140    if wave == "pulse":
141        return pulse_wave(phase, duty)
142    if wave == "triangle":
143        return triangle_wave(phase)
144    if wave == "sine":
145        return sine_wave(phase)
146    raise ValueError(f"unknown wave: {wave}")
147
148
149def generate(out_dir: Path) -> list[Path]:
150    """本文で使う波形のサンプル音を書き出す."""
151    a4 = note_frequency(note_number("A4"))
152    sounds: dict[str, Signal] = {
153        "sine_a4": tone(a4, 1.0, "sine"),
154        "pulse_12_5_a4": tone(a4, 1.0, "pulse", 0.125),
155        "pulse_25_a4": tone(a4, 1.0, "pulse", 0.25),
156        "pulse_50_a4": tone(a4, 1.0, "pulse", 0.5),
157        "triangle_a4": tone(a4, 1.0, "triangle"),
158        "noise_long": noise_wave(1.0, 8),
159        "noise_short": noise_wave(1.0, 8, short_mode=True),
160    }
161    # ノイズの周期インデックスを 0 から 15 まで順に変えて鳴らす
162    sweep = [noise_wave(0.25, index) for index in range(len(NOISE_PERIODS))]
163    sounds["noise_periods"] = np.concatenate([fade_edges(part) for part in sweep])
164
165    paths: list[Path] = []
166    for name, signal in sounds.items():
167        path = out_dir / f"{name}.wav"
168        write_wav(path, normalize(fade_edges(signal), peak=0.4))
169        paths.append(path)
170    return paths
171
172
173if __name__ == "__main__":
174    for written in generate(Path("output")):
175        print(written)

envelope.py

エンベロープ、ピッチスイープ、ビブラート、アルペジオ、疑似エコー。

リスト 3 envelope.py
  1"""音量や音程の時間変化（エンベロープ、スイープ、ビブラート、アルペジオ）.
  2
  3どの関数も「1 サンプルごとの値を並べた配列」を返す。
  4音量の配列は波形に掛け算し、周波数の配列は phase_from_frequency に渡して使う。
  5"""
  6
  7from pathlib import Path
  8
  9import numpy as np
 10
 11from waveforms import fade_edges, note_frequency, note_number, phase_from_frequency, pulse_wave, triangle_wave
 12from wav_io import SAMPLE_RATE, Signal, normalize, seconds_to_samples, write_wav
 13
 14# ファミコンのサウンドドライバが音量や音程を書き換える間隔（1 フレーム）[秒]
 15FRAME = 1.0 / 60.0
 16
 17
 18def adsr(attack: float, decay: float, sustain: float, hold: float, release: float) -> Signal:
 19    """ADSR エンベロープを作る.
 20
 21    attack 秒で 0 から 1 へ上がり、decay 秒で sustain の音量まで下がる。
 22    その音量を hold 秒保った後、release 秒で 0 に戻る。
 23    """
 24    times = np.cumsum([0.0, attack, decay, hold, release])
 25    levels = [0.0, 1.0, sustain, sustain, 0.0]
 26    t = np.arange(seconds_to_samples(float(times[-1]))) / SAMPLE_RATE
 27    return np.interp(t, times, levels)
 28
 29
 30def nes_decay(seconds: float, period: int) -> Signal:
 31    """ファミコンの矩形波・ノイズが持つ減衰エンベロープを再現する.
 32
 33    音量は 15 から始まり、毎秒 240 回のクロックが (period + 1) 回来るたびに
 34    1 ずつ下がって 0 で止まる。period は 0 から 15 の整数である。
 35    """
 36    t = np.arange(seconds_to_samples(seconds)) / SAMPLE_RATE
 37    level = 15.0 - np.floor(t * 240.0 / (period + 1))
 38    return np.maximum(level, 0.0) / 15.0
 39
 40
 41def quantize_volume(envelope: Signal, levels: int = 16) -> Signal:
 42    """音量を 4 bit（16 段階）のような粗い段階に丸める."""
 43    return np.round(envelope * (levels - 1)) / (levels - 1)
 44
 45
 46def exponential_sweep(start: float, end: float, seconds: float) -> Signal:
 47    """周波数を start から end へ指数関数的に（音程として均等に）変化させる."""
 48    return np.geomspace(start, end, seconds_to_samples(seconds))
 49
 50
 51def vibrato(frequency: float, seconds: float, depth: float, rate: float, delay: float = 0.0) -> Signal:
 52    """ビブラート（音程の周期的な揺れ）をかけた周波数を返す.
 53
 54    depth は揺れの幅 [半音]、rate は 1 秒あたりの揺れの回数である。
 55    delay 秒が経過するまでは揺らさない。
 56    """
 57    t = np.arange(seconds_to_samples(seconds)) / SAMPLE_RATE
 58    swing = depth * np.sin(2.0 * np.pi * rate * np.maximum(t - delay, 0.0))
 59    return frequency * 2.0 ** (swing / 12.0)
 60
 61
 62def arpeggio(root: int, offsets: list[int], seconds: float, step: float = FRAME) -> Signal:
 63    """root（MIDI ノート番号）から offsets 半音ずつ離れた音を step 秒ごとに切り替える.
 64
 65    1 チャンネルで和音のように聞かせる「アルペジオ」の手法である。
 66    """
 67    t = np.arange(seconds_to_samples(seconds)) / SAMPLE_RATE
 68    index = np.floor(t / step).astype(np.int64) % len(offsets)
 69    notes = root + np.array(offsets)[index]
 70    return 440.0 * 2.0 ** ((notes - 69) / 12.0)
 71
 72
 73def pseudo_echo(signal: Signal, delay: float, gain: float, repeats: int = 2) -> Signal:
 74    """遅らせて小さくした同じ音を重ね、エコーのように聞かせる."""
 75    offset = seconds_to_samples(delay)
 76    result = np.concatenate([signal, np.zeros(offset * repeats)])
 77    for n in range(1, repeats + 1):
 78        result[offset * n:offset * n + signal.size] += signal * gain ** n
 79    return result
 80
 81
 82def generate(out_dir: Path) -> list[Path]:
 83    """本文で使う音の加工のサンプル音を書き出す."""
 84    c5 = note_frequency(note_number("C5"))
 85    sounds: dict[str, Signal] = {}
 86
 87    # 音量エンベロープのない音と、ADSR をかけた音
 88    env = adsr(0.01, 0.15, 0.5, 0.4, 0.3)
 89    sounds["no_envelope"] = pulse_wave(phase_from_frequency(c5, env.size))
 90    sounds["adsr"] = sounds["no_envelope"] * env
 91
 92    # 16 段階で減衰するファミコン風の音（減衰の速さを 2 通り）
 93    for period in (2, 8):
 94        decay = nes_decay(1.0, period)
 95        sounds[f"nes_decay_{period}"] = pulse_wave(phase_from_frequency(c5, decay.size), 0.25) * decay
 96
 97    # ピッチスイープ（上昇と下降）
 98    up = exponential_sweep(200.0, 1600.0, 0.4)
 99    sounds["sweep_up"] = pulse_wave(phase_from_frequency(up, up.size)) * nes_decay(0.4, 4)
100    down = exponential_sweep(1600.0, 200.0, 0.4)
101    sounds["sweep_down"] = pulse_wave(phase_from_frequency(down, down.size)) * nes_decay(0.4, 4)
102
103    # ビブラート
104    wobble = vibrato(c5, 1.5, depth=0.3, rate=6.0, delay=0.3)
105    sounds["vibrato"] = pulse_wave(phase_from_frequency(wobble, wobble.size), 0.5)
106
107    # アルペジオ（長三和音と短三和音）
108    for name, offsets in (("major", [0, 4, 7]), ("minor", [0, 3, 7])):
109        freq = arpeggio(note_number("C5"), offsets, 1.0)
110        sounds[f"arpeggio_{name}"] = pulse_wave(phase_from_frequency(freq, freq.size), 0.5)
111
112    # 1 フレームごとにデューティ比を切り替える
113    seconds = 1.0
114    num_samples = seconds_to_samples(seconds)
115    frame_index = np.floor(np.arange(num_samples) / SAMPLE_RATE / FRAME).astype(np.int64)
116    duty = np.array([0.125, 0.25, 0.5, 0.25])[frame_index % 4]
117    sounds["duty_switch"] = pulse_wave(phase_from_frequency(c5, num_samples), duty)
118
119    # 疑似エコー
120    phrase = np.concatenate([
121        pulse_wave(phase_from_frequency(note_frequency(note_number(name)), seconds_to_samples(0.15)), 0.5)
122        * nes_decay(0.15, 1)
123        for name in ("C5", "E5", "G5", "C6")
124    ])
125    sounds["echo_off"] = np.concatenate([phrase, np.zeros(seconds_to_samples(0.4))])
126    sounds["echo_on"] = pseudo_echo(phrase, delay=0.2, gain=0.4)
127
128    # 三角波は音量を変えられないため、音を短く切って休符を挟むことで区切りを表す
129    notes = []
130    for name in ("A2", "A3", "F2", "F3", "G2", "G3", "E2", "E3"):
131        tone = triangle_wave(phase_from_frequency(note_frequency(note_number(name)), seconds_to_samples(0.18)))
132        notes.append(fade_edges(tone, 0.003))  # 音の切れ目でクリック音が出ないようにする
133        notes.append(np.zeros(seconds_to_samples(0.07)))
134    sounds["triangle_bass"] = np.concatenate(notes)
135
136    paths: list[Path] = []
137    for name, signal in sounds.items():
138        path = out_dir / f"{name}.wav"
139        write_wav(path, normalize(fade_edges(signal), peak=0.4))
140        paths.append(path)
141    return paths
142
143
144if __name__ == "__main__":
145    for written in generate(Path("output")):
146        print(written)

リスト 4 sound_effects.py
  1"""ゲームでよく使う効果音の作例.
  2
  3どの効果音も「波形」「音程の変化」「音量の変化」の 3 つの組み合わせで作っている。
  4"""
  5
  6from pathlib import Path
  7
  8import numpy as np

 10from envelope import arpeggio, exponential_sweep, nes_decay, quantize_volume, vibrato

 60def explosion() -> Signal:
 61    """爆発: ノイズの周期を徐々に長く（音を低く）しながら、ゆっくり減衰させる."""
 62    seconds = 1.0

 65    t = np.arange(num_samples) / SAMPLE_RATE

 92def tokenize(score: str) -> list[str]:
 93    """楽譜の文字列をステップごとのトークンに分割する."""
 94    return [token for token in score.replace("|", " ").split() if token]

 97def parse(score: str) -> tuple[list[Event], int]:
 98    """楽譜からイベントの一覧と全体のステップ数を取り出す."""
 99    tokens = tokenize(score)
100    events: list[Event] = []
101    current: tuple[int, str] | None = None  # 鳴っている音の (開始ステップ, 値)
102    for step, token in enumerate(tokens + ["."]):  # 末尾に休符を足して最後の音を閉じる
103        if token == "-":
104            if current is None:
105                raise ValueError(f"step {step}: '-' には伸ばす音が必要である")
106            continue
107        if current is not None:
108            events.append(Event(current[0], step - current[0], current[1]))
109            current = None
110        if token != ".":
111            current = (step, token)
112    return events, len(tokens)

115def envelope_for(instrument: Instrument, gate: float) -> Signal:
116    """gate 秒だけ鍵盤を押したときの音量変化（リリース込み）を作る."""
117    times = [0.0, instrument.attack, instrument.attack + instrument.decay]
118    levels = [0.0 if instrument.attack > 0.0 else 1.0, 1.0, instrument.sustain]
119    t = np.arange(seconds_to_samples(gate + instrument.release)) / SAMPLE_RATE
120    held = np.interp(np.minimum(t, gate), times, levels)
121    if instrument.release > 0.0:
122        fade = np.clip(1.0 - (t - gate) / instrument.release, 0.0, 1.0)
123        held = held * np.where(t > gate, fade, 1.0)
124    return held

127def render_note(instrument: Instrument, note: str, gate: float) -> Signal:
128    """音名 note を gate 秒鳴らした音を作る."""
129    envelope = envelope_for(instrument, gate)
130    frequency = note_frequency(note_number(note))
131    if instrument.vibrato_depth > 0.0:
132        seconds = envelope.size / SAMPLE_RATE
133        freq: float | Signal = vibrato(frequency, seconds, instrument.vibrato_depth,
134                                       instrument.vibrato_rate, instrument.vibrato_delay)
135    else:
136        freq = frequency
137    phase = phase_from_frequency(freq, envelope.size)
138    if instrument.wave == "pulse":
139        # 矩形波は音量が 16 段階なので、エンベロープを丸める
140        return pulse_wave(phase, instrument.duty) * quantize_volume(envelope) * instrument.volume
141    if instrument.wave == "triangle":
142        # 三角波は音量を変えられないため、エンベロープは使わず鳴らすか止めるかだけにする。
143        # ただし、音の切れ目でクリック音が出ないように、両端に 3 ミリ秒のフェードを入れる
144        return fade_edges(triangle_wave(phase), 0.003) * instrument.volume
145    raise ValueError(f"音名を鳴らせない波形である: {instrument.wave}")

148def render_drum(drum: Drum) -> Signal:
149    """打楽器 1 打分の音を作る."""
150    num_samples = seconds_to_samples(drum.length)
151    rate = CPU_CLOCK / NOISE_PERIODS[drum.period_index]
152    t = np.arange(num_samples) / SAMPLE_RATE
153    volume = quantize_volume(np.clip(1.0 - t / drum.length, 0.0, 1.0))
154    return noise_from_rate(rate, num_samples, drum.short_mode) * volume * drum.volume
155

157def render_track(track: Track, step_seconds: float, loop: bool) -> Signal:
158    """1 チャンネル分を合成する.
159
160    1 つのチャンネルは同時に 1 音しか出せないため、次の音が始まったら前の音の余韻は切る。
161    loop が True のときは、曲の終わりからはみ出した余韻を曲の先頭に重ねる。
162    """
163    events, total_steps = parse(track.score)
164    total = seconds_to_samples(total_steps * step_seconds)
165    buffer = np.zeros(total * 2)  # はみ出した余韻を受け止めるため長めに確保する
166    starts = [event.start for event in events]
167    for i, event in enumerate(events):
168        gate = event.length * step_seconds
169        if track.instrument.wave == "noise":
170            sound = render_drum(DRUMS[event.value])
171        else:
172            sound = render_note(track.instrument, event.value, gate)
173        begin = seconds_to_samples(event.start * step_seconds)
174        if i + 1 < len(starts):
175            # 次の音の開始位置で打ち切る
176            limit = seconds_to_samples(starts[i + 1] * step_seconds) - begin
177        elif loop and starts:
178            # 最後の音は、次の周回の最初の音の開始位置で打ち切る
179            limit = total + seconds_to_samples(starts[0] * step_seconds) - begin
180        else:
181            limit = sound.size
182        sound = sound[:limit]
183        buffer[begin:begin + sound.size] += sound
184    if loop:
185        result: Signal = buffer[:total].copy()
186        result += buffer[total:total * 2]  # 余韻を先頭へ折り返す
187        return result
188    used = int(np.max(np.nonzero(buffer)[0])) + 1 if np.any(buffer) else total
189    return buffer[:max(total, used)]
190
191
192def render_song(song: Song, loop: bool = True) -> Signal:
193    """全チャンネルを合成してミキシングする."""
194    tracks = [render_track(track, song.step_seconds, loop) * track.gain for track in song.tracks]
195    length = max(track.size for track in tracks)
196    mixed = np.zeros(length)
197    for track in tracks:
198        mixed[:track.size] += track
199    return normalize(mixed, peak=0.8)

33def generate(out_dir: Path) -> list[Path]:
34    """ジングルを WAV ファイルに書き出す.
35
36    Pyxel が書き出す WAV は音量が小さいため、読み込み直して音量を揃える。
37    """
38    out_dir.mkdir(parents=True, exist_ok=True)
39    sound = build_jingle()
40    path = out_dir / "pyxel_jingle.wav"
41    seconds = sound.total_sec() or 1.0
42    sound.save(str(path), seconds + 0.2)
43    with wave.open(str(path), "rb") as wav_file:
44        sample_rate = wav_file.getframerate()
45        frames = wav_file.readframes(wav_file.getnframes())
46    signal = np.frombuffer(frames, dtype="<i2").astype(np.float64) / 32768.0
47    write_wav(path, normalize(signal, peak=0.5), sample_rate)
48    return [path]

45            if event.type == pygame.QUIT:
46                running = False
47            elif event.type == pygame.KEYDOWN:

50                elif event.key == pygame.K_UP:

61            "ESC: quit",
62        ]
