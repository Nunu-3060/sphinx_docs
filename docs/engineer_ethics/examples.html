個人情報のマスキング

ダウンロード：pii_masking.py

 1"""ログに出力する文字列から個人情報をマスキングするサンプルです。
 2
 3ログは障害調査のために多くの人が閲覧し、長期間保存されます。
 4メールアドレスや電話番号をそのまま記録すると、ログ経由で個人情報が
 5漏えいするおそれがあります。このサンプルでは、logging モジュールの
 6フィルターを使い、ログに書き込む前に個人情報を伏せ字にします。
 7
 8実行方法::
 9
10    python pii_masking.py
11"""
12
13import logging
14import re
15
16# メールアドレス（例: taro.yamada@example.com）
17EMAIL_PATTERN = re.compile(r"[\w.+-]+@[\w-]+(?:\.[\w-]+)+")
18
19# ハイフン区切りの国内電話番号（例: 090-1234-5678, 03-1234-5678）
20PHONE_PATTERN = re.compile(r"\b0\d{1,4}-\d{1,4}-\d{3,4}\b")
21
22
23def mask_email(match: re.Match[str]) -> str:
24    """メールアドレスのローカル部を先頭 1 文字だけ残して伏せます。"""
25    local, domain = match.group(0).split("@", 1)
26    return f"{local[0]}***@{domain}"
27
28
29def mask_phone(match: re.Match[str]) -> str:
30    """電話番号の末尾 4 桁だけを残し、それ以外の数字を伏せます。"""
31    phone = match.group(0)
32    head, tail = phone[:-4], phone[-4:]
33    return re.sub(r"\d", "*", head) + tail
34
35
36def mask_pii(text: str) -> str:
37    """文字列に含まれるメールアドレスと電話番号をマスキングします。"""
38    text = EMAIL_PATTERN.sub(mask_email, text)
39    return PHONE_PATTERN.sub(mask_phone, text)
40
41
42class PiiMaskingFilter(logging.Filter):
43    """ログレコードのメッセージから個人情報を取り除くフィルターです。"""
44
45    def filter(self, record: logging.LogRecord) -> bool:
46        # 引数を埋め込んだ後のメッセージをマスキングし、引数は破棄します。
47        record.msg = mask_pii(record.getMessage())
48        record.args = None
49        return True
50
51
52def create_logger() -> logging.Logger:
53    """マスキング用フィルターを組み込んだロガーを作成します。"""
54    logger = logging.getLogger("pii_masking_demo")
55    logger.setLevel(logging.INFO)
56    handler = logging.StreamHandler()
57    handler.setFormatter(logging.Formatter("%(levelname)s: %(message)s"))
58    handler.addFilter(PiiMaskingFilter())
59    logger.addHandler(handler)
60    return logger
61
62
63def main() -> None:
64    """マスキングの前後でログの内容を比較します。"""
65    message = "問い合わせ受付: %s / %s"
66    email = "taro.yamada@example.com"
67    phone = "090-1234-5678"
68
69    print("マスキングなし:", message % (email, phone))
70
71    logger = create_logger()
72    logger.info(message, email, phone)
73
74
75if __name__ == "__main__":
76    main()

パスワードの安全な保存

ダウンロード：password_hashing.py

 1"""パスワードを安全に保存し、照合するサンプルです。
 2
 3パスワードを平文のまま保存したり、ソルトなしの高速なハッシュ関数
 4（MD5 や SHA-256 を 1 回だけ適用するなど）で保存したりすると、
 5データベースが漏えいしたときに多くのパスワードが復元されてしまいます。
 6このサンプルでは、標準ライブラリの hashlib.scrypt を使い、
 7利用者ごとに異なるソルトを付けて計算コストの高いハッシュ値を保存します。
 8
 9パラメーターは OWASP Password Storage Cheat Sheet の推奨値
10（N=2**17, r=8, p=1）に合わせています。
11
12実行方法::
13
14    python password_hashing.py
15"""
16
17import hashlib
18import hmac
19import secrets
20
21SCRYPT_N = 2**17  # CPU とメモリーのコスト
22SCRYPT_R = 8  # ブロックサイズ
23SCRYPT_P = 1  # 並列度
24SALT_BYTES = 16
25KEY_BYTES = 32
26
27# scrypt が使うメモリー量はおよそ 128 * N * r バイト（この設定で 128 MiB）です。
28MAX_MEMORY = 256 * 1024 * 1024
29
30
31def _derive_key(password: str, salt: bytes, n: int, r: int, p: int) -> bytes:
32    """パスワードとソルトから scrypt で鍵を導出します。"""
33    return hashlib.scrypt(
34        password.encode("utf-8"),
35        salt=salt,
36        n=n,
37        r=r,
38        p=p,
39        maxmem=MAX_MEMORY,
40        dklen=KEY_BYTES,
41    )
42
43
44def hash_password(password: str) -> str:
45    """パスワードを保存用の文字列に変換します。
46
47    戻り値は「方式$N$r$p$ソルト$ハッシュ値」の形式です。
48    パラメーターも一緒に保存しておくと、将来コストを引き上げたときにも
49    古いハッシュ値を照合できます。
50    """
51    salt = secrets.token_bytes(SALT_BYTES)
52    key = _derive_key(password, salt, SCRYPT_N, SCRYPT_R, SCRYPT_P)
53    return "$".join(
54        ["scrypt", str(SCRYPT_N), str(SCRYPT_R), str(SCRYPT_P),
55         salt.hex(), key.hex()]
56    )
57
58
59def verify_password(password: str, stored: str) -> bool:
60    """入力されたパスワードが保存済みのハッシュ値と一致するか調べます。"""
61    scheme, n, r, p, salt_hex, key_hex = stored.split("$")
62    if scheme != "scrypt":
63        raise ValueError(f"未対応の方式です: {scheme}")
64    key = _derive_key(password, bytes.fromhex(salt_hex), int(n), int(r),
65                      int(p))
66    # 比較にかかる時間から情報が漏れないよう、定数時間で比較します。
67    return hmac.compare_digest(key, bytes.fromhex(key_hex))
68
69
70def main() -> None:
71    """ハッシュ化と照合の流れを示します。"""
72    stored = hash_password("correct horse battery staple")
73    print("保存する値:", stored)
74
75    # 同じパスワードでもソルトが異なるため、保存される値は毎回変わります。
76    print("2 回目の値:", hash_password("correct horse battery staple"))
77
78    print("正しいパスワード:",
79          verify_password("correct horse battery staple", stored))
80    print("誤ったパスワード:", verify_password("password123", stored))
81
82
83if __name__ == "__main__":
84    main()

SQL インジェクションの比較

ダウンロード：sql_injection.py

 1"""SQL インジェクションの危険性と対策を比較するサンプルです。
 2
 3利用者の入力を文字列連結で SQL 文に埋め込むと、入力に含まれる
 4記号が SQL の一部として解釈され、意図しないデータが取得されます。
 5プレースホルダーを使えば、入力は常に「値」として扱われます。
 6
 7実行方法::
 8
 9    python sql_injection.py
10"""
11
12import sqlite3
13
14
15def create_database() -> sqlite3.Connection:
16    """デモ用のテーブルを持つインメモリーデータベースを作成します。"""
17    conn = sqlite3.connect(":memory:")
18    conn.execute("CREATE TABLE users (name TEXT, email TEXT)")
19    conn.executemany(
20        "INSERT INTO users VALUES (?, ?)",
21        [
22            ("alice", "alice@example.com"),
23            ("bob", "bob@example.com"),
24            ("carol", "carol@example.com"),
25        ],
26    )
27    return conn
28
29
30def find_user_unsafe(conn: sqlite3.Connection,
31                     name: str) -> list[tuple[str, str]]:
32    """【悪い例】入力を文字列連結で SQL 文に埋め込みます。"""
33    sql = f"SELECT name, email FROM users WHERE name = '{name}'"
34    return conn.execute(sql).fetchall()
35
36
37def find_user_safe(conn: sqlite3.Connection,
38                   name: str) -> list[tuple[str, str]]:
39    """【良い例】プレースホルダーを使って入力を値として渡します。"""
40    sql = "SELECT name, email FROM users WHERE name = ?"
41    return conn.execute(sql, (name,)).fetchall()
42
43
44def main() -> None:
45    """通常の入力と攻撃的な入力で、2 つの関数の結果を比較します。"""
46    conn = create_database()
47    inputs = ["alice", "' OR '1'='1"]
48
49    for name in inputs:
50        print(f"入力: {name!r}")
51        print("  悪い例:", find_user_unsafe(conn, name))
52        print("  良い例:", find_user_safe(conn, name))
53
54    conn.close()
55
56
57if __name__ == "__main__":
58    main()

公平性の指標の計算

ダウンロード：fairness_check.py

 1"""選考結果の公平性を属性別に確認するサンプルです。
 2
 3AI モデルや選考ルールが、特定の属性の人を不利に扱っていないかを
 4数値で確認します。ここでは次の 2 つの指標を計算します。
 5
 6* 選考通過率の差（人口統計学的パリティの差）
 7* 格差影響比（最も低い通過率 / 最も高い通過率）
 8
 9格差影響比が 0.8 を下回る場合は、米国の雇用選考の指針で使われる
10「4/5 ルール」に照らして、不利益な影響の疑いがあると判断します。
11指標はあくまで調査のきっかけであり、基準を満たせば公平である、
12という意味ではない点に注意してください。
13
14実行方法::
15
16    python fairness_check.py
17"""
18
19from dataclasses import dataclass
20
21FOUR_FIFTHS_THRESHOLD = 0.8
22
23
24@dataclass(frozen=True)
25class Applicant:
26    """応募者 1 人分の選考結果です。"""
27
28    group: str
29    passed: bool
30
31
32def pass_rates(applicants: list[Applicant]) -> dict[str, float]:
33    """属性ごとの選考通過率を計算します。"""
34    totals: dict[str, int] = {}
35    passes: dict[str, int] = {}
36    for applicant in applicants:
37        totals[applicant.group] = totals.get(applicant.group, 0) + 1
38        if applicant.passed:
39            passes[applicant.group] = passes.get(applicant.group, 0) + 1
40    return {group: passes.get(group, 0) / total
41            for group, total in totals.items()}
42
43
44def parity_difference(rates: dict[str, float]) -> float:
45    """最も高い通過率と最も低い通過率の差を返します。"""
46    return max(rates.values()) - min(rates.values())
47
48
49def disparate_impact(rates: dict[str, float]) -> float:
50    """最も低い通過率を最も高い通過率で割った比を返します。"""
51    highest = max(rates.values())
52    if highest == 0:
53        raise ValueError("全員が不合格のため、比を計算できません。")
54    return min(rates.values()) / highest
55
56
57def make_sample() -> list[Applicant]:
58    """属性 A は 100 人中 60 人、属性 B は 100 人中 42 人が通過したデータです。"""
59    group_a = [Applicant("A", i < 60) for i in range(100)]
60    group_b = [Applicant("B", i < 42) for i in range(100)]
61    return group_a + group_b
62
63
64def main() -> None:
65    """サンプルデータで公平性の指標を計算し、結果を表示します。"""
66    rates = pass_rates(make_sample())
67    for group, rate in sorted(rates.items()):
68        print(f"属性 {group} の通過率: {rate:.2f}")
69
70    difference = parity_difference(rates)
71    ratio = disparate_impact(rates)
72    print(f"通過率の差: {difference:.2f}")
73    print(f"格差影響比: {ratio:.2f}")
74
75    if ratio < FOUR_FIFTHS_THRESHOLD:
76        print("4/5 ルールを下回っています。原因を調査してください。")
77    else:
78        print("4/5 ルールは満たしています。他の観点からも確認してください。")
79
80
81if __name__ == "__main__":
82    main()

識別子の仮名化

ダウンロード：pseudonymize.py

 1"""HMAC を使って識別子を仮名化するサンプルです。
 2
 3分析のためにデータを渡すとき、氏名や電話番号などの識別子を
 4別の値に置き換えることを仮名化といいます。
 5
 6電話番号のように取り得る値の種類が少ない識別子は、単純なハッシュ値
 7（SHA-256 など）に置き換えても、すべての候補のハッシュ値を計算すれば
 8元の値を突き止められます。秘密鍵を使う HMAC で置き換えれば、
 9鍵を知らない人には元の値を推測できません。
10
11仮名化したデータも、鍵や他の情報と照合すれば個人を識別できるため、
12個人情報として適切に管理する必要があります。
13
14実行方法::
15
16    python pseudonymize.py
17"""
18
19import hashlib
20import hmac
21import os
22import secrets
23
24KEY_ENV_NAME = "PSEUDONYM_KEY"
25TOKEN_LENGTH = 16
26
27
28def load_key() -> bytes:
29    """環境変数から秘密鍵を読み込みます。未設定ならデモ用の鍵を生成します。
30
31    実際の運用では、鍵をデータと別の場所で管理し、
32    データの受け渡し先には渡さないようにします。
33    """
34    key_hex = os.environ.get(KEY_ENV_NAME)
35    if key_hex:
36        return bytes.fromhex(key_hex)
37    print(f"{KEY_ENV_NAME} が未設定のため、デモ用の鍵を生成します。")
38    return secrets.token_bytes(32)
39
40
41def pseudonymize(identifier: str, key: bytes) -> str:
42    """識別子を HMAC-SHA256 で仮名に置き換えます。
43
44    同じ鍵と同じ識別子からは常に同じ仮名が得られるため、
45    仮名化した後もデータどうしを突き合わせて分析できます。
46    """
47    digest = hmac.new(key, identifier.encode("utf-8"), hashlib.sha256)
48    return digest.hexdigest()[:TOKEN_LENGTH]
49
50
51def plain_hash(identifier: str) -> str:
52    """【悪い例】鍵を使わない単純なハッシュ値に置き換えます。"""
53    digest = hashlib.sha256(identifier.encode("utf-8"))
54    return digest.hexdigest()[:TOKEN_LENGTH]
55
56
57def main() -> None:
58    """単純なハッシュ値と HMAC による仮名を比較します。"""
59    key = load_key()
60    records = [
61        ("090-1234-5678", "購入金額 12,000 円"),
62        ("080-9876-5432", "購入金額 3,500 円"),
63        ("090-1234-5678", "購入金額 8,000 円"),
64    ]
65
66    for phone, detail in records:
67        print(f"{pseudonymize(phone, key)}  {detail}")
68
69    # 単純なハッシュ値は誰でも同じ計算ができるため、候補を総当たりすれば
70    # 元の値を突き止められます（ここでは候補を 100 件に絞っています）。
71    leaked = plain_hash("090-1234-5678")
72    candidates = [f"090-1234-56{i:02d}" for i in range(100)]
73    found = [c for c in candidates if plain_hash(c) == leaked]
74    print("単純なハッシュ値:", leaked)
75    print("総当たりで判明した元の値:", found)
76
77
78if __name__ == "__main__":
79    main()

ライセンスの一覧表示

ダウンロード：license_report.py

 1"""インストール済みパッケージのライセンスを一覧表示するサンプルです。
 2
 3OSS を利用するときは、ライセンスが定める義務（著作権表示の掲載、
 4ソースコードの開示など）を守る必要があります。まずは自分のプロジェクトが
 5どのライセンスのパッケージに依存しているかを把握することが第一歩です。
 6
 7パッケージのメタデータに記載されたライセンス情報は自己申告であり、
 8誤りや記載漏れがあり得ます。重要な判断をする前には、
 9各パッケージに同梱されているライセンス文書を必ず確認してください。
10
11実行方法::
12
13    python license_report.py
14"""
15
16from importlib import metadata
17
18UNKNOWN = "（記載なし）"
19
20
21def first_value(dist: metadata.Distribution, field: str) -> str | None:
22    """メタデータの指定したフィールドの最初の値を返します。"""
23    values = dist.metadata.get_all(field)
24    if not values:
25        return None
26    value = str(values[0]).strip()
27    return value or None
28
29
30def license_of(dist: metadata.Distribution) -> str:
31    """パッケージのライセンスを、信頼できる情報源から順に調べて返します。"""
32    # 1. PEP 639 で定められた SPDX 形式のライセンス表記
33    expression = first_value(dist, "License-Expression")
34    if expression:
35        return expression
36
37    # 2. 「License :: OSI Approved :: MIT License」のような分類子
38    classifiers = dist.metadata.get_all("Classifier") or []
39    licenses = [str(c).split(" :: ")[-1] for c in classifiers
40                if str(c).startswith("License ::")]
41    if licenses:
42        return ", ".join(licenses)
43
44    # 3. 自由記述の License フィールド（全文が入っていることがあるため 1 行目のみ）
45    text = first_value(dist, "License")
46    if text:
47        return text.splitlines()[0]
48
49    return UNKNOWN
50
51
52def main() -> None:
53    """パッケージ名、バージョン、ライセンスを表形式で表示します。"""
54    rows = sorted(
55        (
56            (first_value(dist, "Name") or UNKNOWN, dist.version,
57             license_of(dist))
58            for dist in metadata.distributions()
59        ),
60        key=lambda row: row[0].lower(),
61    )
62
63    print(f"{'パッケージ':<27}{'バージョン':<11}ライセンス")
64    for name, version, license_name in rows:
65        print(f"{name:<32}{version:<16}{license_name}")
66
67
68if __name__ == "__main__":
69    main()
