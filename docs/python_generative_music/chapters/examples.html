実装例

本章では、これまでに解説した内容を踏まえた、 4 つの実装例を紹介します。いずれのサンプルコードも、本文中で内容を確認できるほか、各節の末尾に掲載したリンクからファイルをダウンロードすることができます。いずれのコードも PEP 8 に準拠し、 flake8 および mypy によるチェックを通過しています。

ランダムメロディジェネレーター

もっとも基本的な例として、乱数を用いてハ長調のメジャースケールからランダムに音を選び、旋律を生成するプログラムです。各音符の長さは、 4 分音符または 8 分音符の中からランダムに選択されます。

 1"""乱数を用いてランダムなメロディを生成し、MIDI ファイルへ出力するサンプル。"""
 2
 3from __future__ import annotations
 4
 5import random
 6from pathlib import Path
 7
 8import pretty_midi
 9
10SCALE = [60, 62, 64, 65, 67, 69, 71, 72]
11DURATIONS = [0.5, 0.25]
12OUTPUT_PATH = Path(__file__).with_name("random_melody.mid")
13
14
15def generate_melody(note_count: int) -> list[tuple[int, float]]:
16    """音階とリズムの候補からランダムに選んで旋律を生成する。"""
17    melody: list[tuple[int, float]] = []
18    for _ in range(note_count):
19        pitch = random.choice(SCALE)
20        duration = random.choice(DURATIONS)
21        melody.append((pitch, duration))
22    return melody
23
24
25def melody_to_midi(melody: list[tuple[int, float]], output_path: Path) -> None:
26    """旋律のリストを MIDI ファイルへ書き出す。"""
27    midi_data = pretty_midi.PrettyMIDI()
28    instrument = pretty_midi.Instrument(program=0)
29
30    start_time = 0.0
31    for pitch, duration in melody:
32        note = pretty_midi.Note(
33            velocity=100,
34            pitch=pitch,
35            start=start_time,
36            end=start_time + duration,
37        )
38        instrument.notes.append(note)
39        start_time += duration
40
41    midi_data.instruments.append(instrument)
42    midi_data.write(str(output_path))
43
44
45def main() -> None:
46    melody = generate_melody(note_count=32)
47    melody_to_midi(melody, OUTPUT_PATH)
48    print(f"{OUTPUT_PATH.name} を出力しました。")
49
50
51if __name__ == "__main__":
52    main()

generate_melody 関数では、あらかじめ定義した音階 SCALE と、音の長さの候補 DURATIONS の中から、それぞれランダムに値を選択し、指定した数の音符からなる旋律を生成しています。 melody_to_midi 関数では、生成した旋律のリストを pretty_midi.Note オブジェクトへ変換した上で、 MIDI ファイルとして出力しています。

このサンプルコードは random_melody_generator.py からダウンロードできます。実行すると、スクリプトと同じディレクトリに random_melody.mid という MIDI ファイルが生成されます。

python random_melody_generator.py

マルコフ連鎖メロディジェネレーター

「アルゴリズム作曲の手法」の章で紹介したマルコフ連鎖を用いて、旋律を生成するプログラムです。 TRANSITION_TABLE には、各音符から遷移しうる次の音符の候補があらかじめ定義されており、 generate_melody 関数がその候補の中から乱数を用いて次の音を選択します。

 1"""マルコフ連鎖を用いて旋律を生成し、MIDI ファイルへ出力するサンプル。"""
 2
 3from __future__ import annotations
 4
 5import random
 6from pathlib import Path
 7
 8import pretty_midi
 9
10TransitionTable = dict[int, list[int]]
11
12TRANSITION_TABLE: TransitionTable = {
13    60: [62, 64],
14    62: [60, 64, 65],
15    64: [62, 65, 67],
16    65: [64, 67],
17    67: [65, 69, 72],
18    69: [67, 72],
19    72: [69, 67],
20}
21
22NOTE_DURATION = 0.4
23OUTPUT_PATH = Path(__file__).with_name("markov_melody.mid")
24
25
26def generate_melody(
27    transition_table: TransitionTable,
28    start_note: int,
29    note_count: int,
30) -> list[int]:
31    """遷移確率表に従って、次に続く音をランダムに選びながら旋律を生成する。"""
32    current_note = start_note
33    melody = [current_note]
34    for _ in range(note_count - 1):
35        current_note = random.choice(transition_table[current_note])
36        melody.append(current_note)
37    return melody
38
39
40def melody_to_midi(melody: list[int], output_path: Path) -> None:
41    """旋律のリストを MIDI ファイルへ書き出す。"""
42    midi_data = pretty_midi.PrettyMIDI()
43    instrument = pretty_midi.Instrument(program=0)
44
45    start_time = 0.0
46    for pitch in melody:
47        note = pretty_midi.Note(
48            velocity=100,
49            pitch=pitch,
50            start=start_time,
51            end=start_time + NOTE_DURATION,
52        )
53        instrument.notes.append(note)
54        start_time += NOTE_DURATION
55
56    midi_data.instruments.append(instrument)
57    midi_data.write(str(output_path))
58
59
60def main() -> None:
61    melody = generate_melody(TRANSITION_TABLE, start_note=60, note_count=32)
62    melody_to_midi(melody, OUTPUT_PATH)
63    print(f"{OUTPUT_PATH.name} を出力しました。")
64
65
66if __name__ == "__main__":
67    main()

ランダムメロディジェネレーターと比較すると、音の並びに一定の傾向が生まれ、より旋律らしい印象になることが確認できます。このサンプルコードは markov_melody_generator.py からダウンロードできます。実行すると、 markov_melody.mid という MIDI ファイルが生成されます。

python markov_melody_generator.py

ユークリッドリズムジェネレーター

「アルゴリズム作曲の手法」の章で紹介したユークリッドリズムを実装し、ドラムパターンとして MIDI ファイルへ出力するプログラムです。 euclidean_rhythm 関数が Bjorklund のアルゴリズムによってリズムパターンを計算し、 pattern_to_midi 関数がそのパターンをキック（ バスドラム ）の MIDI ノートへ変換します。

 1"""ユークリッドリズムのアルゴリズムでリズムパターンを生成し、MIDI ファイルへ出力するサンプル。"""
 2
 3from __future__ import annotations
 4
 5from pathlib import Path
 6
 7import pretty_midi
 8
 9STEP_DURATION = 0.2
10BAR_REPEAT_COUNT = 4
11KICK_NOTE_NUMBER = 36
12OUTPUT_PATH = Path(__file__).with_name("euclidean_rhythm.mid")
13
14
15def euclidean_rhythm(pulses: int, steps: int) -> list[bool]:
16    """Bjorklund のアルゴリズムにより、pulses 個の拍を steps 個のステップへ均等に配置する。"""
17    if pulses <= 0:
18        return [False] * steps
19    if pulses >= steps:
20        return [True] * steps
21
22    groups: list[list[bool]] = [[True] for _ in range(pulses)]
23    remainders: list[list[bool]] = [[False] for _ in range(steps - pulses)]
24
25    while len(remainders) > 1:
26        pair_count = min(len(groups), len(remainders))
27        merged = [groups[i] + remainders[i] for i in range(pair_count)]
28        leftover_groups = groups[pair_count:]
29        leftover_remainders = remainders[pair_count:]
30        groups = merged
31        remainders = (
32            leftover_groups if leftover_groups else leftover_remainders
33        )
34
35    result: list[bool] = []
36    for group in groups + remainders:
37        result.extend(group)
38    return result
39
40
41def pattern_to_midi(
42    pattern: list[bool], repeat_count: int, output_path: Path
43) -> None:
44    """リズムパターンを繰り返しながら、ドラム用の MIDI ファイルへ書き出す。"""
45    midi_data = pretty_midi.PrettyMIDI()
46    instrument = pretty_midi.Instrument(program=0, is_drum=True)
47
48    start_time = 0.0
49    for _ in range(repeat_count):
50        for hit in pattern:
51            if hit:
52                note = pretty_midi.Note(
53                    velocity=100,
54                    pitch=KICK_NOTE_NUMBER,
55                    start=start_time,
56                    end=start_time + STEP_DURATION,
57                )
58                instrument.notes.append(note)
59            start_time += STEP_DURATION
60
61    midi_data.instruments.append(instrument)
62    midi_data.write(str(output_path))
63
64
65def main() -> None:
66    pattern = euclidean_rhythm(pulses=5, steps=8)
67    pattern_to_midi(pattern, BAR_REPEAT_COUNT, OUTPUT_PATH)
68    print(f"{OUTPUT_PATH.name} を出力しました。")
69
70
71if __name__ == "__main__":
72    main()

既定の設定では、 8 ステップの中に 5 拍を配置したパターンを、 4 小節分繰り返した MIDI ファイル euclidean_rhythm.mid を生成します。 euclidean_rhythm 関数の引数を変更することで、さまざまな拍数とステップ数の組み合わせを試すことができます。このサンプルコードは euclidean_rhythm.py からダウンロードできます。

python euclidean_rhythm.py

サイン波シンセサイザー

「音声合成の基礎」の章で紹介した内容に基づき、サイン波に ADSR エンベロープを適用して WAV ファイルへ出力するプログラムです。 generate_sine_wave 関数で波形を生成し、 apply_adsr_envelope 関数で音量変化を適用した後、 write_wav 関数で 16 ビット PCM の WAV ファイルとして書き出します。

 1"""numpy でサイン波を合成し、ADSR エンベロープを適用して WAV ファイルへ出力するサンプル。"""
 2
 3from __future__ import annotations
 4
 5import wave
 6from pathlib import Path
 7
 8import numpy as np
 9
10SAMPLE_RATE = 44100
11FREQUENCY = 440.0
12DURATION = 2.0
13OUTPUT_PATH = Path(__file__).with_name("sine_wave.wav")
14
15
16def generate_sine_wave(
17    frequency: float,
18    duration: float,
19    sample_rate: int,
20) -> np.ndarray:
21    """指定した周波数のサイン波を、指定した長さ分だけ生成する。"""
22    sample_count = int(sample_rate * duration)
23    times = np.linspace(0, duration, sample_count, endpoint=False)
24    return np.sin(2 * np.pi * frequency * times)
25
26
27def apply_adsr_envelope(
28    samples: np.ndarray,
29    sample_rate: int,
30    attack: float,
31    decay: float,
32    sustain_level: float,
33    release: float,
34) -> np.ndarray:
35    """アタック・ディケイ・サステイン・リリースからなる音量変化を適用する。"""
36    total_count = len(samples)
37    attack_count = int(sample_rate * attack)
38    decay_count = int(sample_rate * decay)
39    release_count = int(sample_rate * release)
40    sustain_count = total_count - attack_count - decay_count - release_count
41    sustain_count = max(sustain_count, 0)
42
43    envelope = np.concatenate(
44        [
45            np.linspace(0.0, 1.0, attack_count, endpoint=False),
46            np.linspace(1.0, sustain_level, decay_count, endpoint=False),
47            np.full(sustain_count, sustain_level),
48            np.linspace(sustain_level, 0.0, release_count),
49        ]
50    )
51    envelope = envelope[:total_count]
52    return samples[: len(envelope)] * envelope
53
54
55def write_wav(
56    samples: np.ndarray, sample_rate: int, output_path: Path
57) -> None:
58    """浮動小数点の波形データを 16 ビット PCM の WAV ファイルへ書き出す。"""
59    normalized = np.clip(samples, -1.0, 1.0)
60    pcm_data = (normalized * 32767).astype(np.int16)
61
62    with wave.open(str(output_path), "wb") as wav_file:
63        wav_file.setnchannels(1)
64        wav_file.setsampwidth(2)
65        wav_file.setframerate(sample_rate)
66        wav_file.writeframes(pcm_data.tobytes())
67
68
69def main() -> None:
70    samples = generate_sine_wave(FREQUENCY, DURATION, SAMPLE_RATE)
71    samples = apply_adsr_envelope(
72        samples,
73        SAMPLE_RATE,
74        attack=0.05,
75        decay=0.1,
76        sustain_level=0.6,
77        release=0.3,
78    )
79    write_wav(samples, SAMPLE_RATE, OUTPUT_PATH)
80    print(f"{OUTPUT_PATH.name} を出力しました。")
81
82
83if __name__ == "__main__":
84    main()

実行すると、 440 Hz のサイン波（ 音楽で用いられる基準音、いわゆる A4 の音 ）を 2 秒間鳴らした sine_wave.wav が生成されます。このサンプルコードは sine_wave_synthesis.py からダウンロードできます。

python sine_wave_synthesis.py

サンプルコードの検証方法

ここまでに紹介した 4 つのサンプルコードは、以下のコマンドで静的解析を行うことができます。 mypy の実行には、 mypy.ini をダウンロードし、各サンプルコードと同じディレクトリへ配置した上で使用します。

pip install flake8 mypy
flake8 random_melody_generator.py markov_melody_generator.py euclidean_rhythm.py sine_wave_synthesis.py
mypy --config-file mypy.ini random_melody_generator.py markov_melody_generator.py euclidean_rhythm.py sine_wave_synthesis.py
