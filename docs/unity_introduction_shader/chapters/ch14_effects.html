第 14 章 表現テクニック

この章では、これまでに学んだ内容を組み合わせて、ゲームでよく使われる表現を作る。リムライト、ディゾルブ、トゥーンシェーディングと輪郭線の 3 つを扱う。

この章のサンプルコードは、examples/chapter14 フォルダーにある。

リムライト

リムライトは、オブジェクトの輪郭付近を光らせる表現である。逆光で輪郭が明るく縁取られる様子を表したり、キャラクターを背景から目立たせたりするときに使う。

カメラから見たオブジェクトの輪郭付近では、面がカメラに対して横を向いているため、法線 \(\mathbf{N}\) と視線の方向 \(\mathbf{V}\) がほぼ直交し、内積 \(\mathbf{N} \cdot \mathbf{V}\) が 0 に近くなる。反対に、カメラの正面を向いた面では内積が 1 に近い。そこで、\(1 - \mathbf{N} \cdot \mathbf{V}\) を輪郭らしさの値として使う。

\[I_{rim} = C_{rim} \, (1 - \max(0,\ \mathbf{N} \cdot \mathbf{V}))^{p}\]

\(p\) は、光る範囲を調整する値（_RimPower）である。値が大きいほど、光る範囲が輪郭の近くに狭まる。

RimLight.shader
 1// 第 14 章：オブジェクトの輪郭付近を光らせる（リムライト）シェーダー。
 2Shader "Introduction/Chapter14/RimLight"
 3{
 4    Properties
 5    {
 6        [MainColor] _BaseColor ("Base Color", Color) = (0.2, 0.2, 0.3, 1)
 7        [HDR] _RimColor ("Rim Color", Color) = (0.3, 0.8, 1, 1)
 8        // 値が大きいほど、光る範囲が輪郭の近くに狭まる
 9        _RimPower ("Rim Power", Range(0.5, 8)) = 3
10    }
11
12    SubShader
13    {

16            "RenderType" = "Opaque"
17            "Queue" = "Geometry"
18            "RenderPipeline" = "UniversalPipeline"
19        }

22        {

24            Tags { "LightMode" = "UniversalForward" }

28            #pragma fragment Frag
29

37            };

43                float3 normalWS : TEXCOORD1;
44            };

46            CBUFFER_START(UnityPerMaterial)
47                half4 _BaseColor;
48                half4 _RimColor;
49                half _RimPower;
50            CBUFFER_END
51
52            Varyings Vert(Attributes input)
53            {
54                Varyings output;
55                output.positionWS = TransformObjectToWorld(input.positionOS.xyz);
56                output.positionCS = TransformWorldToHClip(output.positionWS);
57                output.normalWS = TransformObjectToWorldNormal(input.normalOS);
58                return output;
59            }
60
61            half4 Frag(Varyings input) : SV_Target
62            {
63                float3 normalWS = normalize(input.normalWS);
64                float3 viewDirWS = GetWorldSpaceNormalizeViewDir(input.positionWS);
65                Light mainLight = GetMainLight();
66
67                half3 ambient = SampleSH(normalWS) * _BaseColor.rgb;
68                half nDotL = saturate(dot(normalWS, mainLight.direction));
69                half3 diffuse = _BaseColor.rgb * mainLight.color * nDotL;
70
71                // 法線と視線が直交する（内積が 0 に近い）輪郭付近ほど、rim が 1 に近づく
72                half nDotV = saturate(dot(normalWS, viewDirWS));
73                half rim = pow(1.0 - nDotV, _RimPower);
74                half3 rimColor = _RimColor.rgb * rim;
75
76                return half4(ambient + diffuse + rimColor, 1.0);
77            }
78            ENDHLSL
79        }
80    }
81}

RimLight.shader をダウンロード

球に適用すると、輪郭が水色に光る。リムライトの色は、ライトの方向に関係なく加えられるため、ライトが当たらない側の輪郭も光る。

_RimColor には [HDR] 属性を付けている。HDR の色は、各成分に 1 より大きい値を設定できる。URP の Volume で Bloom を有効にしている場合、1 を超える明るさの部分は周囲ににじむように光って見える。

フレネル効果

水面は、真上から見ると水の中が透けて見えるが、浅い角度から見ると空を映して鏡のように見える。このように、視線と面のなす角によって反射の強さが変わる現象を、フレネル効果という。フレネル効果の反射率は、Schlick の近似式でよく近似される。

\[F = F_0 + (1 - F_0)(1 - \mathbf{N} \cdot \mathbf{V})^5\]

\(F_0\) は、面を正面から見たときの反射率である。リムライトの式は、この近似式と同じ \((1 - \mathbf{N} \cdot \mathbf{V})\) のべき乗の形をしている。リムライトは、フレネル効果を表現上の目的で強調したものと考えられる。

ディゾルブ

ディゾルブは、オブジェクトが燃え尽きたり、溶けたりするように、少しずつ消えていく表現である。ノイズの値がしきい値より小さいピクセルを clip 関数で破棄し（第 12 章を参照）、しきい値を変えることで消える範囲を変える。

Dissolve.shader
  1// 第 14 章：ノイズを使って、
  2// オブジェクトが燃え尽きるように消えていく（ディゾルブ）シェーダー。
  3Shader "Introduction/Chapter14/Dissolve"
  4{
  5    Properties
  6    {
  7        [MainColor] _BaseColor ("Base Color", Color) = (1, 1, 1, 1)
  8        // 0 で消えている部分なし、1 ですべて消える
  9        _Threshold ("Threshold", Range(0, 1)) = 0.3
 10        // 消える境目を光らせる幅
 11        _EdgeWidth ("Edge Width", Range(0, 0.2)) = 0.05
 12        [HDR] _EdgeColor ("Edge Color", Color) = (4, 1.5, 0.3, 1)
 13        // ノイズの細かさ（UV 座標に掛ける値）
 14        _NoiseScale ("Noise Scale", Float) = 10
 15    }
 16
 17    SubShader
 18    {
 19        Tags
 20        {
 21            "RenderType" = "TransparentCutout"
 22            "Queue" = "AlphaTest"
 23            "RenderPipeline" = "UniversalPipeline"
 24        }
 25
 26        Pass
 27        {
 28            Name "Dissolve"
 29            Tags { "LightMode" = "UniversalForward" }
 30
 31            // 消えた部分から内側（裏面）が見えるため、裏面も描画する
 32            Cull Off
 33
 34            HLSLPROGRAM
 35            #pragma vertex Vert
 36            #pragma fragment Frag
 37
 38            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
 39            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl"
 40
 41            struct Attributes
 42            {
 43                float4 positionOS : POSITION;
 44                float3 normalOS : NORMAL;
 45                float2 uv : TEXCOORD0;
 46            };
 47
 48            struct Varyings
 49            {
 50                float4 positionCS : SV_POSITION;
 51                float3 normalWS : TEXCOORD0;
 52                float2 uv : TEXCOORD1;
 53            };
 54
 55            CBUFFER_START(UnityPerMaterial)
 56                half4 _BaseColor;
 57                half _Threshold;
 58                half _EdgeWidth;
 59                half4 _EdgeColor;
 60                float _NoiseScale;
 61            CBUFFER_END
 62
 63            // 2 次元の座標から、0 ～ 1 の疑似乱数を作る
 64            float Hash(float2 p)
 65            {
 66                return frac(sin(dot(p, float2(12.9898, 78.233))) * 43758.5453);
 67            }
 68
 69            // バリューノイズ：格子点ごとの乱数を、滑らかに補間したノイズ
 70            float ValueNoise(float2 uv)
 71            {
 72                float2 cell = floor(uv); // 格子の番号
 73                float2 local = frac(uv); // 格子の中での位置（0 ～ 1）
 74
 75                // 格子の四隅の乱数
 76                float a = Hash(cell);
 77                float b = Hash(cell + float2(1.0, 0.0));
 78                float c = Hash(cell + float2(0.0, 1.0));
 79                float d = Hash(cell + float2(1.0, 1.0));
 80
 81                // 3t^2 - 2t^3 で補間の重みを滑らかにし、格子の境目が目立たないようにする
 82                float2 t = local * local * (3.0 - 2.0 * local);
 83                return lerp(lerp(a, b, t.x), lerp(c, d, t.x), t.y);
 84            }
 85
 86            Varyings Vert(Attributes input)
 87            {
 88                Varyings output;
 89                output.positionCS = TransformObjectToHClip(input.positionOS.xyz);
 90                output.normalWS = TransformObjectToWorldNormal(input.normalOS);
 91                output.uv = input.uv;
 92                return output;
 93            }
 94
 95            half4 Frag(Varyings input, bool isFrontFace : SV_IsFrontFace) : SV_Target
 96            {
 97                float noise = ValueNoise(input.uv * _NoiseScale);
 98
 99                // ノイズの値がしきい値より小さいピクセルを破棄する
100                clip(noise - _Threshold);
101
102                // 裏面では法線を反転し、内側にもライティングが正しく当たるようにする
103                float3 normalWS = normalize(input.normalWS) * (isFrontFace ? 1.0 : -1.0);
104                Light mainLight = GetMainLight();
105                half3 ambient = SampleSH(normalWS) * _BaseColor.rgb;
106                half nDotL = saturate(dot(normalWS, mainLight.direction));
107                half3 diffuse = _BaseColor.rgb * mainLight.color * nDotL;
108
109                // しきい値のすぐ上（消える境目）ほど edge が 1 に近づく。
110                // しきい値が 0 のときは何も消えていないため、境目も光らせない。
111                half edge = 1.0 - smoothstep(0.0, _EdgeWidth, noise - _Threshold);
112                edge *= step(0.001, _Threshold);
113
114                half3 color = lerp(ambient + diffuse, _EdgeColor.rgb, edge);
115                return half4(color, 1.0);
116            }
117            ENDHLSL
118        }
119    }
120}

Dissolve.shader をダウンロード

マテリアルの Threshold を 0 から 1 に変えると、オブジェクトがまだらに消えていく。消える境目は、Edge Color の色で光る。

ノイズの生成

ディゾルブでは、場所によって不規則に変わるが、近い場所では近い値になるノイズを使う。ノイズのテクスチャーを使うこともできるが、サンプルでは、テクスチャーを用意しなくても動くように、バリューノイズをシェーダーの中で計算している。

  1. Hash 関数で、2 次元の座標から 0～1 の疑似乱数を作る。同じ座標からは常に同じ値が得られる。

  2. UV 座標を _NoiseScale 倍し、整数の格子に分ける。格子の四隅の点に、Hash 関数で乱数を割り当てる。

  3. 格子の中の点の値を、四隅の乱数から補間して求める。補間の重みに \(3t^2 - 2t^3\) を使うと、格子の境目で値が滑らかにつながる。

Hash 関数で使っている定数（12.9898、78.233、43758.5453）には特別な意味はなく、結果が不規則に見えるように選ばれた値である。

境目の発光

消える境目を光らせるため、ノイズの値としきい値の差が _EdgeWidth より小さい部分に、_EdgeColor の色を混ぜている。smoothstep 関数で、差が 0 のとき 1、_EdgeWidth 以上のとき 0 になる値を作っている。しきい値が 0 のときは何も消えていないため、step 関数で境目の発光を止めている。

消えた部分からはオブジェクトの内側（裏面）が見えるため、Cull Off で裏面も描画している。フラグメントシェーダーの SV_IsFrontFace セマンティクスの引数には、表面なら true、裏面なら false が入る。裏面では法線を反転して、内側のライティングも正しく計算されるようにしている。

C# からしきい値を変える

次のスクリプトは、しきい値を時間とともに変化させる。

DissolveAnimator.cs
 1using UnityEngine;
 2
 3/// <summary>
 4/// Dissolve シェーダーの _Threshold を時間とともに変化させ、
 5/// オブジェクトが消えたり現れたりするようにする。
 6/// </summary>
 7[RequireComponent(typeof(Renderer))]
 8public class DissolveAnimator : MonoBehaviour
 9{
10    private static readonly int ThresholdId = Shader.PropertyToID("_Threshold");
11
12    [Tooltip("完全に消えるまでの秒数")]
13    [SerializeField] private float _duration = 3.0f;
14
15    private Material _material;
16
17    private void Start()
18    {
19        _material = GetComponent<Renderer>().material;
20    }
21
22    private void Update()
23    {
24        // 0（消えていない）と 1（すべて消えた）の間を往復させる
25        float threshold = Mathf.PingPong(Time.time / _duration, 1.0f);
26        _material.SetFloat(ThresholdId, threshold);
27    }
28
29    private void OnDestroy()
30    {
31        Destroy(_material);
32    }
33}

トゥーンシェーディングと輪郭線

トゥーンシェーディングは、陰影をはっきりした数段階の色で塗り分け、アニメやイラストのように見せる表現である。輪郭線と組み合わせて使うことが多い。

ToonOutline.shader
  1// 第 14 章：陰影をはっきり 2 段階に分けるトゥーンシェーディングと、
  2// 輪郭線を描画するシェーダー。
  3Shader "Introduction/Chapter14/ToonOutline"
  4{
  5    Properties
  6    {
  7        [MainColor] _BaseColor ("Base Color", Color) = (1, 0.6, 0.5, 1)
  8        _ShadeColor ("Shade Color", Color) = (0.5, 0.3, 0.4, 1)
  9        // 明るい部分と暗い部分の境目の位置（N・L の値）

 11        // 境目をぼかす幅（0 でくっきり分かれる）
 12        _ShadeSmoothness ("Shade Smoothness", Range(0, 0.5)) = 0.02
 13        _OutlineColor ("Outline Color", Color) = (0.1, 0.05, 0.05, 1)
 14        // 輪郭線の太さ（オブジェクト空間の単位）
 15        _OutlineWidth ("Outline Width", Range(0, 0.1)) = 0.02
 16    }
 17

 19    {

 21        {
 22            "RenderType" = "Opaque"
 23            "Queue" = "Geometry"
 24            "RenderPipeline" = "UniversalPipeline"
 25        }

 35            half _ShadeSmoothness;
 36            half4 _OutlineColor;
 37            float _OutlineWidth;
 38        CBUFFER_END
 39        ENDHLSL
 40

 42        Pass
 43        {
 44            Name "Toon"
 45            Tags { "LightMode" = "UniversalForward" }
 46

 48            #pragma vertex Vert
 49            #pragma fragment Frag
 50
 51            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl"
 52
 53            struct Attributes
 54            {
 55                float4 positionOS : POSITION;
 56                float3 normalOS : NORMAL;
 57            };
 58
 59            struct Varyings
 60            {
 61                float4 positionCS : SV_POSITION;
 62                float3 normalWS : TEXCOORD0;
 63            };
 64
 65            Varyings Vert(Attributes input)
 66            {
 67                Varyings output;
 68                output.positionCS = TransformObjectToHClip(input.positionOS.xyz);
 69                output.normalWS = TransformObjectToWorldNormal(input.normalOS);
 70                return output;
 71            }
 72
 73            half4 Frag(Varyings input) : SV_Target
 74            {
 75                float3 normalWS = normalize(input.normalWS);
 76                Light mainLight = GetMainLight();
 77
 78                // Lambert と異なり、N・L を 0 ～ 1 に切り詰めずに使う（-1 ～ 1）
 79                half nDotL = dot(normalWS, mainLight.direction);
 80                // しきい値の前後で 0 から 1 に急に変わる値を作る
 81                half lit = smoothstep(
 82                    _ShadeThreshold - _ShadeSmoothness,
 83                    _ShadeThreshold + _ShadeSmoothness,
 84                    nDotL);
 85
 86                half3 color = lerp(_ShadeColor.rgb, _BaseColor.rgb, lit) * mainLight.color;
 87                return half4(color, 1.0);
 88            }
 89            ENDHLSL
 90        }
 91
 92        // 2 つ目のパス：法線方向に膨らませたメッシュの裏面だけを描画して、輪郭線にする
 93        Pass
 94        {
 95            Name "Outline"
 96            // URP は、LightMode が SRPDefaultUnlit のパスも、
 97            // UniversalForward のパスと同じタイミングで描画する
 98            Tags { "LightMode" = "SRPDefaultUnlit" }
 99
100            // 表面を描画せず、裏面だけを描画する
101            Cull Front
102
103            HLSLPROGRAM
104            #pragma vertex OutlineVert
105            #pragma fragment OutlineFrag
106
107            struct Attributes
108            {
109                float4 positionOS : POSITION;
110                float3 normalOS : NORMAL;
111            };
112
113            struct Varyings
114            {
115                float4 positionCS : SV_POSITION;
116            };
117
118            Varyings OutlineVert(Attributes input)
119            {
120                Varyings output;
121                // 頂点を法線方向に押し出す
122                float3 offset = normalize(input.normalOS) * _OutlineWidth;
123                float3 positionOS = input.positionOS.xyz + offset;
124                output.positionCS = TransformObjectToHClip(positionOS);
125                return output;
126            }
127
128            half4 OutlineFrag(Varyings input) : SV_Target
129            {

131            }
132            ENDHLSL
133        }

ToonOutline.shader をダウンロード

球に適用すると、明るい部分と暗い部分がくっきり分かれ、周囲に暗い輪郭線が描画される。

陰影の段階化

Lambert 反射では、\(\mathbf{N} \cdot \mathbf{L}\) の値に比例して明るさが滑らかに変わる。トゥーンシェーディングでは、\(\mathbf{N} \cdot \mathbf{L}\) がしきい値（_ShadeThreshold）より大きいかどうかで、明るい色（_BaseColor）と暗い色（_ShadeColor）を切り替える。

step 関数で切り替えると境目がぎざぎざになるため、サンプルでは smoothstep 関数で、しきい値の前後のわずかな幅（_ShadeSmoothness）だけ滑らかに切り替えている。

暗い部分を黒くせず、_ShadeColor で色を指定できるようにしているのは、トゥーン調の表現では、影の色を物体の色より少し青みや赤みのある色にすることが多いためである。

輪郭線

輪郭線は、背面法と呼ばれる方法で描画している。1 つのシェーダーに 2 つのパスを書き、2 つ目のパスで次の処理を行う。

  1. 頂点を法線の方向に _OutlineWidth だけ押し出し、メッシュを一回り大きくする。

  2. Cull Front で表面を描画せず、裏面だけを輪郭線の色で描画する。

一回り大きくしたメッシュの裏面は、元のメッシュの後ろにあるため、元のメッシュで隠れない輪郭の外側の部分だけが見える。これが輪郭線になる。

輪郭線のパスの LightMode は SRPDefaultUnlit である。URP は、1 つのシェーダーの中で LightMode が UniversalForward のパスと SRPDefaultUnlit のパスを、同じ描画の段階で両方とも描画する（第 5 章を参照）。2 つのパスの LightMode が同じだと、片方しか描画されない。

  • Cube のように角で法線が分かれているメッシュでは、角で輪郭線が途切れる。滑らかな法線を別に用意する方法などで対処する。

  • 押し出す量がオブジェクト空間の単位のため、カメラから遠いほど輪郭線が細く見え、Transform の Scale でも太さが変わる。
