11. 結合テスト

9 章のテストダブルを使うと、依存先から切り離してテスト対象をテストできる。しかし、実際のファイルやデータベースと組み合わせたときに正しく動くかは、テストダブルでは確かめられない。この章では、部品と外部の資源を組み合わせて確かめる結合テストの書き方を学ぶ。

11.1 結合テストで確かめること

結合テストでは、単体テストでは確かめられない、部品の間や外部の資源との「つなぎ目」を確かめる。たとえば、次のような欠陥は結合テストで見つかる。

  • ファイルの文字コードや改行コードの扱いの誤り

  • SQL の文法の誤りや、テーブルの制約の違反

  • 外部の API の応答の形式についての思い込み

  • 部品の間での、引数の順序や単位（円とドルなど）の食い違い

結合テストは、単体テストより準備が大がかりで、実行にも時間がかかる。テストピラミッド（3 章）に従い、ロジックの細かな場合分けは単体テストで確かめ、結合テストはつなぎ目の確認に絞るとよい。

この章では、自動テストとして手軽に扱える外部の資源として、ファイル、SQLite のデータベース、HTTP の API を取り上げる。

11.2 ファイルを使うテスト

売上の CSV ファイルを読み込み、商品別の合計を CSV ファイルに書き出すモジュールを題材にする。

ch11_integration/sales_report.py
 1"""売上の CSV ファイルを読み込み、商品別の集計をファイルに書き出す."""
 2
 3import csv
 4from collections import defaultdict
 5from pathlib import Path
 6
 7
 8def load_sales(path: Path) -> list[tuple[str, int]]:
 9    """売上の CSV ファイル（列: product, amount）を読み込む.
10
11    Raises:
12        ValueError: amount が整数として読めない行がある場合。
13    """
14    with path.open(encoding="utf-8", newline="") as f:
15        reader = csv.DictReader(f)
16        rows: list[tuple[str, int]] = []
17        for line_no, row in enumerate(reader, start=2):
18            try:
19                rows.append((row["product"], int(row["amount"])))
20            except ValueError as e:
21                raise ValueError(f"{line_no} 行目の金額が不正です") from e
22        return rows
23
24
25def write_summary(sales: list[tuple[str, int]], path: Path) -> None:
26    """商品別の売上合計を、金額の大きい順に CSV ファイルへ書き出す."""
27    totals: defaultdict[str, int] = defaultdict(int)
28    for product, amount in sales:
29        totals[product] += amount
30    ranking = sorted(totals.items(), key=lambda kv: (-kv[1], kv[0]))
31    with path.open("w", encoding="utf-8", newline="") as f:
32        writer = csv.writer(f)
33        writer.writerow(["product", "total"])
34        writer.writerows(ranking)

ファイルを使うテストでは、8 章の組み込みの fixture tmp_path を使う。tmp_path はテストごとに新しい一時フォルダーを用意するため、テストが作ったファイルがほかのテストに影響したり、作業中のフォルダーに残ったりしない。

ch11_integration/test_sales_report.py
 1"""ファイルの読み書きを含めて sales_report をテストする."""
 2
 3from pathlib import Path
 4
 5import pytest
 6
 7from sales_report import load_sales, write_summary
 8
 9
10@pytest.fixture
11def sales_csv(tmp_path: Path) -> Path:
12    """テスト用の売上ファイルを一時フォルダーに作る."""
13    path = tmp_path / "sales.csv"
14    path.write_text(
15        "product,amount\n"
16        "りんご,300\n"
17        "メロン,2000\n"
18        "りんご,450\n",
19        encoding="utf-8",
20    )
21    return path
22
23
24def test_load_sales(sales_csv: Path) -> None:
25    assert load_sales(sales_csv) == [
26        ("りんご", 300), ("メロン", 2000), ("りんご", 450)]
27
28
29def test_load_sales_with_invalid_amount(tmp_path: Path) -> None:
30    path = tmp_path / "broken.csv"
31    path.write_text("product,amount\nりんご,三百\n", encoding="utf-8")
32    with pytest.raises(ValueError, match="2 行目"):
33        load_sales(path)
34
35
36def test_load_and_write_summary(sales_csv: Path, tmp_path: Path) -> None:
37    output = tmp_path / "summary.csv"
38
39    write_summary(load_sales(sales_csv), output)
40
41    assert output.read_text(encoding="utf-8").splitlines() == [
42        "product,total",
43        "メロン,2000",
44        "りんご,750",
45    ]

sales_csv fixture（10～21 行目）は、テスト用の CSV ファイルを一時フォルダーに作る。テストデータをテストの中で作ると、テストを読むだけで入力の内容が分かる。テストデータを別のファイルとして用意する方法もあるが、その場合はテストと内容を突き合わせて読む必要がある。

test_load_and_write_summary（36～45 行目）は、読み込みから書き出しまでを通して確かめる。書き出したファイルを実際に読み直して比べることで、文字コードや CSV の区切り方も含めて確かめている。

11.3 データベースを使うテスト

注文を SQLite のデータベースに保存するクラスを題材にする。

ch11_integration/order_repository.py
 1"""注文を SQLite のデータベースに保存する."""
 2
 3import sqlite3
 4
 5
 6class OrderRepository:
 7    """注文の保存と集計を担当するクラス."""
 8
 9    def __init__(self, conn: sqlite3.Connection) -> None:
10        self._conn = conn
11
12    def create_table(self) -> None:
13        """注文のテーブルを作る."""
14        self._conn.execute(
15            "CREATE TABLE IF NOT EXISTS orders ("
16            " id INTEGER PRIMARY KEY,"
17            " customer TEXT NOT NULL,"
18            " amount INTEGER NOT NULL CHECK (amount >= 0))")
19
20    def add(self, order_id: int, customer: str, amount: int) -> None:
21        """注文を 1 件保存する.
22
23        Raises:
24            sqlite3.IntegrityError: 注文番号の重複や負の金額の場合。
25        """
26        with self._conn:  # 成功すればコミット、例外ならロールバック
27            self._conn.execute(
28                "INSERT INTO orders (id, customer, amount) VALUES (?, ?, ?)",
29                (order_id, customer, amount))
30
31    def total_by_customer(self) -> dict[str, int]:
32        """顧客ごとの注文金額の合計を返す."""
33        rows = self._conn.execute(
34            "SELECT customer, SUM(amount) FROM orders GROUP BY customer")
35        return {customer: total for customer, total in rows}

SQLite は、":memory:" を指定して接続すると、メモリー上にデータベースを作る。ファイルを作らず、接続を閉じれば消えるため、テストごとに空のデータベースを用意する用途に向いている。

ch11_integration/test_order_repository.py
 1"""SQLite のインメモリデータベースを使って OrderRepository をテストする."""
 2
 3import sqlite3
 4from collections.abc import Iterator
 5
 6import pytest
 7
 8from order_repository import OrderRepository
 9
10
11@pytest.fixture
12def repo() -> Iterator[OrderRepository]:
13    """テストごとに空のデータベースを用意し、終わったら閉じる."""
14    conn = sqlite3.connect(":memory:")
15    repository = OrderRepository(conn)
16    repository.create_table()
17    yield repository  # ここでテストが実行される
18    conn.close()      # 後始末
19
20
21def test_total_by_customer(repo: OrderRepository) -> None:
22    repo.add(1, "佐藤", 1200)
23    repo.add(2, "鈴木", 800)
24    repo.add(3, "佐藤", 300)
25    assert repo.total_by_customer() == {"佐藤": 1500, "鈴木": 800}
26
27
28def test_empty_table(repo: OrderRepository) -> None:
29    assert repo.total_by_customer() == {}
30
31
32def test_duplicate_id_is_rejected(repo: OrderRepository) -> None:
33    repo.add(1, "佐藤", 1200)
34    with pytest.raises(sqlite3.IntegrityError):
35        repo.add(1, "鈴木", 800)
36    # 失敗した INSERT はロールバックされ、最初の注文だけが残る
37    assert repo.total_by_customer() == {"佐藤": 1200}
38
39
40def test_negative_amount_is_rejected(repo: OrderRepository) -> None:
41    with pytest.raises(sqlite3.IntegrityError):
42        repo.add(1, "佐藤", -100)

repo fixture（11～18 行目）は、8.3 節の yield を使う fixture である。テストの前にデータベースとテーブルを作り、テストの後に接続を閉じる。

test_duplicate_id_is_rejected と test_negative_amount_is_rejected は、テーブルの主キーと CHECK 制約が働くことを確かめている。これらは SQL とデータベースが実際に動いて初めて確かめられる性質であり、データベースをテストダブルに置き換えた単体テストでは確かめられない。また、test_duplicate_id_is_rejected の最後の assert は、失敗した INSERT がロールバックされ、データベースに不完全なデータが残らないことを確かめている。

本番で PostgreSQL や MySQL などを使う場合、SQLite でのテストだけでは、SQL の方言や型の違いによる欠陥を見逃すおそれがある。その場合は、本番と同じ種類のデータベースをテスト用に起動して結合テストを行う。コンテナーを使うと、テストのたびに同じ状態のデータベースを用意しやすい。

11.4 HTTP の API を使うテスト

為替レートを Web API から取得するクライアントを題材にする。

ch11_integration/exchange_client.py
 1"""為替レートを Web API から取得する."""
 2
 3import httpx
 4
 5
 6class ExchangeRateError(Exception):
 7    """為替レートを取得できなかったときに送出する例外."""
 8
 9
10class ExchangeClient:
11    """為替レートの API のクライアント.
12
13    API は GET /rates?base=USD&target=JPY に対して
14    {"base": "USD", "target": "JPY", "rate": 150.5} の形の JSON を返すものとする。
15    """
16
17    def __init__(self, client: httpx.Client) -> None:
18        self._client = client
19
20    def rate(self, base: str, target: str) -> float:
21        """base 通貨 1 単位が target 通貨でいくらになるかを返す.
22
23        Raises:
24            ExchangeRateError: API がエラーを返した場合。
25        """
26        try:
27            response = self._client.get(
28                "/rates", params={"base": base, "target": target})
29            response.raise_for_status()
30        except httpx.HTTPError as e:
31            raise ExchangeRateError(f"{base}/{target} を取得できません") from e
32        return float(response.json()["rate"])
33
34    def convert(self, amount: float, base: str, target: str) -> float:
35        """金額を換算し、小数第 2 位に丸めて返す."""
36        return round(amount * self.rate(base, target), 2)

ExchangeClient は、HTTP クライアントの httpx.Client をコンストラクターで受け取る（17～18 行目）。10 章の依存性の注入である。

外部の API を使うテストで本物のサーバーに接続すると、ネットワークの状態や API の利用制限によってテストが不安定になる。httpx には、実際の通信の代わりに関数で応答を返す httpx.MockTransport が用意されている。これを使うと、HTTP のリクエストの組み立てと応答の解析は本物の httpx で行いながら、通信の部分だけを差し替えられる。

ch11_integration/test_exchange_client.py
 1"""httpx.MockTransport で API の応答を差し替えて ExchangeClient をテストする."""
 2
 3from collections.abc import Iterator
 4
 5import httpx
 6import pytest
 7
 8from exchange_client import ExchangeClient, ExchangeRateError
 9
10
11def fake_api(request: httpx.Request) -> httpx.Response:
12    """本物の API の代わりに応答を返す関数."""
13    if request.url.path != "/rates":
14        return httpx.Response(404)
15    base = request.url.params["base"]
16    target = request.url.params["target"]
17    if (base, target) == ("USD", "JPY"):
18        return httpx.Response(
19            200, json={"base": base, "target": target, "rate": 150.5})
20    return httpx.Response(400, json={"error": "unsupported currency"})
21
22
23@pytest.fixture
24def client() -> Iterator[ExchangeClient]:
25    """ネットワークに接続しないクライアント."""
26    http = httpx.Client(transport=httpx.MockTransport(fake_api),
27                        base_url="https://api.example.com")
28    yield ExchangeClient(http)
29    http.close()
30
31
32def test_rate(client: ExchangeClient) -> None:
33    assert client.rate("USD", "JPY") == 150.5
34
35
36def test_convert(client: ExchangeClient) -> None:
37    assert client.convert(12.34, "USD", "JPY") == 1857.17
38
39
40def test_error_response_raises(client: ExchangeClient) -> None:
41    with pytest.raises(ExchangeRateError):
42        client.rate("USD", "XXX")
43
44
45def test_request_parameters() -> None:
46    # 送信されたリクエストを記録して、パラメーターを検証する
47    requests: list[httpx.Request] = []
48
49    def handler(request: httpx.Request) -> httpx.Response:
50        requests.append(request)
51        return httpx.Response(200, json={"rate": 0.0067})
52
53    with httpx.Client(transport=httpx.MockTransport(handler),
54                      base_url="https://api.example.com") as http:
55        ExchangeClient(http).rate("JPY", "USD")
56
57    assert len(requests) == 1
58    assert requests[0].url.params["base"] == "JPY"
59    assert requests[0].url.params["target"] == "USD"

fake_api（11～20 行目）は、リクエストを受け取って応答を返す関数で、本物の API の代わりを務めるフェイクである。サポートしない通貨に対しては、ステータスコード 400 の応答を返す。これにより、test_error_response_raises（40～42 行目）で、API がエラーを返した場合の動作を確かめられる。本物の API では、エラーの応答を意図して返させるのが難しいことが多い。

test_request_parameters（45～59 行目）は、送信されたリクエストを記録するスパイとして handler を使い、クエリーのパラメーターが正しく組み立てられていることを確かめている。

この方法でも、本物の API の応答の形式が想定と異なれば、欠陥は見つからない。本物の API に接続するテストも用意しておき、8 章のマーカーで区別して、普段の実行からは除外する運用がよく行われる。

11.5 テストデータの管理

結合テストでは、テストデータの管理が重要になる。次の点に注意する。

  • テストごとに独立したデータを用意する：あるテストが作ったデータを、別のテストが前提にしてはならない。実行の順序を変えたり、1 つのテストだけを実行したりしたときに失敗するようになる。

  • 後始末を確実に行う：tmp_path や yield を使う fixture を使い、テストが失敗しても後始末が実行されるようにする。

  • 必要最小限のデータにする：テストに関係のないデータが多いと、テストの意図が読み取りにくくなり、実行も遅くなる。

  • 本番のデータをそのまま使わない：個人情報などを含む可能性があるうえ、データが変わるとテストの結果も変わる。

11.6 まとめ

  • 結合テストでは、部品の間や外部の資源とのつなぎ目を確かめる。ロジックの場合分けは単体テストに任せる。

  • ファイルを使うテストでは tmp_path を使い、テストごとに独立した一時フォルダーで読み書きする。

  • SQLite のインメモリデータベースを使うと、テストごとに空のデータベースを手軽に用意できる。

  • HTTP の API は httpx.MockTransport で通信の部分だけを差し替え、エラーの応答なども再現する。

  • テストデータはテストごとに独立させ、後始末を確実に行う。
