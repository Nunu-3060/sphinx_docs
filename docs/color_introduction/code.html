 60def hex_to_rgb(code: str) -> FloatArray:
 61    """``"#1f77b4"`` のようなカラーコードを sRGB 値 (0.0-1.0) に変換する。"""

 65    values = [int(text[i:i + 2], 16) for i in range(0, 6, 2)]
 66    return np.array(values, dtype=float) / 255.0

 85def srgb_to_linear(rgb: ArrayLike) -> FloatArray:
 86    """ガンマ補正された sRGB 値を、光の強さに比例する線形な値に戻す。"""
 87    values = np.asarray(rgb, dtype=float)
 88    return np.where(
 89        values <= 0.04045,
 90        values / 12.92,
 91        ((values + 0.055) / 1.055) ** 2.4,
 92    )

 95def linear_to_srgb(linear: ArrayLike) -> FloatArray:
 96    """線形な値を、画像として保存する sRGB 値にガンマ補正する。

 98    0.0-1.0 の範囲外の値は、範囲内に切り詰めてから変換する。
 99    """

101    return np.where(
102        values <= 0.0031308,
103        values * 12.92,
104        1.055 * values ** (1.0 / 2.4) - 0.055,

112def hsv_to_rgb(hue: ArrayLike, saturation: ArrayLike,
113               value: ArrayLike) -> FloatArray:
114    """HSV を sRGB 値に変換する。
115
116    色相 ``hue`` は 1 周を 1.0 とする値 (0.0 が赤、1/3 が緑、2/3 が青)、
117    彩度 ``saturation`` と明度 ``value`` は 0.0-1.0 の値で与える。
118    引数は同じ形の配列でもよく、戻り値の末尾に RGB の軸が加わる。
119    """
120    h = np.mod(np.asarray(hue, dtype=float), 1.0) * 6.0
121    s = np.asarray(saturation, dtype=float)
122    v = np.asarray(value, dtype=float)
123    h, s, v = np.broadcast_arrays(h, s, v)
124
125    sector = np.floor(h).astype(int) % 6
126    fraction = h - np.floor(h)
127    p = v * (1.0 - s)
128    q = v * (1.0 - s * fraction)
129    t = v * (1.0 - s * (1.0 - fraction))
130
131    # 色相環を 60 度ずつ 6 つに分け、区間ごとに RGB の並びを決める。
132    r = np.choose(sector, [v, q, p, p, t, v])
133    g = np.choose(sector, [t, v, v, q, p, p])
134    b = np.choose(sector, [p, p, t, v, v, q])
135    return np.stack([r, g, b], axis=-1)

147def xyz_to_lab(xyz: ArrayLike) -> FloatArray:
148    """CIE XYZ を CIELAB (L*, a*, b*) に変換する。白色点は D65 とする。"""
149    ratio = np.asarray(xyz, dtype=float) / WHITE_D65
150    delta = 6.0 / 29.0
151    f = np.where(
152        ratio > delta ** 3,
153        np.cbrt(ratio),
154        ratio / (3.0 * delta ** 2) + 4.0 / 29.0,
155    )
156    lightness = 116.0 * f[..., 1] - 16.0
157    a = 500.0 * (f[..., 0] - f[..., 1])
158    b = 200.0 * (f[..., 1] - f[..., 2])
159    return np.stack([lightness, a, b], axis=-1)

167def delta_e_76(lab1: ArrayLike, lab2: ArrayLike) -> FloatArray:
168    """CIELAB の 2 色の間のユークリッド距離 (色差 ΔE*ab) を求める。"""
169    diff = np.asarray(lab1, dtype=float) - np.asarray(lab2, dtype=float)
170    return np.sqrt(np.sum(diff ** 2, axis=-1))

194def oklch_to_oklab(lightness: ArrayLike, chroma: ArrayLike,
195                   hue_deg: ArrayLike) -> FloatArray:
196    """OKLCH (明るさ L、彩度 C、色相角 h [度]) を OKLab に変換する。"""
197    angle = np.deg2rad(np.asarray(hue_deg, dtype=float))
198    c = np.asarray(chroma, dtype=float)
199    lightness_arr, a, b = np.broadcast_arrays(
200        np.asarray(lightness, dtype=float), c * np.cos(angle),
201        c * np.sin(angle))
202    return np.stack([lightness_arr, a, b], axis=-1)

205def oklab_to_oklch(lab: ArrayLike) -> FloatArray:
206    """OKLab を OKLCH (L, C, h [度]) に変換する。h は 0 以上 360 未満。"""
207    values = np.asarray(lab, dtype=float)
208    chroma = np.hypot(values[..., 1], values[..., 2])
209    hue = np.mod(np.rad2deg(np.arctan2(values[..., 2], values[..., 1])),
210                 360.0)
211    return np.stack([values[..., 0], chroma, hue], axis=-1)

214def in_srgb_gamut(lab: ArrayLike, tolerance: float = 1e-6) -> NDArray[
215        np.bool_]:
216    """OKLab の色が sRGB の色域 (RGB が全て 0.0-1.0) に収まるかを返す。"""
217    linear = oklab_to_linear(lab)
218    inside = (linear >= -tolerance) & (linear <= 1.0 + tolerance)
219    return np.all(inside, axis=-1)

222def oklch_to_srgb_in_gamut(lightness: ArrayLike, chroma: ArrayLike,
223                           hue_deg: ArrayLike) -> FloatArray:
224    """OKLCH を sRGB 値に変換する。色域外の色は彩度だけを下げて収める。
225
226    RGB の各成分を単純に切り詰めると、明るさや色相まで変わってしまう。
227    ここでは明るさ L と色相 h を保ったまま、色域に収まる最大の彩度を
228    二分探索で求める。
229    """
230    lightness_arr, chroma_arr, hue_arr = np.broadcast_arrays(
231        np.asarray(lightness, dtype=float), np.asarray(chroma, dtype=float),
232        np.asarray(hue_deg, dtype=float))
233    # low は常に色域内、high は元の彩度から始まる探索範囲の上端。
234    low = np.zeros_like(chroma_arr)
235    high = chroma_arr.copy()
236    for _ in range(30):
237        middle = (low + high) / 2.0
238        ok = in_srgb_gamut(oklch_to_oklab(lightness_arr, middle, hue_arr))
239        low = np.where(ok, middle, low)
240        high = np.where(ok, high, middle)
241    return oklab_to_srgb(oklch_to_oklab(lightness_arr, low, hue_arr))

260def contrast_ratio(rgb1: ArrayLike, rgb2: ArrayLike) -> float:
261    """2 色のコントラスト比 (1.0 から 21.0) を求める。"""
262    lum1 = float(relative_luminance(rgb1))
263    lum2 = float(relative_luminance(rgb2))
264    lighter, darker = max(lum1, lum2), min(lum1, lum2)
265    return (lighter + 0.05) / (darker + 0.05)

328def labeled_rows(rows: Sequence[tuple[str, ArrayLike]],
329                 label_width: int = 180, gap: int = 12,
330                 font_size: int = 18) -> Image.Image:
331    """左端にラベルを付けた画像を、上から順に縦に並べた 1 枚の画像にする。

336    width = label_width + max(a.shape[1] for a in arrays) + gap
337    height = sum(a.shape[0] for a in arrays) + gap * (len(arrays) + 1)
338    canvas = Image.new("RGB", (width, height), "white")
339    draw = ImageDraw.Draw(canvas)
340    label_font = font(font_size)
341

344        canvas.paste(Image.fromarray(array), (label_width, top))

349        draw.text((gap, top + array.shape[0] // 2), label, fill="black",
350                  font=label_font, anchor="lm")
351        top += array.shape[0] + gap

 14def circle_masks(size: int, radius: float,
 15                 offset: float) -> list[FloatArray]:
 16    """正三角形の頂点に中心を置いた、3 つの円のマスク (0.0 か 1.0) を作る。"""
 17    y, x = np.mgrid[0:size, 0:size].astype(float)
 18    center = size / 2.0
 19    masks = []
 20    for angle in (-90.0, 30.0, 150.0):
 21        cx = center + offset * np.cos(np.deg2rad(angle))
 22        cy = center + offset * np.sin(np.deg2rad(angle)) + offset * 0.25
 23        inside = (x - cx) ** 2 + (y - cy) ** 2 <= radius ** 2
 24        masks.append(inside.astype(float))
 25    return masks

 28def additive_mixing(size: int = 360) -> FloatArray:
 29    """黒地に光の三原色 (赤・緑・青) の円を重ね、成分を足し合わせる。"""
 30    masks = circle_masks(size, radius=size * 0.28, offset=size * 0.17)
 31    primaries = [hex_to_rgb("#ff0000"), hex_to_rgb("#00ff00"),
 32                 hex_to_rgb("#0000ff")]
 33    image = np.zeros((size, size, 3))
 34    for mask, color in zip(masks, primaries):
 35        image += mask[..., np.newaxis] * color
 36    return np.clip(image, 0.0, 1.0)

 39def subtractive_mixing(size: int = 360) -> FloatArray:
 40    """白地に色材の三原色 (シアン・マゼンタ・イエロー) の円を重ねる。
 41
 42    色材は光の一部を吸収するので、重なった部分では反射率を掛け合わせる。
 43    """
 44    masks = circle_masks(size, radius=size * 0.28, offset=size * 0.17)
 45    primaries = [hex_to_rgb("#00ffff"), hex_to_rgb("#ff00ff"),
 46                 hex_to_rgb("#ffff00")]
 47    image = np.ones((size, size, 3))
 48    for mask, color in zip(masks, primaries):
 49        # マスクの外側では反射率 1.0 (何も吸収しない) として掛ける。
 50        image *= 1.0 - mask[..., np.newaxis] * (1.0 - color)
 51    return image

 85def mix_scale(base: FloatArray, steps: int) -> list[FloatArray]:
 86    """基準の色に白または黒を sRGB の値で混ぜて、濃淡の段階を作る。
 87
 88    明るい側の半分は白を、暗い側の半分は黒を混ぜる。
 89    """
 90    white, black = np.ones(3), np.zeros(3)
 91    half = steps // 2
 92    light = [base + (white - base) * (1.0 - i / half) for i in range(half)]
 93    dark = [base + (black - base) * (i / (steps - half))
 94            for i in range(steps - half)]
 95    return light + dark

 98def oklch_scale(base: FloatArray, steps: int, lightest: float = 0.96,
 99                darkest: float = 0.30) -> list[FloatArray]:
100    """基準の色の色相を保ち、OKLab の明るさ L を等間隔に変えて段階を作る。
101
102    明るさは ``lightest`` から ``darkest`` まで変える。明るさの両端では
103    表せる彩度が小さくなるため、彩度は中央で最大になるように山形に変え、
104    さらに色域に収まるように下げる。
105    """
106    _, base_chroma, hue = oklab_to_oklch(srgb_to_oklab(base))
107    colors = []
108    for lightness in np.linspace(lightest, darkest, steps):
109        # 明るさ 0.62 付近で基準の彩度になり、両端に向かって小さくなる。
110        weight = 1.0 - abs(lightness - 0.62) / 0.5
111        chroma = base_chroma * max(weight, 0.15)
112        colors.append(oklch_to_srgb_in_gamut(lightness, chroma, hue))
113    return colors

144    sky_top, sky_bottom = hex_to_rgb("#3b2c63"), hex_to_rgb("#f39a4a")
145    image = oklab_to_srgb(
146        srgb_to_oklab(sky_top) * (1.0 - sky_t[..., np.newaxis])
147        + srgb_to_oklab(sky_bottom) * sky_t[..., np.newaxis])
148
149    # 太陽: 地平線に半分沈んだ円。
150    sun_x = width * 0.62
151    sun = (x - sun_x) ** 2 + (y - horizon) ** 2 < (height * 0.1) ** 2
152    image[sun] = hex_to_rgb("#ffd36b")
153
154    # 山並み: 正弦波を重ねた稜線より下を暗い色で塗る。
155    # 太陽を隠さないよう、太陽に近いほど山を低くする。
156    envelope = np.clip((np.abs(x - sun_x) - 40.0) / 100.0, 0.0, 1.0)
157    ridge = horizon - envelope * (
158        40.0 + 18.0 * np.sin(x / 57.0) + 10.0 * np.sin(x / 23.0 + 1.0))
159    image[(y > ridge) & (y < horizon)] = hex_to_rgb("#4a2f4f")
160
161    # 海: 地平線より下。深いところほど暗い青にする。
162    sea_t = np.clip((y - horizon) / (height - horizon), 0.0, 1.0)
163    sea_top, sea_bottom = hex_to_rgb("#6b5a8e"), hex_to_rgb("#1d2b4f")
164    sea = (sea_top * (1.0 - sea_t[..., np.newaxis])
165           + sea_bottom * sea_t[..., np.newaxis])
166    below = y >= horizon
167    image[below] = sea[below]
168
169    # 海面に映る太陽の光: 横縞状の明るい帯。
170    glitter = (below & (np.abs(x - sun_x) < 36.0 - sea_t * 24.0)
171               & (np.sin(y * 0.9 + np.sin(x * 0.2) * 2.0) > 0.3))
172    image[glitter] = hex_to_rgb("#f7b267")
173
174    noise = rng.normal(0.0, 0.015, image.shape)
175    return np.clip(image + noise, 0.0, 1.0)

178def kmeans(points: FloatArray, k: int, iterations: int = 30,
179           seed: int = 0) -> tuple[FloatArray, NDArray[np.intp]]:
180    """k-means 法で点の集まりを ``k`` 個のクラスタに分ける。
181
182    戻り値は (各クラスタの中心, 各点が属するクラスタの番号)。
183    """
184    rng = np.random.default_rng(seed)
185    centers = points[rng.choice(len(points), size=k, replace=False)]
186    labels = np.zeros(len(points), dtype=np.intp)
187    for _ in range(iterations):
188        # 各点を、最も近い中心のクラスタに割り当てる。
189        distances = np.linalg.norm(
190            points[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=-1)
191        labels = np.argmin(distances, axis=1)
192        # 各クラスタの中心を、属する点の平均に動かす。
193        for i in range(k):
194            members = points[labels == i]
195            if len(members) > 0:
196                centers[i] = members.mean(axis=0)
197    return centers, labels

200def extract_palette(image: FloatArray, k: int,
201                    sample: int = 6000) -> list[tuple[FloatArray, float]]:
202    """画像から ``k`` 色を取り出し、(色, 画像に占める割合) の並びを返す。
203
204    距離を見た目の差に近づけるため、画素を OKLab に変換してから分類する。
205    計算を軽くするため、画素の一部を無作為に選んで使う。
206    """
207    rng = np.random.default_rng(0)
208    pixels = srgb_to_oklab(image.reshape(-1, 3))
209    chosen = pixels[rng.choice(len(pixels), size=sample, replace=False)]
210    centers, labels = kmeans(chosen, k)
211    shares = np.bincount(labels, minlength=k) / len(labels)
212    order = np.argsort(centers[:, 0])  # 暗い色から明るい色の順に並べる。
213    return [(oklab_to_srgb(centers[i]), float(shares[i])) for i in order]

 64def sample_field(width: int = 360, height: int = 240) -> FloatArray:
 65    """なだらかな山と谷を重ねた、0.0-1.0 の値を持つ 2 次元のデータ。"""
 66    y, x = np.mgrid[0:height, 0:width].astype(float)
 67    u, v = x / width, y / height
 68    field = (0.6 * u
 69             + 0.25 * np.exp(-((u - 0.3) ** 2 + (v - 0.4) ** 2) / 0.02)
 70             - 0.2 * np.exp(-((u - 0.7) ** 2 + (v - 0.6) ** 2) / 0.03)
 71             + 0.05 * np.sin(u * 20.0) * np.sin(v * 14.0))
 72    value_range = field.max() - field.min()
 73    normalized: FloatArray = (field - field.min()) / value_range
 74    return normalized

 26def naive_rgb_to_cmyk(rgb: FloatArray) -> FloatArray:
 27    """RGB を CMYK の網点の面積率 (0.0-1.0) に変換する簡易的な式。
 28
 29    K (黒) を R、G、B の最大値から決め、残りを C、M、Y に割り振る。
 30    インキや用紙の特性を考慮していないので、実際の印刷の色は再現できない。
 31    """
 32    rgb = np.asarray(rgb, dtype=float)
 33    k = 1.0 - rgb.max(axis=-1)
 34    # K = 1 (黒) の画素では C、M、Y を 0 にする (0 除算を避ける)。
 35    scale = np.where(k < 1.0, 1.0 - k, 1.0)[..., np.newaxis]
 36    cmy = (1.0 - rgb - k[..., np.newaxis]) / scale
 37    return np.concatenate([cmy, k[..., np.newaxis]], axis=-1)

 68def dot_gain(coverage: FloatArray, gain_at_half: float) -> FloatArray:
 69    """網点が太ることによる、実際の面積率の増え方を放物線で近似する。
 70
 71    面積率 50% のときに ``gain_at_half`` だけ増え、0% と 100% では増えない。
 72    """
 73    gained = coverage + gain_at_half * 4.0 * coverage * (1.0 - coverage)
 74    return np.clip(gained, 0.0, 1.0)

 77def printed_tone(coverage: FloatArray, paper: FloatArray,
 78                 ink: FloatArray) -> FloatArray:
 79    """黒インキの網点の面積率から、印刷物の見た目の色を求める。
 80
 81    Murray-Davies の式にならい、光の反射率 (線形な値) を、インキの部分と
 82    紙の部分の面積で重み付けして平均する。
 83    """
 84    reflectance = (coverage[..., np.newaxis] * srgb_to_linear(ink)
 85                   + (1.0 - coverage[..., np.newaxis]) * srgb_to_linear(paper))
 86    return linear_to_srgb(reflectance)
