ch02_dimension.py

  1"""第 2 章: 次元を持つ量を計算し、単位の取り違えを検出する。
  2
  3物理量を「数値」と「次元 (長さ L、質量 M、時間 T、電流 I の指数)」の組で
  4表す小さなクラスを作り、次の 3 点を確かめます。
  5
  6* 掛け算・割り算では次元の指数が足し引きされること
  7* 次元の異なる量どうしの足し算はエラーになること
  8* ポンド重秒 (lbf·s) とニュートン秒 (N·s) を取り違えると、
  9  数値が約 4.45 倍ずれること
 10
 11実行例::
 12
 13    python ch02_dimension.py
 14"""
 15
 16from __future__ import annotations
 17
 18from dataclasses import dataclass
 19
 20# 次元の指数 (L, M, T, I) の組
 21Dimension = tuple[int, int, int, int]
 22
 23DIMENSIONLESS: Dimension = (0, 0, 0, 0)
 24LENGTH: Dimension = (1, 0, 0, 0)
 25MASS: Dimension = (0, 1, 0, 0)
 26TIME: Dimension = (0, 0, 1, 0)
 27CURRENT: Dimension = (0, 0, 0, 1)
 28
 29# 1 lbf (ポンド重) をニュートンに換算する係数
 30NEWTON_PER_POUND_FORCE = 4.4482216152605
 31
 32
 33class DimensionError(ValueError):
 34    """次元が合わない演算をしたときに送出する例外。"""
 35
 36
 37def format_dimension(dim: Dimension) -> str:
 38    """次元の指数を ``L^1 M^1 T^-2`` のような文字列に変換する。"""
 39    symbols = ("L", "M", "T", "I")
 40    parts = [f"{s}^{e}" for s, e in zip(symbols, dim) if e != 0]
 41    return " ".join(parts) if parts else "無次元"
 42
 43
 44@dataclass(frozen=True)
 45class Quantity:
 46    """SI 単位で表した数値と、その次元の組。"""
 47
 48    value: float
 49    dim: Dimension
 50
 51    def __add__(self, other: Quantity) -> Quantity:
 52        if self.dim != other.dim:
 53            raise DimensionError(
 54                f"次元が異なる量は足せません: {format_dimension(self.dim)}"
 55                f" + {format_dimension(other.dim)}")
 56        return Quantity(self.value + other.value, self.dim)
 57
 58    def __mul__(self, other: Quantity) -> Quantity:
 59        dim = tuple(a + b for a, b in zip(self.dim, other.dim))
 60        return Quantity(self.value * other.value, _as_dimension(dim))
 61
 62    def __truediv__(self, other: Quantity) -> Quantity:
 63        dim = tuple(a - b for a, b in zip(self.dim, other.dim))
 64        return Quantity(self.value / other.value, _as_dimension(dim))
 65
 66    def __str__(self) -> str:
 67        return f"{self.value:.6g} [{format_dimension(self.dim)}]"
 68
 69
 70def _as_dimension(values: tuple[int, ...]) -> Dimension:
 71    """長さ 4 のタプルを Dimension 型として返す。"""
 72    length, mass, time, current = values
 73    return (length, mass, time, current)
 74
 75
 76def main() -> None:
 77    mass = Quantity(1200.0, MASS)            # 自動車の質量 1200 kg
 78    distance = Quantity(100.0, LENGTH)       # 100 m
 79    duration = Quantity(4.0, TIME)           # 4 s
 80
 81    # 速度 = 距離 / 時間、運動エネルギー = 質量 × 速度^2 / 2
 82    velocity = distance / duration
 83    half = Quantity(0.5, DIMENSIONLESS)
 84    energy = half * mass * velocity * velocity
 85    print(f"速度: {velocity}")
 86    print(f"運動エネルギー: {energy}  (L^2 M^1 T^-2 = J)")
 87
 88    # 次元の異なる量を足すとエラーになる
 89    try:
 90        print(distance + duration)
 91    except DimensionError as error:
 92        print(f"エラーを検出しました: {error}")
 93
 94    # 同じ次元でも単位を取り違えると検出できない
 95    impulse_lbf_s = 100.0  # 本当は lbf·s で計算した値
 96    impulse_n_s = impulse_lbf_s * NEWTON_PER_POUND_FORCE
 97    print(f"{impulse_lbf_s} lbf・s = {impulse_n_s:.1f} N・s")
 98    print("lbf・s の値を N・s と解釈すると、"
 99          f"力積を {NEWTON_PER_POUND_FORCE:.2f} 分の 1 に見積もります。")
100
101
102if __name__ == "__main__":
103    main()

ch03_pendulum.py

  1"""第 3 章: 振り子の非線形モデルと線形近似モデルを比較する。
  2
  3振り子の運動方程式 θ'' = -(g/L) sin θ を、sin θ ≈ θ と近似した
  4線形モデルと比べ、振れ幅が大きくなると近似がどれだけ外れるかを
  5確かめます。
  6
  7* 周期: 振れ幅ごとに、厳密な周期と線形モデルの周期を表で比較する
  8* 波形: 振れ幅 60° のときの角度の時間変化を図にする (ch03_pendulum.png)
  9
 10実行例::
 11
 12    python ch03_pendulum.py --outdir output
 13"""
 14
 15from __future__ import annotations
 16
 17import argparse
 18import math
 19from pathlib import Path
 20
 21import matplotlib
 22import numpy as np
 23from matplotlib import font_manager
 24from matplotlib.figure import Figure
 25from numpy.typing import NDArray
 26
 27G = 9.80665  # 標準重力加速度 [m/s^2]
 28LENGTH = 1.0  # 振り子の長さ [m]
 29
 30JAPANESE_FONTS = ["Yu Gothic", "Meiryo", "Hiragino Sans", "Noto Sans CJK JP"]
 31
 32
 33def setup_font() -> None:
 34    """図の文字に、見つかった日本語フォントを使う。"""
 35    names = {font.name for font in font_manager.fontManager.ttflist}
 36    for name in JAPANESE_FONTS:
 37        if name in names:
 38            matplotlib.rcParams["font.family"] = name
 39            break
 40    matplotlib.rcParams["axes.unicode_minus"] = False
 41
 42
 43def agm(a: float, b: float) -> float:
 44    """算術幾何平均 (AGM) を返す。"""
 45    while abs(a - b) > 1e-15 * a:
 46        a, b = (a + b) / 2, math.sqrt(a * b)
 47    return a
 48
 49
 50def linear_period() -> float:
 51    """線形モデルの周期 2π√(L/g) を返す。振れ幅によらない。"""
 52    return 2 * math.pi * math.sqrt(LENGTH / G)
 53
 54
 55def exact_period(theta0: float) -> float:
 56    """振れ幅 theta0 [rad] のときの厳密な周期を返す。
 57
 58    厳密な周期は第 1 種完全楕円積分で表され、算術幾何平均を使うと
 59    T = 2π√(L/g) / AGM(1, cos(θ0/2)) で計算できます。
 60    """
 61    return linear_period() / agm(1.0, math.cos(theta0 / 2))
 62
 63
 64def simulate(theta0: float, t_end: float,
 65             dt: float) -> tuple[NDArray[np.float64], NDArray[np.float64]]:
 66    """非線形モデルを 4 次のルンゲ・クッタ法で解き、時刻と角度を返す。"""
 67
 68    def deriv(state: NDArray[np.float64]) -> NDArray[np.float64]:
 69        theta, omega = state
 70        return np.array([omega, -(G / LENGTH) * math.sin(theta)])
 71
 72    steps = int(round(t_end / dt))
 73    times = np.linspace(0.0, steps * dt, steps + 1)
 74    angles = np.empty(steps + 1)
 75    state = np.array([theta0, 0.0])
 76    angles[0] = theta0
 77    for i in range(steps):
 78        k1 = deriv(state)
 79        k2 = deriv(state + dt / 2 * k1)
 80        k3 = deriv(state + dt / 2 * k2)
 81        k4 = deriv(state + dt * k3)
 82        state = state + dt / 6 * (k1 + 2 * k2 + 2 * k3 + k4)
 83        angles[i + 1] = state[0]
 84    return times, angles
 85
 86
 87def print_period_table() -> None:
 88    """振れ幅ごとの周期の比較表を表示する。"""
 89    print("振れ幅[°]  厳密な周期[s]  線形モデル[s]  誤差[%]")
 90    for degree in (5, 10, 20, 30, 45, 60, 90):
 91        exact = exact_period(math.radians(degree))
 92        linear = linear_period()
 93        error = (linear - exact) / exact * 100
 94        print(f"{degree:8d}  {exact:13.4f}  {linear:13.4f}  {error:7.2f}")
 95
 96
 97def plot_waveform(path: Path) -> None:
 98    """振れ幅 60° のときの角度の時間変化を、2 つのモデルで比較する。"""
 99    theta0 = math.radians(60)
100    times, angles = simulate(theta0, t_end=6.0, dt=0.001)
101    omega0 = math.sqrt(G / LENGTH)
102    linear = theta0 * np.cos(omega0 * times)
103
104    fig = Figure(figsize=(7, 4), layout="constrained")
105    ax = fig.add_subplot()
106    ax.plot(times, np.degrees(angles), label=r"非線形モデル ($\sin\theta$)")
107    ax.plot(times, np.degrees(linear), "--", label=r"線形モデル ($\theta$)")
108    ax.set_xlabel("時刻 [s]")
109    ax.set_ylabel("角度 [度]")
110    ax.set_title("振れ幅 60 度では、線形モデルの位相が次第に進む")
111    ax.grid(alpha=0.3)
112    ax.legend(loc="upper center", bbox_to_anchor=(0.5, -0.18), ncols=2)
113    fig.savefig(path, dpi=120)
114
115
116def main() -> None:
117    parser = argparse.ArgumentParser(description=__doc__)
118    parser.add_argument("--outdir", type=Path,
119                        default=Path(__file__).resolve().parent / "output",
120                        help="画像の保存先フォルダー")
121    args = parser.parse_args()
122    outdir: Path = args.outdir
123    outdir.mkdir(parents=True, exist_ok=True)
124
125    setup_font()
126    print_period_table()
127    plot_waveform(outdir / "ch03_pendulum.png")
128    print(f"図を {outdir} に保存しました。")
129
130
131if __name__ == "__main__":
132    main()

ch03_state_machine.py

  1"""第 3 章: 状態機械で電気ポットの制御をモデル化する。
  2
  3電気ポットの制御を、状態 (待機・加熱・保温・異常) と、状態を変える
  4きっかけとなるイベントの組で表します。状態とイベントの組から次の
  5状態を決める表 (状態遷移表) を使うと、
  6
  7* 想定していないイベントを確実に検出できる
  8* 「異常」からはリセット以外で抜けられない、といった性質を
  9  機械的に確かめられる
 10
 11ことを示します。
 12
 13実行例::
 14
 15    python ch03_state_machine.py
 16"""
 17
 18from __future__ import annotations
 19
 20from enum import Enum
 21
 22
 23class State(Enum):
 24    """電気ポットの状態。"""
 25
 26    IDLE = "待機"
 27    HEATING = "加熱"
 28    KEEP_WARM = "保温"
 29    ERROR = "異常"
 30
 31
 32class Event(Enum):
 33    """状態を変えるきっかけとなるイベント。"""
 34
 35    START = "沸騰ボタン"
 36    BOILED = "沸騰を検出"
 37    COOLED = "設定温度を下回った"
 38    DRY = "空だきを検出"
 39    SENSOR_FAULT = "センサーの異常"
 40    RESET = "リセット"
 41
 42
 43# 状態遷移表: (現在の状態, イベント) -> 次の状態
 44TRANSITIONS: dict[tuple[State, Event], State] = {
 45    (State.IDLE, Event.START): State.HEATING,
 46    (State.HEATING, Event.BOILED): State.KEEP_WARM,
 47    (State.KEEP_WARM, Event.COOLED): State.HEATING,
 48    (State.KEEP_WARM, Event.START): State.HEATING,
 49    (State.ERROR, Event.RESET): State.IDLE,
 50}
 51
 52# どの状態でも、安全に関わるイベントでは異常に移る
 53for _state in (State.IDLE, State.HEATING, State.KEEP_WARM):
 54    TRANSITIONS[(_state, Event.DRY)] = State.ERROR
 55    TRANSITIONS[(_state, Event.SENSOR_FAULT)] = State.ERROR
 56
 57
 58class InvalidTransition(Exception):
 59    """状態遷移表にないイベントを受け取ったときに送出する例外。"""
 60
 61
 62class KettleController:
 63    """状態遷移表に従って動く電気ポットのコントローラー。"""
 64
 65    def __init__(self) -> None:
 66        self.state = State.IDLE
 67
 68    def handle(self, event: Event) -> State:
 69        """イベントを処理し、遷移後の状態を返す。"""
 70        key = (self.state, event)
 71        if key not in TRANSITIONS:
 72            raise InvalidTransition(
 73                f"状態「{self.state.value}」では"
 74                f"イベント「{event.value}」を受け付けません")
 75        self.state = TRANSITIONS[key]
 76        return self.state
 77
 78    def heater_on(self) -> bool:
 79        """ヒーターに通電するかどうかを返す (加熱中だけ通電する)。"""
 80        return self.state is State.HEATING
 81
 82
 83def check_properties() -> None:
 84    """状態遷移表が満たすべき性質を確かめる。"""
 85    # 異常状態から抜け出せるのはリセットだけ
 86    exits = [event for (state, event) in TRANSITIONS
 87             if state is State.ERROR]
 88    assert exits == [Event.RESET], exits
 89    # 空だきを検出したら、どの状態からでも異常に移る
 90    for state in State:
 91        if state is not State.ERROR:
 92            assert TRANSITIONS[(state, Event.DRY)] is State.ERROR
 93    print("状態遷移表の性質を確認しました。")
 94
 95
 96def main() -> None:
 97    check_properties()
 98    controller = KettleController()
 99    scenario = [Event.START, Event.BOILED, Event.COOLED, Event.BOILED,
100                Event.DRY, Event.START, Event.RESET]
101    for event in scenario:
102        before = controller.state
103        try:
104            after = controller.handle(event)
105        except InvalidTransition as error:
106            print(f"拒否: {error}")
107            continue
108        heater = "ON" if controller.heater_on() else "OFF"
109        print(f"{before.value:3s} --[{event.value}]--> {after.value:3s}"
110              f"  ヒーター {heater}")
111
112
113if __name__ == "__main__":
114    main()

ch03_overfitting.py

  1"""第 3 章: 過学習を確かめ、交差検証でモデルの複雑さを選ぶ。
  2
  3真の関係が y = sin(2πx) で、測定に雑音が乗った模擬データに、
  4次数の異なる多項式を当てはめます。
  5
  6* 訓練データでの誤差は、次数を上げるほど小さくなる
  7* 訓練に使っていない検証データでの誤差は、ある次数を超えると
  8  大きくなる (過学習)
  9* 5 分割交差検証で、適切な次数を選ぶ
 10
 11図は ch03_overfitting.png として保存します。
 12
 13実行例::

 44@dataclass(frozen=True)
 45class LineFit:
 46    """直線 y = a + b x の当てはめ結果。"""
 47
 48    a: float  # 切片
 49    b: float  # 傾き
 50    u_a: float  # 切片の標準不確かさ
 51    u_b: float  # 傾きの標準不確かさ
 52    residual_std: float  # 残差の標準偏差

 54
 55def fit_line(x: NDArray[np.float64], y: NDArray[np.float64]) -> LineFit:
 56    """最小二乗法で直線を当てはめる。"""
 57    n = len(x)
 58    x_mean = float(np.mean(x))
 59    y_mean = float(np.mean(y))
 60    sxx = float(np.sum((x - x_mean) ** 2))
 61    sxy = float(np.sum((x - x_mean) * (y - y_mean)))
 62
 63    b = sxy / sxx
 64    a = y_mean - b * x_mean
 65
 66    # 残差の標準偏差 (自由度は n - 2)
 67    residuals = y - (a + b * x)
 68    s = math.sqrt(float(np.sum(residuals ** 2)) / (n - 2))
 69
 70    u_b = s / math.sqrt(sxx)
 71    u_a = s * math.sqrt(1 / n + x_mean ** 2 / sxx)
 72    return LineFit(a, b, u_a, u_b, s)

25def measure(func: Callable[[], object], repeat: int) -> list[float]:

 48    """解の公式をそのまま使って 2 次方程式の実数解を求める。"""

 54                           c: float) -> tuple[float, float]:
 55    """桁落ちを避けて 2 次方程式の実数解を求める。

22def forward_difference(f: Function, x: float, h: float) -> float:

27def central_difference(f: Function, x: float, h: float) -> float:
28    """中心差分 (f(x + h) - f(x - h)) / 2h。誤差は h^2 に比例する。"""
29    return (f(x + h) - f(x - h)) / (2 * h)

32def trapezoid(f: Function, a: float, b: float, n: int) -> float:
33    """台形公式で区間 [a, b] を n 等分して積分する。"""
34    h = (b - a) / n
35    inner = sum(f(a + i * h) for i in range(1, n))
36    return h * (f(a) / 2 + inner + f(b) / 2)

39def simpson(f: Function, a: float, b: float, n: int) -> float:

44    odd = sum(f(a + i * h) for i in range(1, n, 2))
45    even = sum(f(a + i * h) for i in range(2, n, 2))
46    return h / 3 * (f(a) + 4 * odd + 2 * even + f(b))

 48def euler_step(f: RightHandSide, t: float, y: Vector, h: float) -> Vector:

 62    k1 = f(t, y)
 63    k2 = f(t + h / 2, y + h / 2 * k1)
 64    k3 = f(t + h / 2, y + h / 2 * k2)

 3「リストに重複する値があるか」を調べる 3 つの方法を比べます。

 6* 並べ替え: 並べ替えてから隣どうしを比べる。計算量は O(n log n)
 7* 集合: 集合 (ハッシュ表) に入れながら調べる。計算量は O(n)
 8
 9データの数 n を 2 倍にしたときに、処理時間が何倍になるかを測定します。
10

17

20from collections.abc import Callable
21

23def has_duplicate_bruteforce(values: list[int]) -> bool:
24    """すべての組を比べる (O(n^2))。"""
25    n = len(values)
26    for i in range(n):
27        for j in range(i + 1, n):
28            if values[i] == values[j]:
29                return True
30    return False
31
32
33def has_duplicate_sort(values: list[int]) -> bool:
34    """並べ替えてから隣どうしを比べる (O(n log n))。"""
35    ordered = sorted(values)
36    return any(a == b for a, b in zip(ordered, ordered[1:]))
37
38
39def has_duplicate_set(values: list[int]) -> bool:
40    """集合に入れながら調べる (O(n))。"""
41    seen: set[int] = set()
42    for value in values:
43        if value in seen:
44            return True
45        seen.add(value)
46    return False
47
48
49def measure(func: Callable[[list[int]], bool], values: list[int]) -> float:
50    """func の処理時間 [ms] を 3 回測定し、最小値を返す。"""
51    best = float("inf")
52    for _ in range(3):
53        start = time.perf_counter()
54        func(values)
55        best = min(best, time.perf_counter() - start)
56    return best * 1000
57
58
59def main() -> None:
60    methods = {
61        "総当たり": has_duplicate_bruteforce,
62        "並べ替え": has_duplicate_sort,
63        "集合": has_duplicate_set,
64    }
65    sizes = [500, 1000, 2000, 4000]
66    rng = random.Random(0)
67    print("データ数  " + "  ".join(f"{name:>12}" for name in methods)
68          + "   (単位: ms、括弧内は n が半分のときに対する倍率)")
69    previous: dict[str, float] = {}
70    for n in sizes:
71        # 重複のないデータ (最後まで調べる最悪の場合)
72        values = rng.sample(range(10 * n), n)
73        cells = []
74        for name, func in methods.items():
75            elapsed = measure(func, values)
76            ratio = f"(×{elapsed / previous[name]:4.1f})" \
77                if name in previous else "        "
78            cells.append(f"{elapsed:7.2f}{ratio}")
79            previous[name] = elapsed
80        print(f"{n:8d}  " + "  ".join(cells))

 44def second_order(t: NDArray[np.float64], zeta: float,
 45                 omega_n: float) -> NDArray[np.float64]:
 46    """2 次系の単位ステップ応答を数値積分 (RK4) で求める。"""
 47    dt = float(t[1] - t[0])
 48
 49    def deriv(state: NDArray[np.float64]) -> NDArray[np.float64]:
 50        y, v = state
 51        acc = omega_n ** 2 * (1.0 - y) - 2 * zeta * omega_n * v
 52        return np.array([v, acc])

 59        k2 = deriv(state + dt / 2 * k1)
 60        k3 = deriv(state + dt / 2 * k2)
 61        k4 = deriv(state + dt * k3)

 53@dataclass
 54class PidController:
 55    """離散時間の PID コントローラー。"""
 56
 57    kp: float  # 比例ゲイン
 58    ki: float  # 積分ゲイン
 59    kd: float  # 微分ゲイン
 60    dt: float  # 計算周期 [s]
 61    integral: float = 0.0
 62    previous_measurement: float | None = None
 63
 64    def update(self, setpoint: float, measurement: float) -> float:
 65        """目標値と測定値から操作量を計算する。"""
 66        error = setpoint - measurement
 67        self.integral += error * self.dt
 68
 69        # 微分は偏差ではなく測定値に対して行う。目標値が急に変わった
 70        # ときに操作量が跳ね上がる現象 (微分キック) を避けるため。
 71        if self.previous_measurement is None:
 72            derivative = 0.0
 73        else:
 74            derivative = -(measurement - self.previous_measurement) / self.dt
 75        self.previous_measurement = measurement
 76
 77        return self.kp * error + self.ki * self.integral + self.kd * derivative

 80def simulate(controller: PidController) -> tuple[Array, Array]:
 81    """閉ループ系をシミュレーションし、時刻と出力を返す。"""
 82    steps = int(round(T_END / CONTROL_PERIOD))
 83    times = np.arange(steps + 1, dtype=np.float64) * CONTROL_PERIOD
 84    outputs = np.empty(steps + 1)
 85    x1 = x2 = 0.0  # 1 つ目と 2 つ目の要素の出力
 86    h = CONTROL_PERIOD / PLANT_SUBSTEPS
 87    for i in range(steps + 1):
 88        outputs[i] = x2
 89        u = controller.update(SETPOINT, x2)
 90        # 次の計算周期まで操作量 u を一定に保ち、制御対象を進める
 91        for _ in range(PLANT_SUBSTEPS):
 92            x1 += h * (u - x1) / TAU1
 93            x2 += h * (x1 - x2) / TAU2
 94    return times, outputs

 99    final = float(np.mean(outputs[-100:]))
100    overshoot = max(0.0, (float(np.max(outputs)) - final) / final * 100)
101    # 最終値の ±2 % の範囲に入ったまま出なくなる時刻を整定時間とする
102    outside = np.nonzero(np.abs(outputs - final) > 0.02 * abs(final))[0]
103    settling = float(times[outside[-1] + 1]) if outside.size else 0.0
104    error = SETPOINT - final
105    return (f"行き過ぎ量 {overshoot:5.1f} %, 整定時間 {settling:5.2f} s, "
106            f"定常偏差 {error:6.3f}")

 1"""第 6 章: 周期処理のタイミングのずれ (ジッターとドリフト) を測定する。
 2
 3制御プログラムは、一定の周期で計算を繰り返します。周期を作る方法を
 42 つ比べます。
 5
 6* 方法 1: 処理の後に「周期と同じ時間だけ待つ」
 7  処理時間の分だけ周期が延び、ずれが積み重なる (ドリフト)
 8* 方法 2: 「次に実行すべき時刻」まで待つ
 9  1 回ごとのばらつき (ジッター) はあるが、ずれは積み重ならない
10
11汎用の OS の上では、どちらの方法でも待ち時間はある程度ばらつきます。
12
13実行例::
14
15    python ch06_jitter.py --period 10 --count 200
16"""
17
18from __future__ import annotations
19

21import statistics
22import time
23
24
25def work() -> None:
26    """制御の計算の代わりに、約 2 ms かかる処理を行う。"""
27    end = time.perf_counter() + 0.002
28    while time.perf_counter() < end:
29        pass
30
31
32def run_fixed_sleep(period: float, count: int) -> list[float]:
33    """方法 1: 処理の後に周期と同じ時間だけ待つ。開始時刻の列を返す。"""
34    starts = []
35    for _ in range(count):
36        starts.append(time.perf_counter())
37        work()
38        time.sleep(period)
39    return starts
40
41
42def run_deadline(period: float, count: int) -> list[float]:
43    """方法 2: 次に実行すべき時刻まで待つ。開始時刻の列を返す。"""
44    starts = []
45    next_time = time.perf_counter()
46    for _ in range(count):
47        starts.append(time.perf_counter())
48        work()
49        next_time += period
50        delay = next_time - time.perf_counter()
51        if delay > 0:
52            time.sleep(delay)
53    return starts
54
55
56def report(name: str, starts: list[float], period: float) -> None:
57    """周期の統計と、累積のずれを表示する。"""
58    intervals = [(b - a) * 1000 for a, b in zip(starts, starts[1:])]
59    drift = (starts[-1] - starts[0] - period * (len(starts) - 1)) * 1000
60    print(f"{name}")
61    print(f"  周期の平均   : {statistics.mean(intervals):7.2f} ms")
62    print(f"  周期の標準偏差: {statistics.stdev(intervals):7.2f} ms")
63    print(f"  周期の最大   : {max(intervals):7.2f} ms")
64    print(f"  累積のずれ   : {drift:7.1f} ms")
65
66
67def main() -> None:
68    parser = argparse.ArgumentParser(description=__doc__)
69    parser.add_argument("--period", type=float, default=10.0,
70                        help="目標の周期 [ms]")
71    parser.add_argument("--count", type=int, default=200,
72                        help="繰り返す回数")
73    args = parser.parse_args()
74    period = args.period / 1000
75
76    print(f"目標の周期 {args.period} ms、{args.count} 回")
77    report("方法 1: 毎回、周期と同じ時間だけ待つ",
78           run_fixed_sleep(period, args.count), period)
79    report("方法 2: 次に実行すべき時刻まで待つ",

21def quantize(x: Array, bits: int) -> Array:
22    """範囲 [-1, 1) の信号を bits ビットで量子化する (中央値に丸める)。"""
23    levels = 2 ** bits
24    step = 2.0 / levels  # 量子化ステップ (1 LSB の大きさ)
25    codes = np.clip(np.floor((x + 1.0) / step), 0, levels - 1)
26    result: Array = -1.0 + (codes + 0.5) * step
27    return result

30def sqnr_db(bits: int, samples: int = 100_000) -> float:
31    """フルスケールの正弦波を量子化したときの SQNR [dB] を返す。"""
32    n = np.arange(samples, dtype=np.float64)
33    # 周期がサンプル数と公約数を持たないように、無理数比の周波数にする
34    x = np.sin(2 * np.pi * n / (100 * math.sqrt(2)))
35    error = quantize(x, bits) - x
36    signal_power = float(np.mean(x ** 2))
37    noise_power = float(np.mean(error ** 2))
38    return 10 * math.log10(signal_power / noise_power)

 57def moving_average(x: Array, window: int) -> Array:
 58    """過去 window 点の平均を出力する (因果的な) 移動平均フィルター。
 59
 60    最初の window - 1 点は、そろっているデータだけで平均します。
 61    """
 62    cumsum = np.cumsum(np.insert(x, 0, 0.0))
 63    out = np.empty_like(x)
 64    for i in range(x.size):
 65        start = max(0, i + 1 - window)
 66        out[i] = (cumsum[i + 1] - cumsum[start]) / (i + 1 - start)
 67    return out

44def total_scores(weights: dict[str, float]) -> dict[str, float]:
45    """重みに従って各案の総合点を計算する。"""
46    total = sum(weights.values())
47    return {
48        option: sum(weights[c] * s for c, s in scores.items()) / total

50    }

54    """総合点が最も高い案を返す。"""

56    return max(totals, key=lambda option: totals[option])

60    """各重みを ±delta だけ変えた組み合わせで、1 位になった回数を数える。"""

62    criteria = list(WEIGHTS)
63    for signs in itertools.product((-1, 0, 1), repeat=len(criteria)):
64        weights = {
65            c: max(0.0, WEIGHTS[c] + sign * delta)
66            for c, sign in zip(criteria, signs)
67        }

 51def evaluate(b: Array, h: Array) -> tuple[Array, Array, Array]:
 52    """幅 b [m] と高さ h [m] から、質量・たわみ・最大応力を計算する。"""
 53    mass = DENSITY * b * h * LENGTH
 54    second_moment = b * h ** 3 / 12  # 断面二次モーメント
 55    deflection = FORCE * LENGTH ** 3 / (3 * YOUNG * second_moment)
 56    stress = FORCE * LENGTH / (b * h ** 2 / 6)  # 曲げモーメント / 断面係数
 57    return mass, deflection, stress

 60def pareto_mask(f1: Array, f2: Array) -> NDArray[np.bool_]:
 61    """2 つの目的 (どちらも小さいほど良い) のパレート最適解の印を返す。"""
 62    order = np.argsort(f1)
 63    mask = np.zeros(f1.size, dtype=np.bool_)
 64    best_f2 = np.inf
 65    # f1 の小さい順に見ていき、それまでより f2 が小さい案だけが残る
 66    for index in order:
 67        if f2[index] < best_f2:
 68            mask[index] = True

41def confidence_interval(data: list[float]) -> tuple[float, float]:
42    """母平均の両側 95 % 信頼区間を返す。"""
43    n = len(data)
44    mean = statistics.mean(data)
45    standard_error = statistics.stdev(data) / math.sqrt(n)
46    t = T_975.get(n - 1, 1.96)  # 表にない自由度では正規分布で近似する
47    return mean - t * standard_error, mean + t * standard_error

50def permutation_test(a: list[float], b: list[float], trials: int,
51                     seed: int) -> float:
52    """平均の差の両側並べ替え検定を行い、p 値を返す。
53
54    「2 つのグループに差はない」と仮定すると、測定値のラベル
55    (改修前・改修後) を入れ替えても差の分布は変わらないはずです。
56    ラベルをランダムに入れ替えたときに、実際の差以上の差が
57    出る割合を p 値とします。
58    """
59    observed = abs(statistics.mean(a) - statistics.mean(b))
60    pooled = a + b
61    rng = random.Random(seed)
62    count = 0
63    for _ in range(trials):
64        rng.shuffle(pooled)
65        diff = abs(statistics.mean(pooled[:len(a)])
66                   - statistics.mean(pooled[len(a):]))
67        if diff >= observed:
68            count += 1
69    return (count + 1) / (trials + 1)

  1"""第 13 章: S-N 曲線とマイナー則で、繰り返し荷重による疲労寿命を見積もる。

  6  N = N_ref × (S / S_ref)^(-k)。疲労限度より小さい応力では

ch14_fault_tree.py

 1"""第 14 章: フォールトツリーから頂上事象の発生確率を計算する。
 2
 3電気ポットの「空だき」による過熱を頂上事象とする、簡単な
 4フォールトツリーを考えます。
 5
 6* 頂上事象: 過熱 = 空だき状態 AND 保護機能がすべて失敗
 7* 空だき状態 = 水位センサーの故障 OR 利用者が水を入れ忘れる
 8* 保護機能の失敗 = 温度ヒューズの故障 AND 制御ソフトウェアの
 9  温度監視の失敗
10
11各基本事象は互いに独立と仮定し、AND ゲートは確率の積、
12OR ゲートは 1 - (1 - p1)(1 - p2)… で計算します。
13
14実行例::
15
16    python ch14_fault_tree.py
17"""
18
19from __future__ import annotations
20
21import math
22from dataclasses import dataclass, field
23
24
25@dataclass
26class Event:
27    """フォールトツリーの事象。
28
29    基本事象は probability を持ち、中間事象・頂上事象は gate ("AND" または
30    "OR") と下位の事象 children を持ちます。
31    """
32
33    name: str
34    probability: float | None = None
35    gate: str | None = None
36    children: list[Event] = field(default_factory=list)
37
38    def evaluate(self) -> float:
39        """この事象の発生確率を計算する。"""
40        if self.probability is not None:
41            return self.probability
42        values = [child.evaluate() for child in self.children]
43        if self.gate == "AND":
44            return math.prod(values)
45        if self.gate == "OR":
46            return 1 - math.prod(1 - p for p in values)
47        raise ValueError(f"未知のゲートです: {self.gate}")
48
49    def show(self, indent: int = 0) -> None:
50        """木構造と各事象の確率を表示する。"""
51        label = f"[{self.gate}] " if self.gate else ""
52        print(f"{'  ' * indent}{label}{self.name}: {self.evaluate():.2e}")
53        for child in self.children:
54            child.show(indent + 1)
55
56
57def build_tree(software_failure: float) -> Event:
58    """フォールトツリーを組み立てる。確率は 1 回の使用あたりの値。"""
59    dry = Event("空だき状態", gate="OR", children=[
60        Event("水位センサーの故障", probability=1e-4),
61        Event("水の入れ忘れ", probability=1e-2),
62    ])
63    protection = Event("保護機能の失敗", gate="AND", children=[
64        Event("温度ヒューズの故障", probability=1e-4),
65        Event("温度監視ソフトウェアの失敗", probability=software_failure),
66    ])
67    return Event("過熱", gate="AND", children=[dry, protection])
68
69
70def main() -> None:
71    top = build_tree(software_failure=1e-3)
72    top.show()
73
74    print("\n温度監視ソフトウェアの失敗確率を変えたときの頂上事象の確率")
75    for p in (1e-1, 1e-2, 1e-3, 1e-4):
76        print(f"  {p:.0e} -> {build_tree(p).evaluate():.2e}")
77    print("\n保護機能をソフトウェアだけにした場合 (温度ヒューズなし)")
78    dry = top.children[0].evaluate()
79    print(f"  空だき状態 {dry:.2e} × ソフトウェアの失敗 1e-03 "
80          f"= {dry * 1e-3:.2e}")
