Tiny の構文解析器（エラー回復付き）。（第 5 章）

tiny_parser.py をダウンロード

  1"""Tiny 言語の構文解析器（第 5 章）。
  2
  3文は再帰下降構文解析で、式は Pratt 構文解析で解析し、抽象構文木を作る。
  4文の途中で構文エラーを見つけたら、次の文の先頭まで読み飛ばして解析を続け
  5（パニックモードのエラー回復）、見つかったエラーをまとめて報告する。
  6"""
  7
  8from __future__ import annotations
  9
 10import sys
 11
 12from tiny_ast import (
 13    BOOL, INT, STR, VOID, ArrayLit, ArrayType, Assign, Binary, Block, BoolLit,
 14    Call, Expr, ExprStmt, FuncDef, If, Index, IntLit, Let, Name, Param,
 15    Program, Return, Stmt, StrLit, Type, Unary, While, dump)
 16from tiny_lexer import Token, TinyError, TokenKind, tokenize
 17
 18
 19class ParseError(TinyError):
 20    """構文エラー。all_errors に見つかったすべてのエラーを持つ。"""
 21
 22    def __init__(self, message: str, line: int = 0, column: int = 0) -> None:
 23        super().__init__(message, line, column)
 24        self.all_errors: list[ParseError] = [self]
 25
 26
 27# 二項演算子の結合力（左、右）。すべて左結合。
 28BINARY_POWER: dict[TokenKind, tuple[int, int]] = {
 29    TokenKind.OR: (1, 2),
 30    TokenKind.AND: (3, 4),
 31    TokenKind.EQ: (5, 6),
 32    TokenKind.NE: (5, 6),
 33    TokenKind.LT: (7, 8),
 34    TokenKind.LE: (7, 8),
 35    TokenKind.GT: (7, 8),
 36    TokenKind.GE: (7, 8),
 37    TokenKind.PLUS: (9, 10),
 38    TokenKind.MINUS: (9, 10),
 39    TokenKind.STAR: (11, 12),
 40    TokenKind.SLASH: (11, 12),
 41    TokenKind.PERCENT: (11, 12),
 42}
 43PREFIX_POWER = 13
 44POSTFIX_POWER = 15  # 関数呼び出し f(...) と添字 a[...]
 45
 46TYPE_NAMES: dict[str, Type] = {"int": INT, "bool": BOOL, "str": STR}
 47
 48# エラー回復で読み飛ばしを止める、文の先頭になるキーワード
 49STATEMENT_START = (TokenKind.LET, TokenKind.IF, TokenKind.WHILE,
 50                   TokenKind.RETURN, TokenKind.FN)
 51
 52# 式の先頭になり得る字句
 53PREFIX_START = (TokenKind.INT, TokenKind.STRING, TokenKind.TRUE,
 54                TokenKind.FALSE, TokenKind.IDENT, TokenKind.MINUS,
 55                TokenKind.NOT, TokenKind.LPAREN, TokenKind.LBRACKET)
 56
 57
 58class Parser:
 59    def __init__(self, tokens: list[Token]) -> None:
 60        self.tokens = tokens
 61        self.pos = 0
 62        self.errors: list[ParseError] = []
 63
 64    # ------------------------------------------------ 字句の操作
 65
 66    def peek(self, offset: int = 0) -> Token:
 67        index = min(self.pos + offset, len(self.tokens) - 1)
 68        return self.tokens[index]
 69
 70    def at(self, kind: TokenKind) -> bool:
 71        return self.peek().kind is kind
 72
 73    def advance(self) -> Token:
 74        token = self.peek()
 75        if token.kind is not TokenKind.EOF:
 76            self.pos += 1
 77        return token
 78
 79    def accept(self, kind: TokenKind) -> bool:
 80        """次の字句が kind なら読み進めて True を返す。"""
 81        if self.at(kind):
 82            self.advance()
 83            return True
 84        return False
 85
 86    def expect(self, kind: TokenKind, what: str) -> Token:
 87        if not self.at(kind):
 88            raise self.error(f"{what} が必要です")
 89        return self.advance()
 90
 91    def error(self, message: str) -> ParseError:
 92        token = self.peek()
 93        found = "ファイルの終わり" if token.kind is TokenKind.EOF \
 94            else f"{token.text!r}"
 95        return ParseError(f"{message}（{found} があります）",
 96                          token.line, token.column)
 97
 98    def synchronize(self) -> None:
 99        """エラー回復: 文の区切りまで字句を読み飛ばす。"""
100        while not self.at(TokenKind.EOF):
101            if self.accept(TokenKind.SEMICOLON):
102                return
103            if self.peek().kind in (*STATEMENT_START, TokenKind.RBRACE):
104                return
105            self.advance()
106
107    # ------------------------------------------------ プログラムと関数
108
109    def parse_program(self) -> Program:
110        functions: list[FuncDef] = []
111        while not self.at(TokenKind.EOF):
112            try:
113                functions.append(self.parse_function())
114            except ParseError as e:
115                self.errors.append(e)
116                self.advance()
117                while not self.at(TokenKind.EOF) and not self.at(TokenKind.FN):
118                    self.advance()
119        if self.errors:
120            first = self.errors[0]
121            first.all_errors = self.errors
122            raise first
123        return Program(functions, line=1)
124
125    def parse_function(self) -> FuncDef:
126        # function = "fn" IDENT "(" [ param { "," param } ] ")"
127        #            [ "->" type ] block
128        line = self.expect(TokenKind.FN, "関数定義 fn").line
129        name = self.expect(TokenKind.IDENT, "関数名").text
130        self.expect(TokenKind.LPAREN, "(")
131        params: list[Param] = []
132        if not self.at(TokenKind.RPAREN):
133            while True:
134                token = self.expect(TokenKind.IDENT, "引数名")
135                self.expect(TokenKind.COLON, ":")
136                params.append(Param(token.text, self.parse_type(),
137                                    line=token.line))
138                if not self.accept(TokenKind.COMMA):
139                    break
140        self.expect(TokenKind.RPAREN, ")")
141        return_type: Type = VOID
142        if self.accept(TokenKind.ARROW):
143            return_type = self.parse_type()
144        body = self.parse_block()
145        return FuncDef(name, params, return_type, body, line=line)
146
147    def parse_type(self) -> Type:
148        # type = "int" | "bool" | "str" | "[" type "]"
149        if self.accept(TokenKind.LBRACKET):
150            element = self.parse_type()
151            self.expect(TokenKind.RBRACKET, "]")
152            return ArrayType(element)
153        token = self.expect(TokenKind.IDENT, "型名")
154        if token.text not in TYPE_NAMES:
155            raise ParseError(f"未知の型 {token.text} です",
156                             token.line, token.column)
157        return TYPE_NAMES[token.text]
158
159    # ------------------------------------------------ 文
160
161    def parse_block(self) -> Block:
162        # block = "{" { statement } "}"
163        line = self.expect(TokenKind.LBRACE, "{").line
164        stmts: list[Stmt] = []
165        while not self.at(TokenKind.RBRACE) and not self.at(TokenKind.EOF):
166            start = self.pos
167            try:
168                stmts.append(self.parse_statement())
169            except ParseError as e:
170                self.errors.append(e)
171                self.synchronize()
172                if self.pos == start:  # 1 字句も進まなければ強制的に進める
173                    self.advance()
174        self.expect(TokenKind.RBRACE, "}")
175        return Block(stmts, line=line)
176
177    def parse_statement(self) -> Stmt:
178        token = self.peek()
179        line = token.line
180        match token.kind:
181            case TokenKind.LBRACE:
182                return self.parse_block()
183            case TokenKind.LET:
184                # "let" IDENT [ ":" type ] "=" expr ";"
185                self.advance()
186                name = self.expect(TokenKind.IDENT, "変数名").text
187                declared: Type | None = None
188                if self.accept(TokenKind.COLON):
189                    declared = self.parse_type()
190                self.expect(TokenKind.ASSIGN, "=")
191                value = self.parse_expr()
192                self.expect(TokenKind.SEMICOLON, ";")
193                return Let(name, declared, value, line=line)
194            case TokenKind.IF:
195                return self.parse_if()
196            case TokenKind.WHILE:
197                # "while" "(" expr ")" block
198                self.advance()
199                self.expect(TokenKind.LPAREN, "(")
200                cond = self.parse_expr()
201                self.expect(TokenKind.RPAREN, ")")
202                return While(cond, self.parse_block(), line=line)
203            case TokenKind.RETURN:
204                # "return" [ expr ] ";"
205                self.advance()
206                result = None
207                if not self.at(TokenKind.SEMICOLON):
208                    result = self.parse_expr()
209                self.expect(TokenKind.SEMICOLON, ";")
210                return Return(result, line=line)
211        # 代入文または式文: expr [ "=" expr ] ";"
212        expr = self.parse_expr()
213        if self.accept(TokenKind.ASSIGN):
214            if not isinstance(expr, (Name, Index)):
215                raise ParseError("代入の左辺が不正です", token.line,
216                                 token.column)
217            value = self.parse_expr()
218            self.expect(TokenKind.SEMICOLON, ";")
219            return Assign(expr, value, line=line)
220        self.expect(TokenKind.SEMICOLON, ";")
221        return ExprStmt(expr, line=line)
222
223    def parse_if(self) -> If:
224        # "if" "(" expr ")" block [ "else" ( block | if ) ]
225        line = self.expect(TokenKind.IF, "if").line
226        self.expect(TokenKind.LPAREN, "(")
227        cond = self.parse_expr()
228        self.expect(TokenKind.RPAREN, ")")
229        then_body = self.parse_block()
230        else_body: Stmt | None = None
231        if self.accept(TokenKind.ELSE):
232            else_body = (self.parse_if() if self.at(TokenKind.IF)
233                         else self.parse_block())
234        return If(cond, then_body, else_body, line=line)
235
236    # ------------------------------------------------ 式（Pratt 構文解析）
237
238    def parse_expr(self, min_power: int = 0) -> Expr:
239        left = self.parse_prefix()
240        while True:
241            token = self.peek()
242            if token.kind in (TokenKind.LPAREN, TokenKind.LBRACKET):
243                if POSTFIX_POWER < min_power:
244                    return left
245                left = self.parse_postfix(left)
246                continue
247            power = BINARY_POWER.get(token.kind)
248            if power is None or power[0] < min_power:
249                return left
250            self.advance()
251            right = self.parse_expr(power[1])
252            left = Binary(token.text, left, right, line=token.line)
253
254    def parse_postfix(self, left: Expr) -> Expr:
255        token = self.advance()
256        if token.kind is TokenKind.LBRACKET:
257            index = self.parse_expr()
258            self.expect(TokenKind.RBRACKET, "]")
259            return Index(left, index, line=token.line)
260        if not isinstance(left, Name):
261            raise ParseError("関数名以外は呼び出せません", token.line,
262                             token.column)
263        args = self.parse_list(TokenKind.RPAREN, ")")
264        return Call(left.name, args, line=left.line)
265
266    def parse_list(self, close: TokenKind, what: str) -> list[Expr]:
267        """「式 , 式 , ...」を閉じ括弧まで読む（開き括弧は読み済み）。"""
268        items: list[Expr] = []
269        if not self.at(close):
270            items.append(self.parse_expr())
271            while self.accept(TokenKind.COMMA):
272                items.append(self.parse_expr())
273        self.expect(close, what)
274        return items
275
276    def parse_prefix(self) -> Expr:
277        token = self.peek()
278        if token.kind not in PREFIX_START:
279            raise self.error("式が必要です")
280        self.advance()
281        line = token.line
282        match token.kind:
283            case TokenKind.INT:
284                assert isinstance(token.value, int)
285                return IntLit(token.value, line=line)
286            case TokenKind.STRING:
287                assert isinstance(token.value, str)
288                return StrLit(token.value, line=line)
289            case TokenKind.TRUE:
290                return BoolLit(True, line=line)
291            case TokenKind.FALSE:
292                return BoolLit(False, line=line)
293            case TokenKind.IDENT:
294                return Name(token.text, line=line)
295            case TokenKind.MINUS | TokenKind.NOT:
296                operand = self.parse_expr(PREFIX_POWER)
297                return Unary(token.text, operand, line=line)
298            case TokenKind.LPAREN:
299                inner = self.parse_expr()
300                self.expect(TokenKind.RPAREN, ")")
301                return inner
302            case TokenKind.LBRACKET:
303                elements = self.parse_list(TokenKind.RBRACKET, "]")
304                return ArrayLit(elements, line=line)
305        raise AssertionError(token.kind)
306
307
308def parse(source: str) -> Program:
309    """ソースコードを構文解析して抽象構文木を返す。"""
310    return Parser(tokenize(source)).parse_program()
311

313def main() -> None:
314    source = "fn main() {\n    let x = 1 + 2 * 3;\n    print(x);\n}\n"
315    if len(sys.argv) > 1:
316        with open(sys.argv[1], encoding="utf-8") as f:
317            source = f.read()
318    try:
319        print(dump(parse(source)))
320    except ParseError as e:
321        for error in e.all_errors:
322            print(f"構文エラー: {error}")

325if __name__ == "__main__":
326    main()
