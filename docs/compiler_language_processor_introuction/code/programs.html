primes.tiny

primes.tiny をダウンロード

 1// エラトステネスのふるいで n 以下の素数を求める
 2fn primes(n: int) -> [int] {
 3    let sieve: [bool] = [];
 4    let i = 0;
 5    while (i <= n) {
 6        push(sieve, true);
 7        i = i + 1;
 8    }
 9    let result: [int] = [];
10    let p = 2;
11    while (p <= n) {
12        if (sieve[p]) {
13            push(result, p);
14            let m = p * p;
15            while (m <= n) {
16                sieve[m] = false;
17                m = m + p;
18            }
19        }
20        p = p + 1;
21    }
22    return result;
23}
24
25fn main() {
26    let ps = primes(50);
27    print(len(ps), "primes:", ps);
28}

sort.tiny

sort.tiny をダウンロード

 1// 配列の挿入ソート
 2fn sort(a: [int]) {
 3    let i = 1;
 4    while (i < len(a)) {
 5        let key = a[i];
 6        let j = i - 1;
 7        while (j >= 0 && a[j] > key) {
 8            a[j + 1] = a[j];
 9            j = j - 1;
10        }
11        a[j + 1] = key;
12        i = i + 1;
13    }
14}
15
16fn main() {
17    let data = [5, 2, 9, 1, 7, 3];
18    print("before:", data);
19    sort(data);
20    print("after: ", data);
21}

 1// 文字列の反転と回文判定
 2fn reverse(s: str) -> str {
 3    let result = "";
 4    let i = len(s) - 1;
 5    while (i >= 0) {
 6        result = result + s[i];
 7        i = i - 1;
 8    }
 9    return result;
10}
11
12fn is_palindrome(s: str) -> bool {
13    return s == reverse(s);
14}
15
16fn main() {

18    let i = 0;
19    while (i < len(words)) {
20        let w = words[i];
21        print(w, reverse(w), is_palindrome(w));
22        i = i + 1;
23    }
24}
