  1"""Tiny 言語の字句解析器（第 3 章）。
  2
  3ソースコードの文字列を先頭から 1 文字ずつ読み進め、トークンの列に変換する。

  6from __future__ import annotations

  8import sys
  9from dataclasses import dataclass
 10from enum import Enum, auto

 13class TinyError(Exception):
 14    """Tiny 処理系が報告するエラーの基底クラス。"""
 15
 16    def __init__(self, message: str, line: int = 0, column: int = 0) -> None:
 17        super().__init__(message)
 18        self.message = message
 19        self.line = line
 20        self.column = column
 21
 22    def __str__(self) -> str:
 23        if self.column > 0:
 24            return f"{self.line}:{self.column}: {self.message}"
 25        if self.line > 0:
 26            return f"{self.line}: {self.message}"
 27        return self.message

124def is_digit(ch: str) -> bool:
125    """ch が ASCII の数字か（str.isdigit は全角数字なども真になる）。"""
126    return ch.isascii() and ch.isdigit()
127
128
129def is_ident_start(ch: str) -> bool:
130    """ch が識別子の先頭に使える文字（ASCII の英字か _）か。"""
131    return ch == "_" or (ch.isascii() and ch.isalpha())
132
133
134def is_ident_char(ch: str) -> bool:
135    """ch が識別子の 2 文字目以降に使える文字か。"""
136    return is_ident_start(ch) or is_digit(ch)

139@dataclass(frozen=True)
140class Token:
141    """トークン。text はソースコード上の字句そのもの。"""
142
143    kind: TokenKind
144    text: str
145    line: int
146    column: int
147    value: int | str | None = None  # 整数・文字列リテラルの値
148
149
150class Lexer:
151    """ソースコードをトークン列に変換する。"""
152
153    def __init__(self, source: str) -> None:
154        self.source = source
155        self.pos = 0
156        self.line = 1
157        self.column = 1
158
159    def peek(self, offset: int = 0) -> str:
160        """現在位置から offset 文字先の文字を返す（末尾なら空文字列）。"""
161        index = self.pos + offset
162        return self.source[index] if index < len(self.source) else ""
163
164    def advance(self) -> str:
165        """1 文字読み進め、読んだ文字を返す。"""
166        ch = self.source[self.pos]
167        self.pos += 1
168        if ch == "\n":
169            self.line += 1
170            self.column = 1
171        else:
172            self.column += 1
173        return ch
174
175    def skip_whitespace_and_comments(self) -> None:
176        while True:
177            ch = self.peek()
178            if ch in (" ", "\t", "\r", "\n"):
179                self.advance()
180            elif ch == "/" and self.peek(1) == "/":
181                while self.peek() not in ("", "\n"):
182                    self.advance()
183            else:
184                return
185
186    def next_token(self) -> Token:
187        """次のトークンを 1 つ切り出す。"""
188        self.skip_whitespace_and_comments()
189        line, column = self.line, self.column
190        start = self.pos
191        ch = self.peek()
192
193        if ch == "":
194            return Token(TokenKind.EOF, "", line, column)
195
196        if is_digit(ch):
197            while is_digit(self.peek()):
198                self.advance()
199            text = self.source[start:self.pos]
200            return Token(TokenKind.INT, text, line, column, int(text))
201
202        if is_ident_start(ch):
203            while is_ident_char(self.peek()):
204                self.advance()
205            text = self.source[start:self.pos]
206            kind = KEYWORDS.get(text, TokenKind.IDENT)
207            return Token(kind, text, line, column)
208
209        if ch == '"':
210            return self.read_string(line, column)
211
212        two = ch + self.peek(1)
213        if two in TWO_CHAR_SYMBOLS:
214            self.advance()
215            self.advance()
216            return Token(TWO_CHAR_SYMBOLS[two], two, line, column)
217
218        if ch in ONE_CHAR_SYMBOLS:
219            self.advance()
220            return Token(ONE_CHAR_SYMBOLS[ch], ch, line, column)
221
222        raise LexError(f"不正な文字 {ch!r} です", line, column)
223
224    def read_string(self, line: int, column: int) -> Token:
225        """文字列リテラルを読む。エスケープシーケンスを解釈する。"""
226        start = self.pos
227        self.advance()  # 開始の "
228        chars: list[str] = []
229        while True:
230            ch = self.peek()
231            if ch in ("", "\n"):
232                raise LexError("文字列リテラルが閉じていません", line, column)
233            self.advance()
234            if ch == '"':
235                break
236            if ch == "\\":
237                esc = self.peek()
238                if esc not in ESCAPES:
239                    raise LexError(
240                        f"不正なエスケープシーケンス \\{esc} です",
241                        self.line, self.column)
242                self.advance()
243                chars.append(ESCAPES[esc])
244            else:
245                chars.append(ch)
246        text = self.source[start:self.pos]
247        return Token(TokenKind.STRING, text, line, column, "".join(chars))
248
249    def tokenize(self) -> list[Token]:
250        """すべてのトークンを切り出す。末尾は必ず EOF トークンになる。"""
251        tokens: list[Token] = []
252        while True:
253            token = self.next_token()
254            tokens.append(token)
255            if token.kind is TokenKind.EOF:
256                return tokens
257
258
259def tokenize(source: str) -> list[Token]:
260    """ソースコードをトークン列に変換する。"""
261    return Lexer(source).tokenize()

264def main() -> None:
265    source = 'let x = 10 * (y + 2); // コメント\nprint("x=", x >= 3);'
