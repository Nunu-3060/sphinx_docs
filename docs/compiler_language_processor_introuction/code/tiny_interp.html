  3意味解析済みの抽象構文木を再帰的にたどり、そのまま実行する。
  4"""

  6from __future__ import annotations

  8import sys
  9from typing import TextIO

 11from tiny_ast import (
 12    ArrayLit, Assign, Binary, Block, BoolLit, Call, Expr, ExprStmt, FuncDef,
 13    If, Index, IntLit, Let, Name, Program, Return, Stmt, StrLit, Unary, While)
 14from tiny_checker import check
 15from tiny_lexer import TinyError
 16from tiny_parser import parse
 17from tiny_runtime import (
 18    BUILTINS, TinyRuntimeError, Value, binary_op, call_builtin, index_value,
 19    store_index, unary_op)

 22class Environment:
 23    """変数の値を保持する環境。ブロックごとに 1 つ作る。"""
 24
 25    def __init__(self, parent: Environment | None = None) -> None:
 26        self.parent = parent
 27        self.values: dict[str, Value] = {}
 28
 29    def define(self, name: str, value: Value) -> None:
 30        self.values[name] = value
 31
 32    def find(self, name: str) -> Environment:
 33        env: Environment | None = self
 34        while env is not None:
 35            if name in env.values:
 36                return env
 37            env = env.parent
 38        raise TinyRuntimeError(f"変数 {name} が見つかりません")
 39
 40    def get(self, name: str) -> Value:
 41        return self.find(name).values[name]
 42
 43    def set(self, name: str, value: Value) -> None:
 44        self.find(name).values[name] = value
 45
 46
 47class ReturnSignal(Exception):
 48    """return 文の実行を関数呼び出し元まで伝える。"""
 49
 50    def __init__(self, value: Value) -> None:
 51        super().__init__()
 52        self.value = value
 53
 54
 55class Interpreter:
 56    def __init__(self, program: Program, out: TextIO = sys.stdout) -> None:
 57        self.functions = {f.name: f for f in program.functions}
 58        self.out = out
 59
 60    def run(self) -> None:
 61        self.call_function(self.functions["main"], [])
 62
 63    def call_function(self, func: FuncDef, args: list[Value]) -> Value:
 64        env = Environment()
 65        for param, arg in zip(func.params, args):
 66            env.define(param.name, arg)
 67        try:
 68            self.exec_block(func.body, Environment(env))
 69        except ReturnSignal as signal:
 70            return signal.value
 71        return None
 72
 73    # ------------------------------------------------ 文の実行
 74
 75    def exec_block(self, block: Block, env: Environment) -> None:
 76        for stmt in block.stmts:
 77            self.exec_stmt(stmt, env)
 78
 79    def exec_stmt(self, stmt: Stmt, env: Environment) -> None:
 80        match stmt:
 81            case Block():
 82                self.exec_block(stmt, Environment(env))
 83            case Let(name, _, value):
 84                env.define(name, self.eval(value, env))
 85            case Assign(Name(name), value):
 86                env.set(name, self.eval(value, env))
 87            case Assign(Index(target, index), value):
 88                base = self.eval(target, env)
 89                i = self.eval(index, env)
 90                store_index(base, i, self.eval(value, env))
 91            case If(cond, then_body, else_body):
 92                if self.eval(cond, env):
 93                    self.exec_block(then_body, Environment(env))
 94                elif else_body is not None:
 95                    self.exec_stmt(else_body, env)
 96            case While(cond, body):
 97                while self.eval(cond, env):
 98                    self.exec_block(body, Environment(env))
 99            case Return(value):
100                result = None if value is None else self.eval(value, env)
101                raise ReturnSignal(result)
102            case ExprStmt(expr):
103                self.eval(expr, env)
104            case _:
105                raise AssertionError(stmt)
106
107    # ------------------------------------------------ 式の評価
108
109    def eval(self, expr: Expr, env: Environment) -> Value:
110        match expr:
111            case IntLit(value) | BoolLit(value) | StrLit(value):
112                return value
113            case ArrayLit(elements):
114                return [self.eval(e, env) for e in elements]
115            case Name(name):
116                return env.get(name)
117            case Index(target, index):
118                return index_value(self.eval(target, env),
119                                   self.eval(index, env))
120            case Unary(op, operand):
121                return unary_op(op, self.eval(operand, env))
122            case Binary("&&", left, right):
123                return bool(self.eval(left, env)) and \
124                    bool(self.eval(right, env))
125            case Binary("||", left, right):
126                return bool(self.eval(left, env)) or \
127                    bool(self.eval(right, env))
128            case Binary(op, left, right):
129                return binary_op(op, self.eval(left, env),
130                                 self.eval(right, env))
131            case Call(callee, args):
132                values = [self.eval(a, env) for a in args]
133                if callee in BUILTINS:
134                    return call_builtin(callee, values, self.out)
135                return self.call_function(self.functions[callee], values)
136        raise AssertionError(expr)
137
138
139def run_source(source: str, out: TextIO = sys.stdout) -> None:
140    """ソースコードを解析し、インタプリタで実行する。"""
141    Interpreter(check(parse(source)), out).run()
