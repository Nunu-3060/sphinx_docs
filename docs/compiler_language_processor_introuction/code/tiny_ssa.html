tiny_ssa.py

支配木と支配辺境の計算、SSA 形式への変換、SSA 形式からの復帰。（第 8 章）

  1"""支配関係の計算と SSA 形式への変換（第 8 章）。
  2
  3* 支配木: Cooper, Harvey, Kennedy の反復アルゴリズムで求める。
  4* SSA 形式: Cytron らの方法（支配辺境に φ 関数を置き、支配木をたどって
  5  変数の名前を付け替える）で構築する。
  6* SSA 形式からの復帰: φ 関数を先行ブロックでのコピーに置き換える。
  7"""

 11import sys

 13from tiny_ir import (
 14    Const, Copy, Function, IRInterpreter, Module, Operand, Phi, Store, Var,

 18def reverse_postorder(func: Function) -> list[str]:
 19    """入口から深さ優先探索したときの帰りがけ順の逆順。"""
 20    order: list[str] = []
 21    visited: set[str] = set()
 22
 23    def visit(label: str) -> None:
 24        visited.add(label)
 25        for succ in func.blocks[label].successors():
 26            if succ not in visited:
 27                visit(succ)
 28        order.append(label)
 29
 30    visit(func.entry)
 31    return order[::-1]
 32
 33
 34def immediate_dominators(func: Function) -> dict[str, str]:
 35    """各基本ブロックの直接支配ブロックを返す（入口は自分自身）。"""
 36    rpo = reverse_postorder(func)
 37    index = {label: i for i, label in enumerate(rpo)}
 38    preds = func.predecessors()
 39    idom: dict[str, str] = {func.entry: func.entry}
 40
 41    def intersect(a: str, b: str) -> str:
 42        # 支配木を上にたどり、a と b の共通の祖先を探す。
 43        while a != b:
 44            while index[a] > index[b]:
 45                a = idom[a]
 46            while index[b] > index[a]:
 47                b = idom[b]
 48        return a
 49
 50    changed = True
 51    while changed:
 52        changed = False
 53        for label in rpo[1:]:
 54            done = [p for p in preds[label] if p in idom]
 55            new_idom = done[0]
 56            for p in done[1:]:
 57                new_idom = intersect(p, new_idom)
 58            if idom.get(label) != new_idom:
 59                idom[label] = new_idom
 60                changed = True
 61    return idom
 62
 63
 64def dominator_tree(func: Function,
 65                   idom: dict[str, str]) -> dict[str, list[str]]:
 66    """支配木の子の一覧を返す。"""
 67    children: dict[str, list[str]] = {label: [] for label in func.blocks}
 68    for label in reverse_postorder(func):
 69        if label != func.entry:
 70            children[idom[label]].append(label)
 71    return children
 72
 73
 74def dominance_frontiers(func: Function,
 75                        idom: dict[str, str]) -> dict[str, set[str]]:
 76    """各基本ブロックの支配辺境を返す。"""
 77    frontier: dict[str, set[str]] = {label: set() for label in func.blocks}
 78    for label, preds in func.predecessors().items():
 79        if len(preds) < 2:
 80            continue
 81        for p in preds:
 82            runner = p
 83            while runner != idom[label]:
 84                frontier[runner].add(label)
 85                runner = idom[runner]
 86    return frontier

 89def to_ssa(func: Function) -> Function:
 90    """関数を SSA 形式に変換する（func を直接書き換える）。"""
 91    idom = immediate_dominators(func)
 92    frontier = dominance_frontiers(func, idom)
 93    children = dominator_tree(func, idom)
 94    preds = func.predecessors()
 95
 96    # 1. 変数ごとに、定義している基本ブロックと定義の数を数える。
 97    def_blocks: dict[str, set[str]] = {p: {func.entry} for p in func.params}
 98    def_count: dict[str, int] = {p: 1 for p in func.params}
 99    for block in func.blocks.values():
100        for instr in block.instrs:
101            dst = defined_var(instr)
102            if dst is not None:
103                def_blocks.setdefault(dst, set()).add(block.label)
104                def_count[dst] = def_count.get(dst, 0) + 1
105    # 定義が 1 つだけの変数は、定義がすべての使用を支配するので名前を変えない。
106    targets = {v for v, n in def_count.items() if n >= 2}
107
108    # 2. 支配辺境に φ 関数を置く。
109    phi_var: dict[int, str] = {}  # φ 関数（の id）→ 元の変数名
110    for var in sorted(targets):
111        worklist = list(def_blocks[var])
112        has_phi: set[str] = set()
113        while worklist:
114            x = worklist.pop()
115            for y in frontier[x]:
116                if y in has_phi:
117                    continue
118                phi = Phi(var, {p: Var(var) for p in preds[y]})
119                func.blocks[y].instrs.insert(0, phi)
120                phi_var[id(phi)] = var
121                has_phi.add(y)
122                if y not in def_blocks[var]:
123                    def_blocks[var].add(y)
124                    worklist.append(y)
125
126    # 3. 支配木をたどりながら名前を付け替える。
127    counter: dict[str, int] = {v: 0 for v in targets}
128    stacks: dict[str, list[str]] = {
129        v: [v] if v in func.params else [] for v in targets}
130
131    def current(op: Operand) -> Operand:
132        if isinstance(op, Var) and op.name in targets:
133            stack = stacks[op.name]
134            return Var(stack[-1]) if stack else Const(None)
135        return op
136
137    def rename(label: str) -> None:
138        block = func.blocks[label]
139        pushed: list[str] = []
140        for instr in block.instrs:
141            if not isinstance(instr, Phi):
142                map_operands(instr, current)
143            if isinstance(instr, Store) or instr.dst not in targets:
144                continue
145            dst = instr.dst
146            counter[dst] += 1
147            new_name = f"{dst}.{counter[dst]}"
148            stacks[dst].append(new_name)
149            pushed.append(dst)
150            instr.dst = new_name
151        map_operands(block.terminator, current)
152        for succ in block.successors():
153            for phi in func.blocks[succ].phis():
154                phi.args[label] = current(Var(phi_var[id(phi)]))
155        for child in children[label]:
156            rename(child)
157        for var in pushed:
158            stacks[var].pop()
159
160    rename(func.entry)
161    return func

164def from_ssa(func: Function) -> Function:
165    """φ 関数をコピー命令に置き換えて SSA 形式から戻す（直接書き換える）。
166
167    x = phi(B1: a, B2: b) を、B1 の末尾の x' = a と B2 の末尾の x' = b、
168    そして元の位置の x = x' に置き換える。新しい変数 x' を経由するので、
169    φ 関数どうしが互いの値を参照していても正しく動く。
170    """
171    for block in list(func.blocks.values()):
172        for i, phi in enumerate(block.phis()):
173            temp = phi.dst + "'"
174            for pred, value in phi.args.items():
175                func.blocks[pred].instrs.append(Copy(temp, value))
176            block.instrs[i] = Copy(phi.dst, Var(temp))
177    return func

180def module_to_ssa(module: Module) -> Module:
181    for func in module.values():
182        to_ssa(func)
183    return module
184
185
186def module_from_ssa(module: Module) -> Module:
187    for func in module.values():
188        from_ssa(func)
189    return module

192def main() -> None:
193    source = ('fn sum(n: int) -> int {\n'
194              '    let s = 0;\n'
195              '    let i = 1;\n'
196              '    while (i <= n) {\n'
197              '        s = s + i;\n'
198              '        i = i + 1;\n'
199              '    }\n'
200              '    return s;\n'
201              '}\n'
202              'fn main() {\n'
203              '    print(sum(10));\n'
204              '}\n')
205    if len(sys.argv) > 1:
206        with open(sys.argv[1], encoding="utf-8") as f:
207            source = f.read()
208    module = compile_to_ir(source)
209    for func in module.values():
210        idom = immediate_dominators(func)
211        frontier = dominance_frontiers(func, idom)
212        print(f"function {func.name}")
213        for label in func.blocks:
214            print(f"  {label:12} idom={idom[label]:12} "
215                  f"DF={sorted(frontier[label])}")
216    module_to_ssa(module)
217    print("--- SSA 形式")
218    for func in module.values():
219        print(format_function(func))
220    IRInterpreter(module).run()
221    module_from_ssa(module)
222    print("--- SSA 形式から戻した結果")
223    for func in module.values():
224        print(format_function(func))
225    IRInterpreter(module).run()
226

228if __name__ == "__main__":
229    main()
