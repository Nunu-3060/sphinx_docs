tiny.py

Tiny 処理系のコマンドラインツール。（付録 A）

tiny.py をダウンロード

  1"""Tiny 言語処理系のコマンドラインツール（付録 A）。
  2
  3使用例:
  4    python tiny.py run programs/fib.tiny       # インタプリタで実行
  5    python tiny.py vm programs/fib.tiny        # 仮想マシンで実行
  6    python tiny.py ir --opt programs/fib.tiny  # 最適化した IR を表示
  7"""
  8
  9from __future__ import annotations
 10
 11import argparse
 12import sys
 13
 14from tiny_ast import dump
 15from tiny_bytecode import compile_source, disassemble
 16from tiny_checker import check
 17from tiny_interp import run_source
 18from tiny_ir import (
 19    IRInterpreter, compile_to_ir, format_function, format_module, to_dot)
 20from tiny_lexer import TinyError, tokenize
 21from tiny_opt import optimize_module
 22from tiny_parser import ParseError, parse
 23from tiny_regalloc import assign_registers
 24from tiny_ssa import from_ssa, module_to_ssa
 25from tiny_vm import VM
 26
 27
 28def command(args: argparse.Namespace, source: str) -> None:
 29    match args.command:
 30        case "tokens":
 31            for token in tokenize(source):
 32                print(f"{token.line:3}:{token.column:<3} "
 33                      f"{token.kind.name:10} {token.text}")
 34        case "ast":
 35            print(dump(parse(source)))
 36        case "check":
 37            check(parse(source))
 38            print("エラーはありません")
 39        case "run":
 40            run_source(source)
 41        case "ir":
 42            module = compile_to_ir(source)
 43            if args.ssa or args.opt:
 44                module_to_ssa(module)
 45            if args.opt:
 46                optimize_module(module)
 47            if args.dot:
 48                print(to_dot(module[args.dot]))
 49            elif args.exec:
 50                IRInterpreter(module).run()
 51            else:
 52                print(format_module(module))
 53        case "bytecode":
 54            program = compile_source(source)
 55            print("\n\n".join(disassemble(f, program)
 56                              for f in program.functions))
 57        case "vm":
 58            VM(compile_source(source)).run()
 59        case "regalloc":
 60            module = optimize_module(module_to_ssa(compile_to_ir(source)))
 61            for func in module.values():
 62                from_ssa(func)
 63                print(format_function(func))
 64                print("--- レジスタ割り当て後")
 65                print(assign_registers(func, args.registers))
 66                print()
 67
 68
 69def main() -> int:
 70    parser = argparse.ArgumentParser(description="Tiny 言語処理系")
 71    sub = parser.add_subparsers(dest="command", required=True)
 72    for name, text in [("tokens", "トークン列を表示する"),
 73                       ("ast", "抽象構文木を表示する"),
 74                       ("check", "意味解析だけを行う"),
 75                       ("run", "木構造インタプリタで実行する"),
 76                       ("bytecode", "バイトコードを表示する"),
 77                       ("vm", "仮想マシンで実行する")]:
 78        sub.add_parser(name, help=text).add_argument("file")
 79    ir = sub.add_parser("ir", help="中間表現を表示する")
 80    ir.add_argument("file")
 81    ir.add_argument("--ssa", action="store_true", help="SSA 形式にする")
 82    ir.add_argument("--opt", action="store_true", help="最適化する")
 83    ir.add_argument("--dot", metavar="FUNC",
 84                    help="関数 FUNC の制御フローグラフを DOT 言語で出力する")
 85    ir.add_argument("--exec", action="store_true", help="IR を実行する")
 86    reg = sub.add_parser("regalloc", help="レジスタ割り当ての結果を表示する")
 87    reg.add_argument("file")
 88    reg.add_argument("-k", "--registers", type=int, default=4,
 89                     help="レジスタの数（既定値 4）")
 90    args = parser.parse_args()
 91
 92    with open(args.file, encoding="utf-8") as f:
 93        source = f.read()
 94    try:
 95        command(args, source)
 96    except ParseError as e:
 97        for error in e.all_errors:
 98            print(f"{args.file}:{error}", file=sys.stderr)
 99        return 1
100    except TinyError as e:
101        print(f"{args.file}:{e}", file=sys.stderr)
102        return 1
103    return 0
104
105
106if __name__ == "__main__":
107    sys.exit(main())
