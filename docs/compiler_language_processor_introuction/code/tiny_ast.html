191def dump(node: object, indent: int = 0) -> str:
192    """構文木を字下げ付きの文字列に変換する。"""
193    pad = "  " * indent
194    if isinstance(node, list):
195        if not node:
196            return pad + "[]"
197        return "\n".join(dump(item, indent) for item in node)
198    if not isinstance(node, Node):
199        return pad + (str(node) if isinstance(node, Type) else repr(node))
200    lines = [pad + type(node).__name__]
201    for f in fields(node):
202        if f.name in ("line", "ty"):
203            continue
204        value = getattr(node, f.name)
205        if isinstance(value, (Node, list)) and value:
206            lines.append(f"{pad}  {f.name}:")
207            lines.append(dump(value, indent + 2))
208        else:
209            shown = str(value) if isinstance(value, Type) else repr(value)
210            lines.append(f"{pad}  {f.name}: {shown}")
211    return "\n".join(lines)
