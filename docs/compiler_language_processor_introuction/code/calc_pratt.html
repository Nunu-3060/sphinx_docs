Pratt 構文解析による算術式の解析。（第 4 章）

calc_pratt.py をダウンロード

 1"""Pratt 構文解析（演算子順位構文解析）による算術式の解析（第 4 章）。
 2
 3演算子ごとに「結合力」を表で与え、1 つの関数で任意の優先順位の式を解析する。
 4結果は構造が分かるように括弧付きの S 式の文字列で返す。
 5"""
 6
 7from __future__ import annotations
 8
 9from tiny_lexer import Token, TinyError, TokenKind, tokenize
10
11# 二項演算子の結合力（左、右）。数が大きいほど強く結び付く。
12# 左結合の演算子は 右 = 左 + 1 とする。
13BINARY_POWER: dict[TokenKind, tuple[int, int]] = {
14    TokenKind.OR: (1, 2),
15    TokenKind.AND: (3, 4),
16    TokenKind.EQ: (5, 6),
17    TokenKind.NE: (5, 6),
18    TokenKind.LT: (7, 8),
19    TokenKind.LE: (7, 8),
20    TokenKind.GT: (7, 8),
21    TokenKind.GE: (7, 8),
22    TokenKind.PLUS: (9, 10),
23    TokenKind.MINUS: (9, 10),
24    TokenKind.STAR: (11, 12),
25    TokenKind.SLASH: (11, 12),
26    TokenKind.PERCENT: (11, 12),
27}
28
29PREFIX_POWER = 13  # 単項演算子 - と ! の結合力
30
31
32class PrattParser:
33    def __init__(self, source: str) -> None:
34        self.tokens = tokenize(source)
35        self.pos = 0
36
37    def peek(self) -> Token:
38        return self.tokens[self.pos]
39
40    def advance(self) -> Token:
41        token = self.tokens[self.pos]
42        self.pos += 1
43        return token
44
45    def parse(self) -> str:
46        result = self.expr(0)
47        token = self.peek()
48        if token.kind is not TokenKind.EOF:
49            raise TinyError(f"余分な字句 {token.text!r} があります",
50                            token.line, token.column)
51        return result
52
53    def expr(self, min_power: int) -> str:
54        """結合力が min_power 以上の演算子だけを取り込んで式を解析する。"""
55        left = self.prefix()
56        while True:
57            op = self.peek()
58            power = BINARY_POWER.get(op.kind)
59            if power is None or power[0] < min_power:
60                return left
61            self.advance()
62            right = self.expr(power[1])
63            left = f"({op.text} {left} {right})"
64
65    def prefix(self) -> str:
66        token = self.advance()
67        if token.kind in (TokenKind.MINUS, TokenKind.NOT):
68            operand = self.expr(PREFIX_POWER)
69            return f"({token.text} {operand})"
70        if token.kind in (TokenKind.INT, TokenKind.IDENT):
71            return token.text
72        if token.kind is TokenKind.LPAREN:
73            inner = self.expr(0)
74            if self.advance().kind is not TokenKind.RPAREN:
75                raise TinyError(") が必要です", token.line, token.column)
76            return inner
77        raise TinyError(f"式が必要ですが {token.text!r} があります",
78                        token.line, token.column)
79
80
81def main() -> None:
82    for source in ["1 + 2 * 3", "a - b - c", "-x * y",
83                   "a < b && b < c || !d", "(1 + 2) * 3"]:
84        print(f"{source:22} => {PrattParser(source).parse()}")
85
86
87if __name__ == "__main__":
88    main()
