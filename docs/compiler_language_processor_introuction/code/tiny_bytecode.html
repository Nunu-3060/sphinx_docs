  1"""Tiny 言語のバイトコード生成器と逆アセンブラ（第 10 章）。
  2
  3意味解析済みの抽象構文木から、スタック型仮想マシン用の命令列を生成する。
  4"""
  5
  6from __future__ import annotations
  7
  8import sys
  9from dataclasses import dataclass, field
 10from enum import IntEnum, auto
 11
 12import tiny_ast as A
 13from tiny_checker import always_returns, check
 14from tiny_parser import parse
 15from tiny_runtime import Value, format_value
 16
 17
 18class Op(IntEnum):
 19    """仮想マシンの命令。引数を取る命令は、その意味をコメントに書く。"""
 20
 21    CONST = auto()          # 定数表の arg 番目の値を積む
 22    LOAD = auto()           # 局所変数 arg の値を積む
 23    STORE = auto()          # 値を降ろして局所変数 arg に書き込む
 24    ADD = auto()
 25    SUB = auto()
 26    MUL = auto()
 27    DIV = auto()
 28    MOD = auto()
 29    EQ = auto()
 30    NE = auto()
 31    LT = auto()
 32    LE = auto()
 33    GT = auto()
 34    GE = auto()
 35    NEG = auto()
 36    NOT = auto()
 37    JUMP = auto()           # arg 番地へ飛ぶ
 38    JUMP_IF_FALSE = auto()  # 値を降ろし、偽なら arg 番地へ飛ぶ
 39    JUMP_IF_FALSE_OR_POP = auto()  # 偽なら値を残して飛び、真なら降ろす
 40    JUMP_IF_TRUE_OR_POP = auto()   # 真なら値を残して飛び、偽なら降ろす
 41    BUILD_ARRAY = auto()    # 値を arg 個降ろして配列を作る
 42    INDEX = auto()          # 配列と添字を降ろし、要素を積む
 43    STORE_INDEX = auto()    # 配列・添字・値を降ろし、要素に書き込む
 44    CALL = auto()           # arg 番目の関数を呼ぶ
 45    PRINT = auto()          # 値を arg 個降ろして表示する
 46    LEN = auto()
 47    PUSH = auto()
 48    TO_STR = auto()
 49    POP = auto()            # 値を 1 つ降ろして捨てる
 50    RETURN = auto()         # 値を降ろし、呼び出し元に戻る
 51

 53BINARY_OPS: dict[str, Op] = {
 54    "+": Op.ADD, "-": Op.SUB, "*": Op.MUL, "/": Op.DIV, "%": Op.MOD,
 55    "==": Op.EQ, "!=": Op.NE, "<": Op.LT, "<=": Op.LE, ">": Op.GT,
 56    ">=": Op.GE,
 57}
 58BUILTIN_OPS: dict[str, Op] = {
 59    "len": Op.LEN, "push": Op.PUSH, "to_str": Op.TO_STR,
 60}

 62Instruction = tuple[Op, int]

 65@dataclass
 66class CodeObject:
 67    """1 つの関数をコンパイルした結果。"""
 68
 69    name: str
 70    num_params: int
 71    num_locals: int = 0
 72    code: list[Instruction] = field(default_factory=list)
 73    constants: list[Value] = field(default_factory=list)
 74    local_names: list[str] = field(default_factory=list)
 75

 77@dataclass
 78class CompiledProgram:
 79    functions: list[CodeObject]
 80    main: int  # main 関数の番号

 82
 83class FunctionCompiler:
 84    def __init__(self, func: A.FuncDef, func_index: dict[str, int]) -> None:
 85        self.func = func
 86        self.func_index = func_index
 87        self.code = CodeObject(func.name, len(func.params))
 88        self.scopes: list[dict[str, int]] = [{}]
 89        self.next_slot = 0
 90
 91    # ------------------------------------------------ 補助
 92
 93    def emit(self, op: Op, arg: int = 0) -> int:
 94        """命令を追加し、その番地を返す。"""
 95        self.code.code.append((op, arg))
 96        return len(self.code.code) - 1
 97
 98    def patch(self, address: int) -> None:
 99        """address のジャンプ命令の飛び先を、次に生成する命令の番地にする。"""
100        op, _ = self.code.code[address]
101        self.code.code[address] = (op, len(self.code.code))
102
103    def constant(self, value: Value) -> int:
104        constants = self.code.constants
105        for i, c in enumerate(constants):
106            if type(c) is type(value) and c == value:
107                return i
108        constants.append(value)
109        return len(constants) - 1
110
111    def declare(self, name: str) -> int:
112        """局所変数にスロット（番号）を割り当てる。"""
113        slot = self.next_slot
114        self.next_slot += 1
115        self.scopes[-1][name] = slot
116        self.code.num_locals = max(self.code.num_locals, self.next_slot)
117        names = self.code.local_names
118        if slot == len(names):
119            names.append(name)
120        elif name not in names[slot].split("/"):
121            names[slot] += "/" + name
122        return slot
123
124    def lookup(self, name: str) -> int:
125        for scope in reversed(self.scopes):
126            if name in scope:
127                return scope[name]
128        raise AssertionError(name)
129
130    # ------------------------------------------------ 関数と文
131
132    def compile(self) -> CodeObject:
133        for param in self.func.params:
134            self.declare(param.name)
135        self.compile_block(self.func.body)
136        if self.func.return_type == A.VOID:
137            # 値を返さない関数は、末尾に達したら None を返す。
138            self.emit(Op.CONST, self.constant(None))
139            self.emit(Op.RETURN)
140        return self.code
141
142    def compile_block(self, block: A.Block) -> None:
143        # ブロックを抜けたら、そのブロックの変数のスロットを再利用する。
144        self.scopes.append({})
145        saved = self.next_slot
146        for stmt in block.stmts:
147            self.compile_stmt(stmt)
148        self.next_slot = saved
149        self.scopes.pop()
150
151    def compile_stmt(self, stmt: A.Stmt) -> None:
152        match stmt:
153            case A.Block():
154                self.compile_block(stmt)
155            case A.Let(name, _, value):
156                self.compile_expr(value)
157                self.emit(Op.STORE, self.declare(name))
158            case A.Assign(A.Name(name), value):
159                self.compile_expr(value)
160                self.emit(Op.STORE, self.lookup(name))
161            case A.Assign(A.Index(target, index), value):
162                self.compile_expr(target)
163                self.compile_expr(index)
164                self.compile_expr(value)
165                self.emit(Op.STORE_INDEX)
166            case A.If(cond, then_body, else_body):
167                self.compile_expr(cond)
168                to_else = self.emit(Op.JUMP_IF_FALSE)
169                self.compile_block(then_body)
170                if else_body is None:
171                    self.patch(to_else)
172                elif always_returns(then_body):
173                    # then 節が必ず return するなら、飛び越す命令は不要。
174                    self.patch(to_else)
175                    self.compile_stmt(else_body)
176                else:
177                    to_end = self.emit(Op.JUMP)
178                    self.patch(to_else)
179                    self.compile_stmt(else_body)
180                    self.patch(to_end)
181            case A.While(cond, body):
182                start = len(self.code.code)
183                self.compile_expr(cond)
184                to_end = self.emit(Op.JUMP_IF_FALSE)
185                self.compile_block(body)
186                self.emit(Op.JUMP, start)
187                self.patch(to_end)
188            case A.Return(value):
189                if value is None:
190                    self.emit(Op.CONST, self.constant(None))
191                else:
192                    self.compile_expr(value)
193                self.emit(Op.RETURN)
194            case A.ExprStmt(expr):
195                self.compile_expr(expr)
196                self.emit(Op.POP)
197
198    # ------------------------------------------------ 式
199
200    def compile_expr(self, expr: A.Expr) -> None:
201        """式の値を計算し、スタックに 1 つ積む命令を生成する。"""
202        match expr:
203            case A.IntLit(value) | A.BoolLit(value) | A.StrLit(value):
204                self.emit(Op.CONST, self.constant(value))
205            case A.Name(name):
206                self.emit(Op.LOAD, self.lookup(name))
207            case A.ArrayLit(elements):
208                for e in elements:
209                    self.compile_expr(e)
210                self.emit(Op.BUILD_ARRAY, len(elements))
211            case A.Index(target, index):
212                self.compile_expr(target)
213                self.compile_expr(index)
214                self.emit(Op.INDEX)
215            case A.Unary(op, operand):
216                self.compile_expr(operand)
217                self.emit(Op.NEG if op == "-" else Op.NOT)
218            case A.Binary("&&" | "||" as op, left, right):
219                self.compile_expr(left)
220                jump = Op.JUMP_IF_FALSE_OR_POP if op == "&&" \
221                    else Op.JUMP_IF_TRUE_OR_POP
222                to_end = self.emit(jump)
223                self.compile_expr(right)
224                self.patch(to_end)
225            case A.Binary(op, left, right):
226                self.compile_expr(left)
227                self.compile_expr(right)
228                self.emit(BINARY_OPS[op])
229            case A.Call(callee, args):
230                for arg in args:
231                    self.compile_expr(arg)
232                if callee == "print":
233                    self.emit(Op.PRINT, len(args))
234                elif callee in BUILTIN_OPS:
235                    self.emit(BUILTIN_OPS[callee])
236                else:
237                    self.emit(Op.CALL, self.func_index[callee])
238            case _:
239                raise AssertionError(expr)
240
241
242def compile_program(program: A.Program) -> CompiledProgram:
243    """意味解析済みのプログラムをバイトコードに変換する。"""
244    func_index = {f.name: i for i, f in enumerate(program.functions)}
245    functions = [FunctionCompiler(f, func_index).compile()
246                 for f in program.functions]
247    return CompiledProgram(functions, func_index["main"])
248
249
250def compile_source(source: str) -> CompiledProgram:
251    return compile_program(check(parse(source)))

254# ---------------------------------------------------------------- 逆アセンブラ

256def disassemble(code: CodeObject, program: CompiledProgram) -> str:
257    lines = [f"function {code.name}: params={code.num_params} "
258             f"locals={code.num_locals}"]
259    for address, (op, arg) in enumerate(code.code):
260        text = f"{address:4}  {op.name:20}"
261        if op is Op.CONST:
262            text += f"{arg:3} ({format_value(code.constants[arg], True)})"
263        elif op in (Op.LOAD, Op.STORE):
264            text += f"{arg:3} ({code.local_names[arg]})"
265        elif op is Op.CALL:
266            text += f"{arg:3} ({program.functions[arg].name})"
267        elif op.name.startswith("JUMP") or op in (Op.BUILD_ARRAY, Op.PRINT):
268            text += f"{arg:3}"
269        lines.append(text.rstrip())
270    return "\n".join(lines)

273def main() -> None:
274    source = ('fn fact(n: int) -> int {\n'
275              '    if (n <= 1) { return 1; }\n'
276              '    return n * fact(n - 1);\n'
277              '}\n'
278              'fn main() {\n'
279              '    print("10! =", fact(10));\n'
280              '}\n')
281    if len(sys.argv) > 1:
282        with open(sys.argv[1], encoding="utf-8") as f:
283            source = f.read()
284    program = compile_source(source)
285    print("\n\n".join(disassemble(f, program) for f in program.functions))
