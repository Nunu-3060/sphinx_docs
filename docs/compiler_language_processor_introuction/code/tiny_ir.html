139class BasicBlock:
140    label: str
141    instrs: list[Instr]

144    def successors(self) -> list[str]:
145        term = self.terminator
146        if isinstance(term, Jump):
147            return [term.target]
148        if isinstance(term, Branch):
149            return [term.if_true, term.if_false]

164        return next(iter(self.blocks))

167        preds: dict[str, list[str]] = {label: [] for label in self.blocks}

179def defined_var(instr: Instr) -> str | None:

186def used_operands(item: Instr | Terminator) -> list[Operand]:
187    """命令が読むオペランドを返す。"""
188    match item:
189        case Copy(_, src):
190            return [src]
191        case BinOp(_, _, left, right):
192            return [left, right]
193        case UnOp(_, _, operand):
194            return [operand]
195        case Call(_, _, args) | NewArray(_, args):
196            return list(args)
197        case Load(_, base, index):
198            return [base, index]
199        case Store(base, index, value):
200            return [base, index, value]
201        case Phi(_, args):
202            return list(args.values())
203        case Branch(cond, _, _):
204            return [cond]
205        case Return(value):
206            return [] if value is None else [value]
207    return []

210def used_vars(item: Instr | Terminator) -> list[str]:
211    return [op.name for op in used_operands(item) if isinstance(op, Var)]

214def map_operands(item: Instr | Terminator,
215                 f: Callable[[Operand], Operand]) -> None:
216    """命令が読むオペランドを f で置き換える（命令を直接書き換える）。"""
217    match item:
218        case Copy():
219            item.src = f(item.src)
220        case BinOp():
221            item.left, item.right = f(item.left), f(item.right)
222        case UnOp():
223            item.operand = f(item.operand)
224        case Call():
225            item.args = [f(a) for a in item.args]
226        case NewArray():
227            item.elements = [f(e) for e in item.elements]
228        case Load():
229            item.base, item.index = f(item.base), f(item.index)
230        case Store():
231            item.base, item.index = f(item.base), f(item.index)
232            item.value = f(item.value)
233        case Phi():
234            item.args = {label: f(v) for label, v in item.args.items()}
235        case Branch():
236            item.cond = f(item.cond)
237        case Return():
238            if item.value is not None:
239                item.value = f(item.value)

242# ---------------------------------------------------------------- 表示

244def format_item(item: Instr | Terminator) -> str:
245    match item:
246        case Copy(dst, src):
247            return f"{dst} = {src}"
248        case BinOp(dst, op, left, right):
249            return f"{dst} = {left} {op} {right}"
250        case UnOp(dst, op, operand):
251            return f"{dst} = {op} {operand}"
252        case Call(dst, func, args):
253            text = f"call {func}({', '.join(map(str, args))})"
254            return text if dst is None else f"{dst} = {text}"
255        case NewArray(dst, elements):
256            return f"{dst} = [{', '.join(map(str, elements))}]"
257        case Load(dst, base, index):
258            return f"{dst} = {base}[{index}]"
259        case Store(base, index, value):
260            return f"{base}[{index}] = {value}"
261        case Phi(dst, args):
262            pairs = ", ".join(f"{label}: {v}" for label, v in args.items())
263            return f"{dst} = phi({pairs})"
264        case Jump(target):
265            return f"jump {target}"
266        case Branch(cond, if_true, if_false):
267            return f"branch {cond}, {if_true}, {if_false}"
268        case Return(value):
269            return "return" if value is None else f"return {value}"
270    raise AssertionError(item)

273def block_lines(block: BasicBlock) -> list[str]:
274    lines = [f"{block.label}:"]
275    lines += ["    " + format_item(i) for i in block.instrs]
276    lines.append("    " + format_item(block.terminator))
277    return lines

280def format_function(func: Function) -> str:
281    lines = [f"function {func.name}({', '.join(func.params)})"]
282    for block in func.blocks.values():
283        lines += block_lines(block)
284    return "\n".join(lines)

291def to_dot(func: Function) -> str:
292    """制御フローグラフを Graphviz の DOT 言語で表す。"""
293    def escape(text: str) -> str:
294        return text.replace("\\", "\\\\").replace('"', '\\"')
295
296    lines = [f'digraph "{func.name}" {{',
297             '  node [shape=box, fontname="monospace"];']
298    for block in func.blocks.values():
299        label = "".join(escape(line) + "\\l" for line in block_lines(block))
300        lines.append(f'  "{block.label}" [label="{label}"];')
301    for block in func.blocks.values():
302        term = block.terminator
303        if isinstance(term, Branch):
304            lines.append(f'  "{block.label}" -> "{term.if_true}" '
305                         '[label="true"];')
306            lines.append(f'  "{block.label}" -> "{term.if_false}" '
307                         '[label="false"];')
308        elif isinstance(term, Jump):
309            lines.append(f'  "{block.label}" -> "{term.target}";')
310    lines.append("}")
311    return "\n".join(lines)

316class Lowering:
317    """1 つの関数の構文木を IR に変換する。"""
318
319    def __init__(self, func: A.FuncDef) -> None:
320        self.func = func
321        self.blocks: dict[str, BasicBlock] = {}
322        self.label: str | None = "entry"  # 組み立て中の基本ブロック
323        self.instrs: list[Instr] = []
324        self.label_count = 0
325        self.temp_count = 0
326        self.scopes: list[dict[str, str]] = [{}]
327        self.used_names: set[str] = set()
328
329    # ------------------------------------------------ 組み立ての補助
330
331    def new_id(self) -> int:
332        """ラベルの番号を作る。1 つの構文から作るラベルには同じ番号を使う。"""
333        self.label_count += 1
334        return self.label_count
335
336    def new_temp(self) -> str:
337        self.temp_count += 1
338        return f"%{self.temp_count}"
339
340    def emit(self, instr: Instr) -> None:
341        if self.label is None:  # return の後ろの到達不能なコード
342            self.start_block(f"dead{self.new_id()}")
343        self.instrs.append(instr)
344
345    def terminate(self, term: Terminator) -> None:
346        """組み立て中の基本ブロックを終端命令で閉じる。"""
347        if self.label is None:
348            self.start_block(f"dead{self.new_id()}")
349        assert self.label is not None
350        self.blocks[self.label] = BasicBlock(self.label, self.instrs, term)
351        self.label = None
352
353    def start_block(self, label: str) -> None:
354        if self.label is not None:  # 前のブロックから流れ込む
355            self.terminate(Jump(label))
356        self.label = label
357        self.instrs = []
358
359    def declare(self, name: str) -> str:
360        """変数を宣言し、IR 上の名前を返す。同名の変数は名前を変える。"""
361        ir_name = name
362        n = 1
363        while ir_name in self.used_names:
364            n += 1
365            ir_name = f"{name}#{n}"
366        self.used_names.add(ir_name)
367        self.scopes[-1][name] = ir_name
368        return ir_name
369
370    def lookup(self, name: str) -> str:
371        for scope in reversed(self.scopes):
372            if name in scope:
373                return scope[name]
374        raise AssertionError(name)
375
376    # ------------------------------------------------ 関数と文
377
378    def lower(self) -> Function:
379        params = [self.declare(p.name) for p in self.func.params]
380        self.lower_block(self.func.body)
381        if self.label is not None:
382            self.terminate(Return(None))
383        return Function(self.func.name, params,
384                        remove_unreachable(self.blocks, "entry"))
385
386    def lower_block(self, block: A.Block) -> None:
387        self.scopes.append({})
388        for stmt in block.stmts:
389            self.lower_stmt(stmt)
390        self.scopes.pop()
391
392    def lower_stmt(self, stmt: A.Stmt) -> None:
393        match stmt:
394            case A.Block():
395                self.lower_block(stmt)
396            case A.Let(name, _, value):
397                src = self.lower_expr(value)
398                self.emit(Copy(self.declare(name), src))
399            case A.Assign(A.Name(name), value):
400                self.emit(Copy(self.lookup(name), self.lower_expr(value)))
401            case A.Assign(A.Index(target, index), value):
402                base = self.lower_expr(target)
403                i = self.lower_expr(index)
404                self.emit(Store(base, i, self.lower_expr(value)))
405            case A.If(cond, then_body, else_body):
406                n = self.new_id()
407                then_label, end_label = f"then{n}", f"endif{n}"
408                else_label = end_label if else_body is None else f"else{n}"
409                c = self.lower_expr(cond)
410                self.terminate(Branch(c, then_label, else_label))
411                self.start_block(then_label)
412                self.lower_block(then_body)
413                if else_body is not None:
414                    self.goto(end_label)
415                    self.start_block(else_label)
416                    self.lower_stmt(else_body)
417                self.goto(end_label)
418                self.start_block(end_label)
419            case A.While(cond, body):
420                n = self.new_id()
421                cond_label, body_label = f"cond{n}", f"body{n}"
422                end_label = f"endwhile{n}"
423                self.start_block(cond_label)
424                c = self.lower_expr(cond)
425                self.terminate(Branch(c, body_label, end_label))
426                self.start_block(body_label)
427                self.lower_block(body)
428                self.goto(cond_label)
429                self.start_block(end_label)
430            case A.Return(value):
431                result = None if value is None else self.lower_expr(value)
432                self.terminate(Return(result))
433            case A.ExprStmt(expr):
434                self.lower_expr(expr)
435
436    def goto(self, label: str) -> None:
437        """組み立て中のブロックがあれば label へのジャンプで閉じる。"""
438        if self.label is not None:
439            self.terminate(Jump(label))
440
441    # ------------------------------------------------ 式
442
443    def lower_expr(self, expr: A.Expr) -> Operand:
444        match expr:
445            case A.IntLit(value) | A.BoolLit(value) | A.StrLit(value):
446                return Const(value)
447            case A.Name(name):
448                return Var(self.lookup(name))
449            case A.ArrayLit(elements):
450                values = [self.lower_expr(e) for e in elements]
451                dst = self.new_temp()
452                self.emit(NewArray(dst, values))
453                return Var(dst)
454            case A.Index(target, index):
455                base = self.lower_expr(target)
456                i = self.lower_expr(index)
457                dst = self.new_temp()
458                self.emit(Load(dst, base, i))
459                return Var(dst)
460            case A.Unary(op, operand):
461                src = self.lower_expr(operand)
462                dst = self.new_temp()
463                self.emit(UnOp(dst, op, src))
464                return Var(dst)
465            case A.Binary("&&" | "||" as op, left, right):
466                return self.lower_logical(op, left, right)
467            case A.Binary(op, left, right):
468                lhs = self.lower_expr(left)
469                rhs = self.lower_expr(right)
470                dst = self.new_temp()
471                self.emit(BinOp(dst, op, lhs, rhs))
472                return Var(dst)
473            case A.Call(callee, args):
474                values = [self.lower_expr(a) for a in args]
475                if expr.ty == A.VOID:
476                    self.emit(Call(None, callee, values))
477                    return Const(None)
478                dst = self.new_temp()
479                self.emit(Call(dst, callee, values))
480                return Var(dst)
481        raise AssertionError(expr)
482
483    def lower_logical(self, op: str, left: A.Expr, right: A.Expr) -> Operand:
484        """短絡評価: 左辺だけで結果が決まれば右辺を評価しない。"""
485        result = self.new_temp()
486        n = self.new_id()
487        rhs_label, end_label = f"rhs{n}", f"endlogic{n}"
488        lhs = self.lower_expr(left)
489        self.emit(Copy(result, lhs))
490        if op == "&&":
491            self.terminate(Branch(lhs, rhs_label, end_label))
492        else:
493            self.terminate(Branch(lhs, end_label, rhs_label))
494        self.start_block(rhs_label)
495        self.emit(Copy(result, self.lower_expr(right)))
496        self.start_block(end_label)
497        return Var(result)

500def remove_unreachable(blocks: dict[str, BasicBlock],
501                       entry: str) -> dict[str, BasicBlock]:
502    """入口から到達できない基本ブロックを取り除く（順序は保つ）。"""
503    reachable: set[str] = set()
504    stack = [entry]
505    while stack:
506        label = stack.pop()
507        if label not in reachable:
508            reachable.add(label)
509            stack.extend(blocks[label].successors())
510    return {k: b for k, b in blocks.items() if k in reachable}

524class IRInterpreter:
525    """IR を直接実行する。steps は実行した命令の数。"""
526
527    def __init__(self, module: Module, out: TextIO = sys.stdout) -> None:
528        self.module = module
529        self.out = out
530        self.steps = 0
531
532    def run(self) -> None:
533        self.call("main", [])
534
535    def call(self, name: str, args: list[Value]) -> Value:
536        func = self.module[name]
537        env: dict[str, Value] = dict(zip(func.params, args))
538
539        def value(op: Operand) -> Value:
540            return op.value if isinstance(op, Const) else env[op.name]
541
542        label, previous = func.entry, ""
543        while True:
544            block = func.blocks[label]
545            # φ 関数はすべて同時に評価する（先に値を集めてから書き込む）。
546            phis = block.phis()
547            incoming = [value(phi.args[previous]) for phi in phis]
548            for phi, v in zip(phis, incoming):
549                env[phi.dst] = v
550            self.steps += len(phis)
551            for instr in block.instrs[len(phis):]:
552                self.steps += 1
553                self.execute(instr, env, value)
554            self.steps += 1
555            term = block.terminator
556            if isinstance(term, Return):
557                return None if term.value is None else value(term.value)
558            previous = label
559            if isinstance(term, Jump):
560                label = term.target
561            else:
562                label = term.if_true if value(term.cond) else term.if_false
563
564    def execute(self, instr: Instr, env: dict[str, Value],
565                value: Callable[[Operand], Value]) -> None:
566        match instr:
567            case Copy(dst, src):
568                env[dst] = value(src)
569            case BinOp(dst, op, left, right):
570                env[dst] = binary_op(op, value(left), value(right))
571            case UnOp(dst, op, operand):
572                env[dst] = unary_op(op, value(operand))
573            case Call(dst, func, args):
574                values = [value(a) for a in args]
575                if func in BUILTINS:
576                    result = call_builtin(func, values, self.out)
577                else:
578                    result = self.call(func, values)
579                if dst is not None:
580                    env[dst] = result
581            case NewArray(dst, elements):
582                env[dst] = [value(e) for e in elements]
583            case Load(dst, base, index):
584                env[dst] = index_value(value(base), value(index))
585            case Store(base, index, v):
586                store_index(value(base), value(index), value(v))
587            case Phi():
588                raise AssertionError("φ 関数は基本ブロックの先頭に置きます")
