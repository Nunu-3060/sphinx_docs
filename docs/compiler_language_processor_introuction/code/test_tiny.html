test_tiny.py

すべての実行方式の結果が一致することを確かめるテスト。（付録 A）

test_tiny.py をダウンロード

  1"""Tiny 処理系のテスト（付録 A）。
  2
  3すべての実行方式（木構造インタプリタ、IR、SSA 形式の IR、最適化した IR、
  4SSA 形式から戻した IR、仮想マシン）が同じ出力になることを確かめる。
  5
  6    python -m unittest test_tiny.py
  7"""
  8
  9from __future__ import annotations
 10
 11import io
 12import unittest
 13from collections.abc import Callable
 14from pathlib import Path
 15
 16from tiny_bytecode import compile_source
 17from tiny_checker import TypeCheckError
 18from tiny_interp import run_source
 19from tiny_ir import IRInterpreter, compile_to_ir
 20from tiny_lexer import LexError
 21from tiny_opt import optimize_module
 22from tiny_parser import ParseError
 23from tiny_runtime import TinyRuntimeError
 24from tiny_ssa import module_from_ssa, module_to_ssa
 25from tiny_vm import VM
 26
 27PROGRAMS = Path(__file__).parent / "programs"
 28
 29
 30def run_interp(source: str, out: io.StringIO) -> None:
 31    run_source(source, out)
 32
 33
 34def run_ir(source: str, out: io.StringIO) -> None:
 35    IRInterpreter(compile_to_ir(source), out).run()
 36
 37
 38def run_ssa(source: str, out: io.StringIO) -> None:
 39    IRInterpreter(module_to_ssa(compile_to_ir(source)), out).run()
 40
 41
 42def run_opt(source: str, out: io.StringIO) -> None:
 43    module = optimize_module(module_to_ssa(compile_to_ir(source)))
 44    IRInterpreter(module, out).run()
 45
 46
 47def run_from_ssa(source: str, out: io.StringIO) -> None:
 48    module = optimize_module(module_to_ssa(compile_to_ir(source)))
 49    IRInterpreter(module_from_ssa(module), out).run()
 50
 51
 52def run_vm(source: str, out: io.StringIO) -> None:
 53    VM(compile_source(source), out).run()
 54
 55
 56RUNNERS: dict[str, Callable[[str, io.StringIO], None]] = {
 57    "interp": run_interp, "ir": run_ir, "ssa": run_ssa, "opt": run_opt,
 58    "from_ssa": run_from_ssa, "vm": run_vm,
 59}

 62def outputs(source: str) -> dict[str, str]:
 63    results = {}
 64    for name, runner in RUNNERS.items():
 65        out = io.StringIO()
 66        runner(source, out)
 67        results[name] = out.getvalue()
 68    return results
 69

 71def main_with(body: str) -> str:
 72    return "fn main() {\n" + body + "\n}\n"

 75class SameOutputTest(unittest.TestCase):
 76    def assert_output(self, source: str, expected: str) -> None:
 77        for name, result in outputs(source).items():
 78            with self.subTest(runner=name):
 79                self.assertEqual(result, expected)
 80
 81    def test_programs(self) -> None:
 82        for path in sorted(PROGRAMS.glob("*.tiny")):
 83            if path.name == "type_error.tiny":
 84                continue
 85            with self.subTest(program=path.name):
 86                results = outputs(path.read_text(encoding="utf-8"))
 87                self.assertEqual(len(set(results.values())), 1, results)
 88
 89    def test_integer_division(self) -> None:
 90        self.assert_output(main_with("print(7 / 2, -7 / 2, 7 % -2, -7 % 2);"),
 91                           "3 -3 1 -1\n")
 92
 93    def test_short_circuit(self) -> None:
 94        source = ("fn t(s: str) -> bool { print(s); return true; }\n"
 95                  "fn f(s: str) -> bool { print(s); return false; }\n"
 96                  + main_with("print(f(\"a\") && t(\"b\"));"
 97                              "print(t(\"c\") || f(\"d\"));"))
 98        self.assert_output(source, "a\nfalse\nc\ntrue\n")
 99
100    def test_shadowing(self) -> None:
101        self.assert_output(
102            main_with("let x = 1; { let x = 2; print(x); } print(x);"),
103            "2\n1\n")
104
105    def test_strings_and_arrays(self) -> None:
106        self.assert_output(
107            main_with('let a = ["x", "y"]; push(a, "z\\n");'
108                      'print(a, len(a), "ab" + "c", "abc"[1]);'),
109            '["x", "y", "z\n"] 3 abc b\n')
110
111    def test_constant_branch(self) -> None:
112        self.assert_output(
113            main_with("let a = 3 * 4; if (a > 100) { print(1); }"
114                      " else { print(a + 1); }"),
115            "13\n")

118class ErrorTest(unittest.TestCase):
119    def test_lex_error(self) -> None:
120        for body in ["let x = 1 @ 2;", "let x = １;", "let 変数 = 1;"]:
121            with self.subTest(body=body):
122                with self.assertRaises(LexError):
123                    run_source(main_with(body))
124
125    def test_parse_errors_are_collected(self) -> None:
126        source = main_with("let x = ;\nlet y = 1\nprint(x);")
127        with self.assertRaises(ParseError) as cm:
128            run_source(source)
129        self.assertEqual(len(cm.exception.all_errors), 2)
130
131    def test_type_errors(self) -> None:
132        bad = [
133            main_with('let x = 1 + "a";'),
134            main_with("print(y);"),
135            main_with("let a = [];"),
136            main_with('let s = "abc"; s[0] = "x";'),
137            "fn f() -> int { if (true) { return 1; } }\n" + main_with(""),
138            "fn main(x: int) {}\n",
139        ]
140        for source in bad:
141            with self.subTest(source=source):
142                with self.assertRaises(TypeCheckError):
143                    run_source(source)
144
145    def test_runtime_errors(self) -> None:
146        for body in ["let z = 0; print(1 / z);", "let a = [1]; print(a[1]);",
147                     "let a = [1]; let i = 0 - 1; print(a[i]);"]:
148            for name, runner in RUNNERS.items():
149                with self.subTest(body=body, runner=name):
150                    with self.assertRaises(TinyRuntimeError):
151                        runner(main_with(body), io.StringIO())
