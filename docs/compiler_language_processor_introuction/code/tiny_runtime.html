tiny_runtime.py

演算の意味と組み込み関数をまとめた実行時ライブラリ。（第 7 章）

tiny_runtime.py をダウンロード

  1"""Tiny 言語の実行時ライブラリ（第 7 章）。
  2
  3演算の意味と組み込み関数をまとめる。インタプリタ・IR インタプリタ・
  4仮想マシン・最適化器（定数畳み込み）がすべてこのモジュールを使うため、
  5どの実行方式でも同じ結果になる。
  6"""
  7
  8from __future__ import annotations
  9
 10from typing import TextIO, Union
 11
 12from tiny_lexer import TinyError
 13
 14# Tiny の値。配列は Python のリストで表す。None は「値なし」を表す。
 15Value = Union[int, bool, str, list["Value"], None]
 16
 17BUILTINS = ("print", "len", "push", "to_str")
 18
 19
 20class TinyRuntimeError(TinyError):
 21    """実行時エラー（ゼロ除算・範囲外アクセスなど）。"""
 22
 23
 24def int_div(a: int, b: int) -> int:
 25    """0 に向かって切り捨てる整数除算（C 言語と同じ）。"""
 26    if b == 0:
 27        raise TinyRuntimeError("0 で除算しました")
 28    q = abs(a) // abs(b)
 29    return q if (a >= 0) == (b >= 0) else -q
 30
 31
 32def int_mod(a: int, b: int) -> int:
 33    """int_div と組になる剰余。a == int_div(a, b) * b + int_mod(a, b) となる。"""
 34    return a - int_div(a, b) * b
 35
 36
 37def binary_op(op: str, left: Value, right: Value) -> Value:
 38    """二項演算を実行する（&& と || は短絡評価のため含めない）。"""
 39    if op == "==":
 40        return left == right
 41    if op == "!=":
 42        return left != right
 43    if op == "+":
 44        if isinstance(left, str) and isinstance(right, str):
 45            return left + right
 46    if not isinstance(left, int) or not isinstance(right, int):
 47        raise TinyRuntimeError(f"演算 {op} の型が不正です")
 48    match op:
 49        case "+":
 50            return left + right
 51        case "-":
 52            return left - right
 53        case "*":
 54            return left * right
 55        case "/":
 56            return int_div(left, right)
 57        case "%":
 58            return int_mod(left, right)
 59        case "<":
 60            return left < right
 61        case "<=":
 62            return left <= right
 63        case ">":
 64            return left > right
 65        case ">=":
 66            return left >= right
 67    raise TinyRuntimeError(f"未知の演算子 {op} です")
 68
 69
 70def unary_op(op: str, operand: Value) -> Value:
 71    """単項演算を実行する。"""
 72    if op == "-" and isinstance(operand, int):
 73        return -operand
 74    if op == "!" and isinstance(operand, bool):
 75        return not operand
 76    raise TinyRuntimeError(f"演算 {op} の型が不正です")
 77
 78
 79def index_value(base: Value, index: Value) -> Value:
 80    """base[index] を返す。範囲外ならエラー。"""
 81    if not isinstance(index, int) or not isinstance(base, (list, str)):
 82        raise TinyRuntimeError("添字演算の型が不正です")
 83    if not 0 <= index < len(base):
 84        raise TinyRuntimeError(f"添字 {index} が範囲外です")
 85    return base[index]
 86
 87
 88def store_index(base: Value, index: Value, value: Value) -> None:
 89    """base[index] = value を実行する。範囲外ならエラー。"""
 90    if not isinstance(index, int) or not isinstance(base, list):
 91        raise TinyRuntimeError("添字への代入の型が不正です")
 92    if not 0 <= index < len(base):
 93        raise TinyRuntimeError(f"添字 {index} が範囲外です")
 94    base[index] = value
 95
 96
 97def format_value(value: Value, nested: bool = False) -> str:
 98    """値を print 用の文字列に変換する。"""
 99    if isinstance(value, bool):
100        return "true" if value else "false"
101    if isinstance(value, str):
102        return f'"{value}"' if nested else value
103    if isinstance(value, list):
104        return "[" + ", ".join(format_value(v, True) for v in value) + "]"
105    return str(value)
106
107
108def call_builtin(name: str, args: list[Value], out: TextIO) -> Value:
109    """組み込み関数を呼び出す。"""
110    match name:
111        case "print":
112            out.write(" ".join(format_value(a) for a in args) + "\n")
113            return None
114        case "len":
115            target = args[0]
116            if not isinstance(target, (list, str)):
117                raise TinyRuntimeError("len の引数が不正です")
118            return len(target)
119        case "push":
120            array = args[0]
121            if not isinstance(array, list):
122                raise TinyRuntimeError("push の引数が不正です")
123            array.append(args[1])
124            return None
125        case "to_str":
126            return format_value(args[0])
127    raise TinyRuntimeError(f"未知の組み込み関数 {name} です")
