  2
  3名前解決（変数・関数が宣言されているか）と型検査を行い、
  4各式の ty 属性に型を書き込む。

 20class TypeCheckError(TinyError):
 21    """意味解析で見つかったエラー。"""

 24class Scope:
 25    """記号表。ブロックごとに 1 つ作り、外側のスコープへの参照を持つ。"""
 26
 27    def __init__(self, parent: Scope | None = None) -> None:
 28        self.parent = parent
 29        self.symbols: dict[str, Type] = {}
 30
 31    def declare(self, name: str, ty: Type, line: int) -> None:
 32        if name in self.symbols:
 33            raise TypeCheckError(f"変数 {name} はこのスコープで宣言済みです", line)
 34        self.symbols[name] = ty
 35
 36    def lookup(self, name: str) -> Type | None:
 37        scope: Scope | None = self
 38        while scope is not None:
 39            if name in scope.symbols:
 40                return scope.symbols[name]
 41            scope = scope.parent
 42        return None

 45ARITHMETIC_OPS = ("-", "*", "/", "%")
 46COMPARISON_OPS = ("<", "<=", ">", ">=")
 47EQUALITY_OPS = ("==", "!=")
 48LOGICAL_OPS = ("&&", "||")

 51def always_returns(stmt: Stmt) -> bool:
 52    """文の実行が必ず return で終わるかを（保守的に）判定する。"""
 53    if isinstance(stmt, Return):
 54        return True
 55    if isinstance(stmt, Block):
 56        return any(always_returns(s) for s in stmt.stmts)
 57    if isinstance(stmt, If):
 58        return (stmt.else_body is not None and always_returns(stmt.then_body)
 59                and always_returns(stmt.else_body))
 60    return False

 63class Checker:
 64    def __init__(self, program: Program) -> None:
 65        self.program = program
 66        self.functions: dict[str, FuncDef] = {}
 67        self.current: FuncDef | None = None
 68
 69    def check(self) -> None:
 70        for func in self.program.functions:
 71            if func.name in self.functions or func.name in BUILTINS:
 72                raise TypeCheckError(f"関数 {func.name} は定義済みです",
 73                                     func.line)
 74            self.functions[func.name] = func
 75        main = self.functions.get("main")
 76        if main is None:
 77            raise TypeCheckError("関数 main がありません")
 78        if main.params or main.return_type != VOID:
 79            raise TypeCheckError("main は引数も戻り値も持てません", main.line)
 80        for func in self.program.functions:
 81            self.check_function(func)
 82
 83    def check_function(self, func: FuncDef) -> None:
 84        self.current = func
 85        scope = Scope()
 86        for param in func.params:
 87            scope.declare(param.name, param.ty, param.line)
 88        self.check_block(func.body, Scope(scope))
 89        if func.return_type != VOID and not always_returns(func.body):
 90            raise TypeCheckError(
 91                f"関数 {func.name} の末尾に return がありません", func.line)
 92
 93    # ------------------------------------------------ 文
 94
 95    def check_block(self, block: Block, scope: Scope) -> None:
 96        for stmt in block.stmts:
 97            self.check_stmt(stmt, scope)
 98
 99    def check_stmt(self, stmt: Stmt, scope: Scope) -> None:
100        match stmt:
101            case Block():
102                self.check_block(stmt, Scope(scope))
103            case Let(name, declared, value):
104                ty = self.check_expr(value, scope, declared)
105                if ty == VOID:
106                    raise TypeCheckError("値を持たない式は代入できません",
107                                         stmt.line)
108                scope.declare(name, ty, stmt.line)
109            case Assign(target, value):
110                if isinstance(target, Index) and \
111                        self.check_expr(target.target, scope) == STR:
112                    raise TypeCheckError("文字列の要素には代入できません",
113                                         stmt.line)
114                target_type = self.check_expr(target, scope)
115                self.check_expr(value, scope, target_type)
116            case If(cond, then_body, else_body):
117                self.check_expr(cond, scope, BOOL)
118                self.check_block(then_body, Scope(scope))
119                if else_body is not None:
120                    self.check_stmt(else_body, scope)
121            case While(cond, body):
122                self.check_expr(cond, scope, BOOL)
123                self.check_block(body, Scope(scope))
124            case Return(value):
125                assert self.current is not None
126                expected = self.current.return_type
127                if value is None:
128                    if expected != VOID:
129                        raise TypeCheckError("戻り値が必要です", stmt.line)
130                elif expected == VOID:
131                    raise TypeCheckError(
132                        f"関数 {self.current.name} は値を返せません",
133                        stmt.line)
134                else:
135                    self.check_expr(value, scope, expected)
136            case ExprStmt(expr):
137                self.check_expr(expr, scope)
138
139    # ------------------------------------------------ 式
140
141    def check_expr(self, expr: Expr, scope: Scope,
142                   expected: Type | None = None) -> Type:
143        """式の型を求めて expr.ty に記録する。
144
145        expected が与えられたら、式の型がそれと一致することを確かめる。
146        空の配列リテラル [] の要素型は expected から決める。
147        """
148        if isinstance(expr, ArrayLit) and not expr.elements:
149            if not isinstance(expected, ArrayType):
150                raise TypeCheckError("空の配列の型を決められません",
151                                     expr.line)
152            ty: Type = expected
153        else:
154            ty = self.infer(expr, scope, expected)
155        if expected is not None and ty != expected:
156            raise TypeCheckError(
157                f"型 {expected} が必要ですが、型 {ty} の式があります",
158                expr.line)
159        expr.ty = ty
160        return ty
161
162    def infer(self, expr: Expr, scope: Scope, expected: Type | None) -> Type:
163        match expr:
164            case IntLit():
165                return INT
166            case BoolLit():
167                return BOOL
168            case StrLit():
169                return STR
170            case Name(name):
171                ty = scope.lookup(name)
172                if ty is None:
173                    raise TypeCheckError(f"変数 {name} は宣言されていません",
174                                         expr.line)
175                return ty
176            case ArrayLit(elements):
177                hint = expected.element \
178                    if isinstance(expected, ArrayType) else None
179                first = self.check_expr(elements[0], scope, hint)
180                for element in elements[1:]:
181                    self.check_expr(element, scope, first)
182                return ArrayType(first)
183            case Index(target, index):
184                target_type = self.check_expr(target, scope)
185                self.check_expr(index, scope, INT)
186                if isinstance(target_type, ArrayType):
187                    return target_type.element
188                if target_type == STR:
189                    return STR
190                raise TypeCheckError(f"型 {target_type} の値は添字で参照できません",
191                                     expr.line)
192            case Unary(op, operand):
193                return self.check_expr(operand, scope,
194                                       INT if op == "-" else BOOL)
195            case Binary(op, left, right):
196                return self.check_binary(op, left, right, scope)
197            case Call():
198                return self.check_call(expr, scope)
199        raise AssertionError(expr)
200
201    def check_binary(self, op: str, left: Expr, right: Expr,
202                     scope: Scope) -> Type:
203        if op in LOGICAL_OPS:
204            self.check_expr(left, scope, BOOL)
205            self.check_expr(right, scope, BOOL)
206            return BOOL
207        left_type = self.check_expr(left, scope)
208        if op in EQUALITY_OPS:
209            if left_type == VOID:
210                raise TypeCheckError("値を持たない式は比較できません",
211                                     left.line)
212            self.check_expr(right, scope, left_type)
213            return BOOL
214        if op == "+" and left_type == STR:
215            self.check_expr(right, scope, STR)
216            return STR
217        if left_type != INT:
218            raise TypeCheckError(
219                f"演算子 {op} は型 {left_type} に使えません", left.line)
220        self.check_expr(right, scope, INT)
221        return BOOL if op in COMPARISON_OPS else INT
222
223    def check_call(self, call: Call, scope: Scope) -> Type:
224        name, args = call.callee, call.args
225        if name in BUILTINS:
226            return self.check_builtin(call, scope)
227        func = self.functions.get(name)
228        if func is None:
229            raise TypeCheckError(f"関数 {name} は定義されていません",
230                                 call.line)
231        if len(args) != len(func.params):
232            raise TypeCheckError(
233                f"関数 {name} の引数は {len(func.params)} 個です", call.line)
234        for arg, param in zip(args, func.params):
235            self.check_expr(arg, scope, param.ty)
236        return func.return_type
237
238    def check_builtin(self, call: Call, scope: Scope) -> Type:
239        name, args = call.callee, call.args
240        if name == "print":
241            for arg in args:
242                if self.check_expr(arg, scope) == VOID:
243                    raise TypeCheckError("値を持たない式は表示できません",
244                                         arg.line)
245            return VOID
246        arity = 2 if name == "push" else 1
247        if len(args) != arity:
248            raise TypeCheckError(f"関数 {name} の引数は {arity} 個です",
249                                 call.line)
250        ty = self.check_expr(args[0], scope)
251        if name == "len":
252            if not isinstance(ty, ArrayType) and ty != STR:
253                raise TypeCheckError("len の引数は配列か文字列です",
254                                     call.line)
255            return INT
256        if name == "push":
257            if not isinstance(ty, ArrayType):
258                raise TypeCheckError("push の第 1 引数は配列です", call.line)
259            self.check_expr(args[1], scope, ty.element)
260            return VOID
261        # to_str
262        if ty not in (INT, BOOL, STR):
263            raise TypeCheckError(f"型 {ty} は to_str で変換できません",
264                                 call.line)
265        return STR

268def check(program: Program) -> Program:
269    """意味解析を行い、型情報を書き込んだ構文木を返す。"""
270    Checker(program).check()
271    return program
