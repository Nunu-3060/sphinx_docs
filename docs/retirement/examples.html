実行の方法

実行するには、次のものが必要です。

  • Python（3.10 以降）と matplotlib

  • Graphviz（dot コマンドを実行できるようにしておきます）

  • 日本語のフォント（BIZ UDGothic、Yu Gothic、Meiryo などのいずれか）

ダウンロードしたファイルを置いたフォルダーで、次のように入力して実行します。--out の後に、画像を保存するフォルダーを指定します。

1pip install matplotlib
2python make_figures.py --out figures

figures フォルダーに、9 つの PNG 画像が作成されます。本書をビルドするときは、Sphinx の設定ファイル（conf.py）からこのスクリプトを呼び出し、自動的に図を作成しています。

図を作成するスクリプト（make_figures.py）

Graphviz の dot ファイルを PNG 画像に変換し、matplotlib で 5 つのグラフを描きます。図の色は、本文の表や注記と調和する、落ち着いた青とオレンジを中心にしています。

 56def choose_font(candidates: list[str]) -> str:
 57    """候補のうち、インストールされている最初のフォントの名前を返す。"""
 58    installed = {f.name for f in font_manager.fontManager.ttflist}

 73def render_dot(src: Path, dst: Path) -> None:
 74    """Graphviz の dot ファイルを PNG 画像に変換する。"""
 75    dot = shutil.which('dot')
 76    if dot is None:
 77        raise RuntimeError('Graphviz の dot コマンドが見つかりません。')
 78    subprocess.run(
 79        [dot, '-Tpng', f'-Gdpi={DPI}', str(src), '-o', str(dst)],
 80        check=True,
 81    )

 84def style_axes(ax: Axes) -> None:
 85    """グラフの枠と目盛り線を目立たない見た目にする。"""

 93def plot_month_end(dst: Path) -> None:
 94    """月末に退職する場合と、その前日に退職する場合の保険の違いを描く。
 95
 96    3 月 28 日から 4 月 3 日までの 7 日間について、どの保険に加入しているかを
 97    2 本の帯で示す。日付は、3 月 28 日を 0 とした日数で表す。
 98    """
 99    days = ['3/28', '3/29', '3/30', '3/31', '4/1', '4/2', '4/3']
100    # (行の見出し, 退職日の位置, 資格喪失日の位置)
101    cases = [
102        ('3 月 31 日に退職', 3, 4),
103        ('3 月 30 日に退職', 2, 3),
104    ]
105
106    fig, ax = plt.subplots(figsize=(8, 2.8))
107    for row, (title, retire, lose) in enumerate(cases):
108        y = len(cases) - 1 - row
109        # 会社の健康保険・厚生年金保険（3 月 28 日から退職日まで）
110        ax.barh(y, lose, left=0, height=0.5, color=BLUE,
111                edgecolor='white', linewidth=2)
112        # 国民健康保険・国民年金（資格喪失日から）
113        ax.barh(y, len(days) - lose, left=lose, height=0.5, color=ORANGE,
114                edgecolor='white', linewidth=2)
115        ax.annotate(f'資格喪失日 {days[lose]}', xy=(lose, y + 0.25),
116                    xytext=(lose, y + 0.45), ha='center', fontsize=9,
117                    color=TEXT_SUB,
118                    arrowprops={'arrowstyle': '-', 'color': TEXT_SUB})
119        month = '会社の保険' if lose >= 4 else '国民健康保険・国民年金'
120        ax.text(len(days) + 0.15, y, f'3 月分の保険料：\n{month}',
121                va='center', fontsize=9, color=TEXT)
122
123    ax.set_yticks(range(len(cases)))
124    ax.set_yticklabels([c[0] for c in reversed(cases)])
125    ax.set_xticks([i + 0.5 for i in range(len(days))])
126    ax.set_xticklabels(days)
127    ax.set_xlim(0, len(days))
128    ax.set_ylim(-0.5, len(cases) + 0.05)
129    style_axes(ax)
130    ax.grid(False)
131    ax.axvline(4, color=TEXT_SUB, linewidth=1, linestyle=':')
132    ax.text(4, len(cases) - 0.15, '月の変わり目', ha='center', fontsize=9,
133            color=TEXT_SUB)
134    ax.legend(
135        handles=[
136            Patch(color=BLUE, label='会社の健康保険・厚生年金保険'),
137            Patch(color=ORANGE, label='国民健康保険・国民年金'),
138        ],
139        loc='upper center', bbox_to_anchor=(0.5, -0.18), ncol=2,
140        frameon=False, fontsize=9,
141    )
142    fig.savefig(dst, dpi=DPI, bbox_inches='tight', facecolor='white')
143    plt.close(fig)

198def retirement_deduction(years: int) -> int:
199    """勤続年数から退職所得控除額（万円）を求める。"""
200    if years <= 20:
201        return max(80, 40 * years)
202    return 800 + 70 * (years - 20)
203
204
205def plot_retirement_deduction(dst: Path) -> None:
206    """勤続年数と退職所得控除額の関係を描く。"""
207    years = list(range(1, 41))
208    amounts = [retirement_deduction(y) for y in years]
209
210    fig, ax = plt.subplots(figsize=(8, 3.6))
211    ax.plot(years, amounts, color=BLUE, linewidth=2)
212    for y in (20, 30):
213        value = retirement_deduction(y)
214        ax.plot(y, value, marker='o', markersize=7, color=BLUE,
215                markeredgecolor='white', markeredgewidth=2)
216        ax.annotate(f'{y} 年：{value:,} 万円', xy=(y, value),
217                    xytext=(-12, 10), textcoords='offset points',
218                    ha='right', fontsize=9, color=TEXT)
219    ax.text(15, 230, '1 年あたり 40 万円', ha='center', fontsize=9,
220            color=TEXT_SUB)
221    ax.text(31, 900, '1 年あたり 70 万円', ha='center', fontsize=9,
222            color=TEXT_SUB)
223
224    ax.set_xlim(0, 41)
225    ax.set_ylim(0, 2400)
226    ax.set_xticks([1, 5, 10, 15, 20, 25, 30, 35, 40])
227    ax.set_xlabel('勤続年数（年）')
228    ax.set_ylabel('退職所得控除額（万円）')
229    ax.yaxis.set_major_formatter(
230        FuncFormatter(lambda value, _pos: f'{value:,.0f}'))
231    for side in ('top', 'right'):
232        ax.spines[side].set_visible(False)
233    ax.grid(axis='y', color=GRID, linewidth=0.8)
234    ax.set_axisbelow(True)
235    fig.savefig(dst, dpi=DPI, bbox_inches='tight', facecolor='white')
236    plt.close(fig)

239def plot_schedule(dst: Path) -> None:
240    """退職日から逆算した、引き継ぎと年次有給休暇の消化の日程の例を描く。
241
242    横軸は退職日を 0 とした週数で、退職日の約 3 か月前（12 週前）に
243    退職の意思を伝え、残りの有給休暇（約 15 日）を最後の 3 週間で

245    """
246    notice = -12
247    handover_start = -10
248    last_day = -3
249    retire = 0
250
251    fig, ax = plt.subplots(figsize=(8, 2.9))

262                edgecolor='white', linewidth=2)
263        ax.text((start + end) / 2, y, label, ha='center', va='center',
264                fontsize=9, color='white')
265    ax.plot(notice, len(rows) - 1, marker='D', markersize=9, color=BLUE,

269
270    for x, label in ((last_day, '最終出社日'), (retire, '退職日')):
271        ax.axvline(x, color=TEXT_SUB, linewidth=1, linestyle=':')

288def plot_resident_tax(dst: Path) -> None:
289    """住民税の課税の対象となる所得と、納める期間の関係を描く。
290
291    横軸は、退職した年（N 年）の 1 月から翌年（N + 1 年）の 12 月までの
292    24 か月で、0 が N 年 1 月を表す。住民税は、前年の所得にかかり、
293    6 月から翌年 5 月までの 12 回に分けて納める。
294    """
295    # (行の見出し, 開始の月, 終了の月, 色)
296    rows = [
297        ('N − 2 年の所得', 0, 5, GRAY),
298        ('N − 1 年の所得', 5, 17, BLUE),
299        ('N 年の所得\n（退職した年）', 17, 24, ORANGE),
300    ]
301    # 退職した月と、住民税の残りの納め方
302    periods = [
303        (0, 4, '1〜4 月に退職：\n5 月分まで一括'),
304        (4, 5, '5 月'),
305        (5, 12, '6〜12 月に退職：次から選ぶ\n一括、普通徴収、\n転職先での特別徴収'),
306    ]
307
308    fig, ax = plt.subplots(figsize=(8.5, 3.6))
309    n = len(rows) + 1
310    for i, (title, start, end, color) in enumerate(rows):
311        y = n - 1 - i
312        ax.barh(y, end - start, left=start, height=0.5, color=color,
313                edgecolor='white', linewidth=2)
314    ax.text(11, n - 2, '6 月から翌年 5 月までの 12 回で納める',
315            ha='center', va='center', fontsize=9, color='white')
316    ax.text(20.5, n - 3, '翌年 6 月から納める', ha='center',
317            va='center', fontsize=9, color='white')
318    for start, end, label in periods:
319        ax.barh(0, end - start, left=start, height=0.8, color='white',
320                edgecolor=TEXT_SUB, linewidth=1)
321        if end - start > 1:
322            ax.text((start + end) / 2, 0, label, ha='center', va='center',
323                    fontsize=8.5, color=TEXT)
324    ax.annotate('5 月に退職：5 月分を差し引く', xy=(4.5, -0.4),
325                xytext=(4.5, -1.1), ha='center', fontsize=8.5, color=TEXT,
326                arrowprops={'arrowstyle': '-', 'color': TEXT_SUB})
327
328    ax.axvline(12, color=TEXT_SUB, linewidth=1, linestyle=':')
329    ax.text(6, n - 0.3, '退職した年（N 年）', ha='center', fontsize=9,
330            color=TEXT_SUB)
331    ax.text(18, n - 0.3, '翌年（N + 1 年）', ha='center', fontsize=9,
332            color=TEXT_SUB)
333
334    labels = ['退職した月と\n住民税の納め方'] + [r[0] for r in reversed(rows)]
335    ax.set_yticks(range(n))
336    ax.set_yticklabels(labels)
337    ax.set_xticks([m + 0.5 for m in range(24)])
338    ax.set_xticklabels([str(m % 12 + 1) for m in range(24)], fontsize=8)
339    ax.set_xlim(0, 24)
340    ax.set_ylim(-1.4, n + 0.1)
341    ax.set_xlabel('月')
342    style_axes(ax)
343    ax.grid(False)
344    fig.savefig(dst, dpi=DPI, bbox_inches='tight', facecolor='white')
345    plt.close(fig)

348def make_all(out_dir: Path) -> list[Path]:
349    """すべての図を作成し、作成したファイルの一覧を返す。"""
350    out_dir.mkdir(parents=True, exist_ok=True)
351    made: list[Path] = []
352
353    dot_names = (
354        'retirement_flow', 'deadlines', 'health_insurance_choice',
355        'pension_types',
356    )
357    for name in dot_names:
358        dst = out_dir / f'{name}.png'
359        render_dot(HERE / f'{name}.dot', dst)
360        made.append(dst)
361
362    plots = {
363        'month_end.png': plot_month_end,
364        'schedule.png': plot_schedule,
365        'benefit_timeline.png': plot_benefit_timeline,
366        'retirement_deduction.png': plot_retirement_deduction,
367        'resident_tax.png': plot_resident_tax,
368    }
369    for filename, plot in plots.items():
370        dst = out_dir / filename
371        plot(dst)
372        made.append(dst)
373    return made
