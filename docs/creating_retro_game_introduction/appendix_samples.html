 4スペースキーを押した回数 (btnp)、押し続けたときのリピート回数 (btnp の hold と repeat)、
 5離した回数 (btnr) を数えて表示する。

 9

15        self.press_count = 0

17        self.release_count = 0

20        # btn: 押されている間、毎フレーム True になる

29

37        if pyxel.btnr(pyxel.KEY_SPACE):
38            self.release_count += 1
39

42        pyxel.text(4, 4, "ARROW KEYS: MOVE", 7)

第 3 章：速度と加速度による動き。motion_demo.py をダウンロード

リスト 5 ch03_basics/motion_demo.py
 1"""第 3 章: 速度と加速度による動き。
 2
 3ボールに重力 (下向きの加速度) をかけ、壁と床で跳ね返らせる。
 4スペースキーで新しいボールを投げる。
 5"""
 6
 7import pyxel
 8
 9SCREEN_W = 160
10SCREEN_H = 120
11GRAVITY = 0.3
12BOUNCE = 0.8
13RADIUS = 4
14
15
16class Ball:
17    def __init__(self) -> None:
18        self.x = 20.0
19        self.y = 20.0
20        self.vx = pyxel.rndf(1, 3)
21        self.vy = pyxel.rndf(-3, 0)
22        self.col = pyxel.rndi(8, 15)
23
24    def update(self) -> None:
25        # 加速度で速度を変え、速度で位置を変える
26        self.vy += GRAVITY
27        self.x += self.vx
28        self.y += self.vy
29
30        # 壁や床に当たったら、画面内に戻して速度の向きを反転させる
31        if self.x < RADIUS:
32            self.x = RADIUS
33            self.vx = -self.vx
34        if self.x > SCREEN_W - RADIUS:
35            self.x = SCREEN_W - RADIUS
36            self.vx = -self.vx
37        if self.y > SCREEN_H - RADIUS:
38            self.y = SCREEN_H - RADIUS
39            self.vy = -self.vy * BOUNCE
40
41    def draw(self) -> None:
42        pyxel.circ(self.x, self.y, RADIUS, self.col)
43
44
45class App:
46    def __init__(self) -> None:
47        self.balls = [Ball()]
48
49    def update(self) -> None:
50        if pyxel.btnp(pyxel.KEY_SPACE):
51            self.balls.append(Ball())
52        for ball in self.balls:
53            ball.update()
54
55    def draw(self) -> None:
56        pyxel.cls(0)
57        pyxel.text(4, 4, "SPACE: THROW A BALL", 7)
58        for ball in self.balls:
59            ball.draw()
60
61
62def main() -> None:
63    pyxel.init(SCREEN_W, SCREEN_H, title="Motion Demo")
64    app = App()
65    pyxel.run(app.update, app.draw)
66
67
68if __name__ == "__main__":
69    main()

リスト 6 ch03_basics/collision_demo.py
 1"""第 3 章: 当たり判定。
 2
 3矢印キーで動かす四角形と円が、固定された四角形と円に重なると色が変わる。
 4四角形どうしは AABB 判定、円どうしは中心間の距離で判定する。
 5"""
 6
 7import pyxel
 8
 9
10def rects_overlap(
11    x1: float, y1: float, w1: float, h1: float,
12    x2: float, y2: float, w2: float, h2: float,
13) -> bool:
14    """2 つの矩形が重なっていれば True を返す (AABB 判定)。"""
15    return x1 < x2 + w2 and x2 < x1 + w1 and y1 < y2 + h2 and y2 < y1 + h1
16
17
18def circles_overlap(x1: float, y1: float, r1: float, x2: float, y2: float, r2: float) -> bool:
19    """2 つの円が重なっていれば True を返す。平方根を避けるため 2 乗どうしで比べる。"""
20    dx = x2 - x1
21    dy = y2 - y1
22    return dx * dx + dy * dy < (r1 + r2) * (r1 + r2)
23
24
25class App:
26    def __init__(self) -> None:
27        self.x = 50.0
28        self.y = 40.0
29
30    def update(self) -> None:
31        if pyxel.btn(pyxel.KEY_LEFT):
32            self.x -= 1.5
33        if pyxel.btn(pyxel.KEY_RIGHT):
34            self.x += 1.5
35        if pyxel.btn(pyxel.KEY_UP):
36            self.y -= 1.5
37        if pyxel.btn(pyxel.KEY_DOWN):
38            self.y += 1.5
39
40    def draw(self) -> None:
41        pyxel.cls(1)
42        pyxel.text(4, 4, "ARROW KEYS: MOVE", 7)
43
44        # 四角形どうし: 動かす四角形 (self.x, self.y, 16, 16) と固定の四角形
45        rect_hit = rects_overlap(self.x, self.y, 16, 16, 60, 30, 24, 16)
46        pyxel.rect(60, 30, 24, 16, 8 if rect_hit else 13)
47        pyxel.rectb(self.x, self.y, 16, 16, 10)
48
49        # 円どうし: 動かす円 (四角形の中心、半径 6) と固定の円
50        cx = self.x + 8
51        cy = self.y + 8
52        circle_hit = circles_overlap(cx, cy, 6, 110, 80, 14)
53        pyxel.circ(110, 80, 14, 8 if circle_hit else 13)
54        pyxel.circb(cx, cy, 6, 11)
55
56        rect_label = "HIT" if rect_hit else "---"
57        circle_label = "HIT" if circle_hit else "---"
58        pyxel.text(4, 110, f"RECT: {rect_label}  CIRCLE: {circle_label}", 7)
59
60
61def main() -> None:
62    pyxel.init(160, 120, title="Collision Demo")
63    app = App()
64    pyxel.run(app.update, app.draw)
65

68    main()

リスト 7 ch03_basics/random_demo.py
 1"""第 3 章: 乱数とタイマー。
 2
 310 フレームごとに、ランダムな位置と色の星を出現させる。
 4星は出現してから 60 フレーム (2 秒) たつと消える。
 5"""
 6
 7import pyxel
 8
 9SPAWN_INTERVAL = 10
10LIFETIME = 60
11
12
13class Star:
14    def __init__(self) -> None:
15        self.x = pyxel.rndi(0, 159)
16        self.y = pyxel.rndi(12, 119)
17        self.col = pyxel.rndi(8, 15)
18        self.born = pyxel.frame_count
19
20    def is_expired(self) -> bool:
21        return pyxel.frame_count - self.born >= LIFETIME
22
23    def draw(self) -> None:
24        # 残り時間が短くなったら点滅させる
25        remaining = LIFETIME - (pyxel.frame_count - self.born)
26        if remaining < 20 and pyxel.frame_count % 4 < 2:
27            return
28        pyxel.circ(self.x, self.y, 2, self.col)
29
30
31class App:
32    def __init__(self) -> None:
33        self.stars: list[Star] = []
34
35    def update(self) -> None:
36        if pyxel.frame_count % SPAWN_INTERVAL == 0:
37            self.stars.append(Star())
38        self.stars = [star for star in self.stars if not star.is_expired()]
39
40    def draw(self) -> None:
41        pyxel.cls(0)
42        pyxel.text(4, 4, f"FRAME {pyxel.frame_count}  STARS {len(self.stars)}", 7)
43        for star in self.stars:
44            star.draw()
45
46
47def main() -> None:
48    pyxel.init(160, 120, title="Random Demo")
49    app = App()
50    pyxel.run(app.update, app.draw)
51
