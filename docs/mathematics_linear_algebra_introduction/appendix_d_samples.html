付録 D サンプルコード一覧

本書で使ったサンプルコードの一覧である。ファイル名をクリックすると、そのファイルをダウンロードできる。すべてのファイルをまとめた examples.zip もある。ZIP ファイルを展開すると、examples フォルダーにすべてのファイルが置かれる。

実行と検査の方法

サンプルは、examples フォルダーで実行する。図を描くサンプルは補助モジュール la_plot.py を import するので、個別にダウンロードした場合は、一覧の「補助モジュール」の列に示したファイルも同じフォルダーに置く。

1$ cd examples
2$ python ch02_vector_ops.py
3$ python ch04_transforms.py

計算の結果を表示するサンプル（report() を持つもの）は、結果を画面に表示する。図を描くサンプル（create_figure() を持つもの）は、図をウィンドウに表示する。両方を持つサンプルは、結果を表示してから図を表示する。

サンプルは、flake8（PEP 8 などのコーディング規約の検査）と mypy（型ヒントの検査）で違反がないことを確認している。examples フォルダーで次のコマンドを実行すると、同じフォルダーの setup.cfg の設定で検査できる。

1$ cd examples
2$ python -m flake8 .
3$ python -m mypy .

 5* draw_vector()  矢印でベクトルを描く
 6* draw_grid()    方眼を行列で変換した図を描く
 7* letter_f()     向きの分かる図形（F の字）の頂点を返す

10from collections.abc import Sequence
11
12import matplotlib.pyplot as plt
13import numpy as np
14from matplotlib import font_manager
15from matplotlib.axes import Axes
16from numpy.typing import ArrayLike, NDArray
17
18# Windows、macOS、Linux で日本語を表示できる代表的なフォント
19FONT_CANDIDATES = [
20    "Yu Gothic",
21    "Meiryo",
22    "Hiragino Sans",
23    "Noto Sans CJK JP",
24    "Noto Sans JP",
25    "IPAexGothic",
26]

29def setup_font() -> None:
30    """日本語を表示できるフォントを matplotlib に設定する.
31
32    候補のフォントのうち、最初に見つかったものを使う。どれも見つからない
33    場合は既定のフォントのままになり、日本語の文字は豆腐（□）で表示される。
34    """
35    available = {font.name for font in font_manager.fontManager.ttflist}
36    for name in FONT_CANDIDATES:
37        if name in available:
38            plt.rcParams["font.family"] = name
39            break
40    # 日本語フォントの多くはマイナス記号（U+2212）を持たないため、
41    # 負の数の目盛りにハイフンを使う
42    plt.rcParams["axes.unicode_minus"] = False
43    # 数式の文字は、本文の数式に近い STIX フォントで描く
44    plt.rcParams["mathtext.fontset"] = "stix"
45

47def setup_plane(ax: Axes, xlim: tuple[float, float], ylim: tuple[float, float]) -> None:
48    """縦横の目盛りが等しい座標平面を用意し、x 軸と y 軸を描く."""
49    ax.set_xlim(*xlim)
50    ax.set_ylim(*ylim)
51    ax.set_aspect("equal")
52    ax.grid(True, color="0.9")
53    ax.axhline(0, color="0.5", linewidth=0.8)
54    ax.axvline(0, color="0.5", linewidth=0.8)
55    ax.set_axisbelow(True)

58def draw_vector(ax: Axes, vector: ArrayLike, origin: ArrayLike = (0.0, 0.0), color: str = "C0",
59                label: str | None = None, offset: tuple[float, float] = (0.1, 0.1),
60                linestyle: str = "-") -> None:
61    """origin を始点とする矢印でベクトルを描き、終点の近くに label を書く."""
62    start = np.asarray(origin, dtype=float)
63    end = start + np.asarray(vector, dtype=float)
64    ax.annotate("", xy=(end[0], end[1]), xytext=(start[0], start[1]),
65                arrowprops={"arrowstyle": "-|>", "color": color, "linewidth": 2,
66                            "linestyle": linestyle, "shrinkA": 0, "shrinkB": 0,
67                            "mutation_scale": 15})
68    if label is not None:
69        ax.text(end[0] + offset[0], end[1] + offset[1], label, color=color, fontsize=12)

72def draw_grid(ax: Axes, matrix: ArrayLike, extent: int = 3, color: str = "C0") -> None:
73    """-extent から extent までの方眼の直線を、行列 matrix で変換して描く."""
74    a = np.asarray(matrix, dtype=float)
75    t = np.linspace(-extent, extent, 2)
76    for k in range(-extent, extent + 1):
77        for line in (np.vstack([t, np.full(2, k)]), np.vstack([np.full(2, k), t])):
78            moved = a @ line
79            ax.plot(moved[0], moved[1], color=color, linewidth=0.6, alpha=0.5)

 1"""内積と正射影: u を v の方向の成分と、v に直交する成分に分ける.
 2

 4"""
 5
 6import matplotlib.pyplot as plt
 7import numpy as np
 8from matplotlib.figure import Figure
 9
10import la_plot
11
12
13def create_figure() -> Figure:
14    """u の v 方向への正射影 p と、残りの成分 u - p を描く."""
15    u = np.array([2.0, 3.0])
16    v = np.array([4.0, 1.0])
17    p = (u @ v) / (v @ v) * v  # 正射影
18    fig, ax = plt.subplots(figsize=(5.5, 4.2))
19    la_plot.setup_plane(ax, (-0.5, 4.8), (-0.5, 3.6))
20
21    # v を含む直線
22    t = np.array([-0.2, 1.15])
23    ax.plot(t * v[0], t * v[1], color="0.6", linestyle=":")
24    la_plot.draw_vector(ax, v, color="C1", label=r"$\mathbf{v}$", offset=(0.1, -0.3))
25    la_plot.draw_vector(ax, u, color="C0", label=r"$\mathbf{u}$", offset=(-0.2, 0.15))
26    la_plot.draw_vector(ax, p, color="C2", label=r"$\mathbf{p}$", offset=(-0.1, -0.45))
27    la_plot.draw_vector(ax, u - p, origin=p, color="C3", linestyle="--",
28                        label=r"$\mathbf{u}-\mathbf{p}$", offset=(0.1, -0.4))
29
30    # 直角の記号
31    d1 = v / np.linalg.norm(v) * 0.25
32    d2 = (u - p) / np.linalg.norm(u - p) * 0.25
33    corner = np.array([p + d1, p + d1 + d2, p + d2])
34    ax.plot(corner[:, 0], corner[:, 1], color="0.3", linewidth=0.8)
35
36    ax.set_title(r"$\mathbf{u}$ の $\mathbf{v}$ 方向への正射影")
37    fig.tight_layout()
38    return fig
39

42    la_plot.setup_font()
43    create_figure()
44    plt.show()

ch02_vector_ops.py

 1"""ベクトルの演算: 和、スカラー倍、内積、ノルム、なす角、正射影、外積.
 2
 3使い方: python ch02_vector_ops.py
 4"""
 5
 6import numpy as np
 7from numpy.typing import NDArray
 8
 9Vector = NDArray[np.float64]
10
11
12def angle_between(u: Vector, v: Vector) -> float:
13    """2 つのベクトルのなす角を度で返す."""
14    cos_theta = (u @ v) / (np.linalg.norm(u) * np.linalg.norm(v))
15    # 丸め誤差で cos が [-1, 1] をわずかに超えると arccos が nan になるので、範囲内に収める
16    return float(np.degrees(np.arccos(np.clip(cos_theta, -1.0, 1.0))))
17
18
19def project(u: Vector, v: Vector) -> Vector:
20    """u の v 方向への正射影を返す."""
21    return (u @ v) / (v @ v) * v
22
23
24def report() -> None:
25    """2 次元と 3 次元のベクトルの演算の結果を表示する."""
26    u = np.array([3.0, 1.0])
27    v = np.array([1.0, 2.0])
28    print("u + v =", u + v)
29    print("2 * u =", 2 * u)
30    print("u @ v =", u @ v)  # 内積（np.dot(u, v) と同じ）
31    print(f"|u| = {np.linalg.norm(u):.4f}")
32    print(f"|v| = {np.linalg.norm(v):.4f}")
33    print(f"なす角 = {angle_between(u, v):.1f} 度")
34    p = project(u, v)
35    print("u の v 方向への正射影 p =", p)
36    print("(u - p) @ v =", (u - p) @ v)  # 残りの成分は v と直交する

 1"""ガウスの消去法で連立一次方程式を解く.
 2
 3前進消去で拡大係数行列を上三角の形にし、後退代入で解を求める。
 4pivoting=True のときは、各列で絶対値が最大の行をピボットの行に選ぶ
 5（部分ピボット選択。第 13 章で扱う）。
 6
 7使い方: python ch05_gauss.py
 8"""
 9
10import numpy as np
11from numpy.typing import ArrayLike, NDArray
12
13Vector = NDArray[np.float64]
14
15
16def gaussian_elimination(a: ArrayLike, b: ArrayLike, pivoting: bool = False,
17                         verbose: bool = False) -> Vector:
18    """ガウスの消去法で a x = b を解き、x を返す."""
19    # 拡大係数行列 [a | b] を作る（元の配列は書き換えない）
20    m = np.hstack([np.array(a, dtype=float), np.array(b, dtype=float).reshape(-1, 1)])
21    n = m.shape[0]
22
23    # 前進消去: k 列目の、k 行目より下の成分を 0 にする
24    for k in range(n - 1):
25        if pivoting:
26            p = k + int(np.argmax(np.abs(m[k:, k])))
27            if p != k:
28                m[[k, p]] = m[[p, k]]  # 行を入れ替える
29                if verbose:
30                    print(f"行 {k + 1} と行 {p + 1} を入れ替える")
31        if m[k, k] == 0.0:
32            raise ZeroDivisionError(f"ピボットが 0 になった（{k + 1} 列目）")
33        for i in range(k + 1, n):
34            factor = m[i, k] / m[k, k]
35            m[i] -= factor * m[k]  # 行 i から、行 k の factor 倍を引く
36        if verbose:
37            print(f"{k + 1} 列目の消去の後:")
38            print(m)
39
40    # 後退代入: 下の行から順に未知数を求める
41    x = np.zeros(n)
42    for i in range(n - 1, -1, -1):
43        x[i] = (m[i, -1] - m[i, i + 1:n] @ x[i + 1:]) / m[i, i]
44    return x

48    """3 元連立一次方程式を消去法で解き、途中経過と解を表示する."""

55        print("解 x =", x)

 1"""LU 分解: 係数行列を下三角行列 L と上三角行列 U の積に分解して解く.
 2
 3一度分解しておけば、右辺 b が変わっても、前進代入と後退代入だけで
 4解を求められる。ここでは簡単のため、ピボット選択をしない。
 5
 6使い方: python ch05_lu.py
 7"""
 8
 9import numpy as np
10from numpy.typing import ArrayLike, NDArray
11
12Matrix = NDArray[np.float64]
13
14
15def lu_decompose(a: ArrayLike) -> tuple[Matrix, Matrix]:
16    """a = L U となる単位下三角行列 L と上三角行列 U を返す."""
17    u = np.array(a, dtype=float)
18    n = u.shape[0]
19    lower = np.eye(n)
20    for k in range(n - 1):
21        for i in range(k + 1, n):
22            lower[i, k] = u[i, k] / u[k, k]  # 消去に使った倍率を L に記録する
23            u[i] -= lower[i, k] * u[k]
24    return lower, u
25
26
27def lu_solve(lower: Matrix, upper: Matrix, b: ArrayLike) -> Matrix:

ch08_rank.py

 1"""行列のランクと、零空間の基底を求める.
 2
 3行基本変形で被約階段行列（RREF）を作り、ピボットのある列の数からランクを、
 4自由変数から零空間の基底を求める。
 5
 6使い方: python ch08_rank.py
 7"""
 8
 9import numpy as np
10from numpy.typing import ArrayLike, NDArray
11
12Matrix = NDArray[np.float64]
13
14
15def rref(a: ArrayLike, tol: float = 1e-10) -> tuple[Matrix, list[int]]:
16    """被約階段行列と、ピボットのある列の番号のリストを返す."""
17    m = np.array(a, dtype=float)
18    rows, cols = m.shape
19    pivots: list[int] = []
20    r = 0  # 次にピボットを置く行
21    for c in range(cols):
22        if r == rows:
23            break
24        p = r + int(np.argmax(np.abs(m[r:, c])))
25        if abs(m[p, c]) < tol:
26            continue  # この列にはピボットがない（自由変数の列）
27        m[[r, p]] = m[[p, r]]
28        m[r] /= m[r, c]  # ピボットを 1 にする
29        for i in range(rows):
30            if i != r:
31                m[i] -= m[i, c] * m[r]  # ピボットの上下を 0 にする
32        pivots.append(c)
33        r += 1
34    m[np.abs(m) < tol] = 0.0
35    return m, pivots
36
37
38def null_space_basis(a: ArrayLike) -> list[Matrix]:
39    """零空間の基底を、自由変数を 1 つずつ 1 にした解として返す."""
40    r, pivots = rref(a)
41    cols = r.shape[1]
42    free = [c for c in range(cols) if c not in pivots]
43    basis = []
44    for f in free:
45        x = np.zeros(cols)
46        x[f] = 1.0
47        for row, p in enumerate(pivots):
48            x[p] = -r[row, f] + 0.0  # + 0.0 で -0.0 を 0.0 にする
49        basis.append(x)
50    return basis
51
52
53def report() -> None:
54    """ランク、被約階段行列、4 つの基本部分空間の次元を表示する."""
55    a = np.array([[1.0, 2.0, 1.0, 3.0],
56                  [2.0, 4.0, 0.0, 2.0],
57                  [3.0, 6.0, 1.0, 5.0]])
58    r, pivots = rref(a)
59    m, n = a.shape
60    rank = len(pivots)
61    print("被約階段行列 =")
62    print(r)
63    print("ピボットのある列（0 から数える）:", pivots)
64    print("ランク =", rank, " np.linalg.matrix_rank =", np.linalg.matrix_rank(a))
65    with np.printoptions(suppress=True):  # 丸め誤差の小さな値を 0 と表示する
66        for x in null_space_basis(a):
67            print("零空間の基底ベクトル", x, " a @ x =", a @ x)
68    print(f"列空間の次元 {rank}、零空間の次元 {n - rank}（R^{n} の中）")
69    print(f"行空間の次元 {rank}、左零空間の次元 {m - rank}（R^{m} の中）")
70    y = np.array([1.0, 1.0, -1.0])
71    print("左零空間の基底ベクトル", y, " y @ a =", y @ a)

 9import numpy as np
10from numpy.typing import NDArray
11
12Matrix = NDArray[np.float64]

15def gram_schmidt(a: Matrix) -> Matrix:
16    """a の列ベクトルを直交化し、正規直交基底を列に並べた行列を返す."""
17    q = np.zeros_like(a, dtype=float)
18    for j in range(a.shape[1]):

20        for i in range(j):
21            v -= (q[:, i] @ a[:, j]) * q[:, i]  # 求めた基底の方向の成分を除く
22        q[:, j] = v / np.linalg.norm(v)  # 長さを 1 にする

16LINKS = {

30    for source, targets in LINKS.items():

33            m[PAGES.index(target), j] = 1.0 / len(targets)  # j 列目はページ j からの遷移確率

37def power_iteration(g: Matrix, tol: float = 1e-10) -> tuple[Matrix, int]:
38    """べき乗法で、固有値 1 の固有ベクトル（成分の和が 1）と反復回数を返す."""
39    n = g.shape[0]
40    x = np.full(n, 1.0 / n)
41    for count in range(1, 1001):
42        x_next = g @ x
43        if np.abs(x_next - x).sum() < tol:

20def make_data() -> Matrix:
21    """相関のある 2 次元の架空のデータを、200 行 2 列の配列で返す."""
22    rng = np.random.default_rng(seed=3)
23    cov = np.array([[3.0, 1.8],
24                    [1.8, 1.5]])
25    return rng.multivariate_normal([2.0, 1.0], cov, size=200)

28def pca(data: Matrix) -> tuple[Matrix, Matrix, Matrix]:
29    """平均、固有値（大きい順）、固有ベクトル（列、同じ順）を返す."""
30    mean = data.mean(axis=0)
31    centered = data - mean
32    cov = centered.T @ centered / (len(data) - 1)  # 共分散行列
33    values, vectors = np.linalg.eigh(cov)
34    order = np.argsort(values)[::-1]  # 大きい順に並べ替える
35    values, vectors = values[order], vectors[:, order]
36    # 固有ベクトルの符号は任意なので、各列の絶対値が最大の成分が正になるようにそろえる
37    signs = np.sign(vectors[np.argmax(np.abs(vectors), axis=0), np.arange(vectors.shape[1])])
38    return mean, values, vectors * signs

 23def rotation_x(degrees: float) -> Matrix:
 24    """x 軸のまわりの回転（同次座標）."""
 25    c, s = np.cos(np.radians(degrees)), np.sin(np.radians(degrees))

 32    return np.array([[c, 0, s, 0], [0, 1, 0, 0], [-s, 0, c, 0], [0, 0, 0, 1]], dtype=float)

 35def translation(tx: float, ty: float, tz: float) -> Matrix:
 36    """平行移動（同次座標）."""

 42def perspective(focal: float) -> Matrix:
 43    """透視投影. 変換後の w 成分が元の z になり、w で割ると (f x / z, f y / z) になる."""
 44    return np.array([[focal, 0, 0, 0], [0, focal, 0, 0], [0, 0, 1, 0], [0, 0, 1, 0]], dtype=float)

 47def cube() -> tuple[Matrix, list[tuple[int, int]]]:
 48    """1 辺の長さ 2 の立方体の頂点（4 行 8 列の同次座標）と辺のリストを返す."""
 49    corners = np.array(list(product((-1.0, 1.0), repeat=3))).T
 50    vertices = np.vstack([corners, np.ones(8)])
 51    # 座標が 1 つだけ異なる頂点の組が辺になる
 52    edges = [(i, j) for i, j in combinations(range(8), 2)
 53             if np.sum(corners[:, i] != corners[:, j]) == 1]
 54    return vertices, edges
