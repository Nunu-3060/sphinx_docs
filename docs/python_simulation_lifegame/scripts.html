utility.py

hsv2rgb 関数を提供する共通ユーティリティモジュール。他のどのモジュールにも依存せず、LifeGameImage.py と TkWidget.py の双方から利用される。

ダウンロード

 1import numpy as np
 2
 3
 4def hsv2rgb(
 5    h: np.typing.NDArray[np.float64] | float,
 6    s: np.typing.NDArray[np.float64] | float,
 7    v: np.typing.NDArray[np.float64] | float,
 8) -> np.typing.NDArray[np.uint8]:
 9    phase_offsets: np.typing.NDArray[np.float64]
10    rgb: np.typing.NDArray[np.float64]
11    if isinstance(h, float) and isinstance(s, float) and isinstance(v, float):
12        phase_offsets = np.array([0, 2/3, 1/3], np.float64)
13        rgb = ((np.clip(np.abs(np.mod(h + phase_offsets, 1) * 6 - 3) - 1, 0, 1) - 1) * s + 1) * v
14    elif isinstance(h, np.ndarray) and isinstance(s, np.ndarray) and isinstance(v, np.ndarray):
15        if h.shape != s.shape or h.shape != v.shape:
16            raise TypeError("adjust the array lengths to be the same.")
17        elif len(h.shape) != 2:
18            raise TypeError("input 2D numpy array.")
19        phase_offsets = np.zeros((h.shape[0], h.shape[1], 3), np.float64)
20        phase_offsets[:, :, 1] = 2 / 3
21        phase_offsets[:, :, 2] = 1 / 3
22        hue: np.typing.NDArray[np.float64] = h[:, :, None]
23        saturate: np.typing.NDArray[np.float64] = s[:, :, None]
24        value: np.typing.NDArray[np.float64] = v[:, :, None]
25        rgb = ((np.clip(np.abs(np.mod(hue + phase_offsets, 1) * 6 - 3) - 1, 0, 1) - 1) * saturate + 1) * value
26    else:
27        text: str = "\n".join([
28            "h, s, and v must all be float or np.ndarray.",
29            f"h: {type(h).__name__}",
30            f"s: {type(s).__name__}",
31            f"v: {type(v).__name__}",
32        ])
33        raise TypeError(text)
34    rgb = np.clip(rgb, 0, 1)
35    return (rgb * 255).round(0).astype(np.uint8)

LifeGame.py

ライフゲームの中核となるロジックを実装したモジュール。格子の状態（lattice）の保持、近傍セル数の計算、次世代への更新（next()）、周期境界条件・固定境界条件の切り替えを行う。

ダウンロード

 1from collections.abc import Callable
 2
 3import numpy as np
 4
 5
 6class LifeGame:
 7    def __init__(
 8        self,
 9        rule: list[list[int]] | None = None,
10        width: int = 256,
11        height: int = 256,
12        probability: float = 0.3,
13        tb: bool = True,
14        lr: bool = True,
15    ) -> None:
16
17        if rule is None:
18            rule = [[3], [2, 3]]
19        self.rule: list[list[int]] = rule
20
21        rng: np.random.Generator = np.random.default_rng()
22        self.lattice: np.typing.NDArray[np.int64] = np.zeros((height, width), dtype=np.int64)
23        self.lattice[rng.random(size=self.lattice.shape) < probability] = 1
24
25        if not tb:
26            def deco(func: Callable[[], None]) -> Callable[[], None]:
27                self.lattice[0, :] = 0
28                self.lattice[-1, :] = 0
29
30                def tmp() -> None:
31                    func()
32                    self.lattice[0, :] = 0
33                    self.lattice[-1, :] = 0
34                    return None
35                return tmp
36            self.next = deco(self.next)
37
38        if not lr:
39            def deco(func: Callable[[], None]) -> Callable[[], None]:
40                self.lattice[:, 0] = 0
41                self.lattice[:, -1] = 0
42
43                def tmp() -> None:
44                    func()
45                    self.lattice[:, 0] = 0
46                    self.lattice[:, -1] = 0
47                    return None
48                return tmp
49            self.next = deco(self.next)
50
51        return None
52
53    def next(self) -> None:
54        # 可読性のために PEP8 に違反した記述にしている。
55        neighbour: np.typing.NDArray[np.int64] = (
56            np.roll(self.lattice, shift=(-1, -1), axis=(0, 1)) +
57            np.roll(self.lattice, shift=( 1, -1), axis=(0, 1)) +
58            np.roll(self.lattice, shift=(-1,  1), axis=(0, 1)) +
59            np.roll(self.lattice, shift=( 1,  1), axis=(0, 1)) +
60            np.roll(self.lattice, shift=( 0, -1), axis=(0, 1)) +
61            np.roll(self.lattice, shift=( 0,  1), axis=(0, 1)) +
62            np.roll(self.lattice, shift=(-1,  0), axis=(0, 1)) +
63            np.roll(self.lattice, shift=( 1,  0), axis=(0, 1))
64        )
65        lattice: np.typing.NDArray[np.int64] = np.zeros_like(self.lattice)
66        for i, neighbour_counts in enumerate(self.rule):
67            b: np.typing.NDArray[np.bool] = self.lattice == i
68            for j in neighbour_counts:
69                lattice[b & (neighbour == j)] = 1
70        self.lattice = lattice
71        return None

LifeGameImage.py

LifeGame を継承し、格子の状態を色付き画像（PIL の Image および Tkinter 表示用の PhotoImage）として出力する機能を追加したモジュール。配色に用いる hsv2rgb 関数は utility.py からインポートして利用している。

ダウンロード

 1import numpy as np
 2from PIL import Image, ImageTk
 3
 4from LifeGame import LifeGame
 5from utility import hsv2rgb
 6
 7
 8class LifeGameImage(LifeGame):
 9    def __init__(
10        self,
11        rule: list[list[int]] | None = None,
12        width: int = 256,
13        height: int = 256,
14        probability: float = 0.3,
15        tb: bool = True,
16        lr: bool = True,
17        c0: np.typing.NDArray[np.uint8] | None = None,
18        c1: np.typing.NDArray[np.uint8] | None = None,
19        pixel: int = 2,
20    ) -> None:
21
22        super().__init__(rule, width, height, probability, tb, lr)
23
24        # c0, c1 片方のみの指定は不完全な入力とする。
25        if c0 is None or c1 is None:
26            rng: np.random.Generator = np.random.default_rng()
27            hue: float = float(rng.random())
28            saturation: float = 0.875
29            value: float = 0.875
30            c0 = hsv2rgb(hue, saturation, 1-value)
31            c1 = hsv2rgb(hue, saturation, value)
32        self.c0: np.typing.NDArray[np.uint8] = c0
33        self.c1: np.typing.NDArray[np.uint8] = c1
34
35        self.image_size: tuple[int, int] = (pixel*width, pixel*height)
36
37        self.image_array: np.typing.NDArray[np.uint8] = np.zeros((height, width, 3), dtype=np.uint8)
38
39        return None
40
41    def image_np(self) -> np.typing.NDArray[np.uint8]:
42        self.image_array[self.lattice == 0] = self.c0
43        self.image_array[self.lattice == 1] = self.c1
44        return self.image_array
45
46    def image_pil(self) -> Image.Image:
47        im: Image.Image = Image.fromarray(self.image_np())
48        return im.resize(self.image_size, resample=Image.Resampling.NEAREST)
49
50    def image_tk(self) -> ImageTk.PhotoImage:
51        return ImageTk.PhotoImage(self.image_pil())

FrameLifeGame.py

LifeGameImage が生成する画像を Tkinter の ttk.Frame 上に表示し、一定間隔ごとに次世代へ自動更新するウィジェットを実装したモジュール。

LifeGame の近傍セル数計算を np.roll の代わりに scipy.signal.convolve2d で行う代替実装。クラス名・インターフェースは LifeGame.py と同一である。

 2
 3import numpy as np
 4from scipy.signal import convolve2d
 5
 6
 7class LifeGame:
 8    def __init__(
 9        self,
10        rule: list[list[int]] | None = None,
11        width: int = 256,
12        height: int = 256,
13        probability: float = 0.3,
14        tb: bool = True,
15        lr: bool = True,
16    ) -> None:
17
18        if rule is None:
19            rule = [[3], [2, 3]]
20        self.rule: list[list[int]] = rule
21
22        self.kernel: np.typing.NDArray[np.int64] = np.array([
23            [1, 1, 1],
24            [1, 0, 1],
25            [1, 1, 1],
26        ], dtype=np.int64)
27
28        rng: np.random.Generator = np.random.default_rng()
29        self.lattice: np.typing.NDArray[np.int64] = np.zeros((height, width), dtype=np.int64)
30        self.lattice[rng.random(size=self.lattice.shape) < probability] = 1
31
32        if not tb:
33            def deco(func: Callable[[], None]) -> Callable[[], None]:
34                self.lattice[0, :] = 0
35                self.lattice[-1, :] = 0
36
37                def tmp() -> None:
38                    func()
39                    self.lattice[0, :] = 0
40                    self.lattice[-1, :] = 0
41                    return None
42                return tmp
43            self.next = deco(self.next)
44
45        if not lr:
46            def deco(func: Callable[[], None]) -> Callable[[], None]:
47                self.lattice[:, 0] = 0
48                self.lattice[:, -1] = 0
49
50                def tmp() -> None:
51                    func()
52                    self.lattice[:, 0] = 0
53                    self.lattice[:, -1] = 0
54                    return None
55                return tmp
56            self.next = deco(self.next)
57
58        return None
59
60    def next(self) -> None:
61        neighbour: np.typing.NDArray[np.int64] = convolve2d(
62            self.lattice,
63            self.kernel,
64            mode="same",
65            boundary="wrap",
66        )
67        lattice: np.typing.NDArray[np.int64] = np.zeros_like(self.lattice)
68        for i, neighbour_counts in enumerate(self.rule):
69            b: np.typing.NDArray[np.bool] = self.lattice == i
70            for j in neighbour_counts:
71                lattice[b & (neighbour == j)] = 1
72        self.lattice = lattice
73        return None

lifegame_fullHD_0001.py

LifeGame.py （np.roll 版）と LifeGame_scipy.py （convolve2d 版）とで、盤面サイズを変えながら処理速度を比較するベンチマークスクリプト。

 2
 3from LifeGame import LifeGame as LGn
 4from LifeGame_scipy import LifeGame as LGs

 7def calculation(lg: LGn | LGs) -> float:
 8    start: float = time.perf_counter()
 9    for _ in range(256):
10        lg.next()
11    end: float = time.perf_counter()
12    return end - start

15def main() -> None:
16    print("Processing speed comparison")
17    for pixel in [4, 2, 1]:
18        width: int = 1920 // pixel
19        height: int = 1080 // pixel
20        print("\n".join([
21            42*"=",
22            f"width: {width:>4}, height: {height:>4}",
23            42*"=",
24        ]))
25        for _ in range(4):
26            lap_n: float = calculation(LGn(width=width, height=height))
27            lap_s: float = calculation(LGs(width=width, height=height))
28            result: str = "numpy" if lap_n < lap_s else "scipy"
29            print(f"result: {result}, numpy : {lap_n:.2f}, scipy: {lap_s:.2f}")
