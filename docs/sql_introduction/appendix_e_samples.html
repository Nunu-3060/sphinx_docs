create_shop_db.py

 1"""サンプルデータベース shop.db を作成するスクリプト.
 2
 3schema.sql（テーブル定義）と data.sql（データ）を読み込み、
 4このスクリプトと同じフォルダーに shop.db を作成します。
 5shop.db が既にある場合は削除してから作り直すため、
 6データを変更してしまったときに初期状態へ戻す用途にも使えます。
 7
 8使い方::
 9
10    python create_shop_db.py
11"""
12
13import sqlite3
14from pathlib import Path
15
16HERE = Path(__file__).resolve().parent
17DB_PATH = HERE / "shop.db"
18SQL_FILES = [HERE / "schema.sql", HERE / "data.sql"]
19
20
21def create_database(db_path: Path, sql_files: list[Path]) -> None:
22    """SQL ファイルを順に実行してデータベースを作成します."""
23    if db_path.exists():
24        db_path.unlink()
25
26    conn = sqlite3.connect(db_path)
27    try:
28        conn.execute("PRAGMA foreign_keys = ON")
29        for sql_file in sql_files:
30            conn.executescript(sql_file.read_text(encoding="utf-8"))
31        conn.commit()
32    finally:
33        conn.close()
34
35
36def main() -> None:
37    """shop.db を作成し、各テーブルの行数を表示します."""
38    create_database(DB_PATH, SQL_FILES)
39
40    conn = sqlite3.connect(DB_PATH)
41    try:
42        tables = [
43            "categories", "products", "customers", "orders", "order_items",
44        ]
45        print(f"{DB_PATH.name} を作成しました。")
46        for table in tables:
47            # テーブル名はプレースホルダーで渡せないため文字列に埋め込みます。
48            # 埋め込むのは上で定義した固定の名前だけなので安全です。
49            (count,) = conn.execute(f"SELECT COUNT(*) FROM {table}").fetchone()
50            print(f"  {table}: {count} 行")
51    finally:
52        conn.close()
53
54
55if __name__ == "__main__":
56    main()

run_query.py

  1"""SQL ファイル（または SQL 文）を実行し、結果を表形式で表示するスクリプト.
  2
  3SQL 文を 1 文ずつ実行し、文ごとに結果を表示します。
  4途中の文でエラーが発生しても、エラー内容を表示して次の文へ進みます。
  5
  6既定では shop.db をメモリ上に複製してから実行するため、
  7UPDATE や DELETE を実行しても shop.db の内容は変わりません。
  8変更を shop.db に保存したい場合は --save を指定します。
  9
 10使い方::
 11
 12    python run_query.py sql/ch04_select.sql
 13    python run_query.py -e "SELECT * FROM products LIMIT 3;"
 14    python run_query.py --save my_changes.sql
 15"""
 16
 17import argparse
 18import sqlite3
 19import sys
 20import unicodedata
 21from collections.abc import Sequence
 22from pathlib import Path
 23from typing import Any
 24
 25HERE = Path(__file__).resolve().parent
 26DEFAULT_DB = HERE / "shop.db"
 27
 28
 29def split_statements(sql: str) -> list[str]:
 30    """SQL の文字列を 1 文ずつのリストに分割します.
 31
 32    sqlite3.complete_statement() は、文字列がセミコロンで終わる
 33    完全な SQL 文かどうかを判定する関数です。
 34    文字列リテラルやコメントの中のセミコロンは文の終わりとみなしません。
 35    セミコロンが現れるたびに判定し、完全な文になった時点で区切ります。
 36    """
 37    statements: list[str] = []
 38    start = 0
 39    for i, ch in enumerate(sql):
 40        if ch == ";" and sqlite3.complete_statement(sql[start:i + 1]):
 41            statements.append(sql[start:i + 1].strip())
 42            start = i + 1
 43    # 末尾に残ったコメントや空白は無視します。
 44    rest = sql[start:]
 45    code_lines = [
 46        line for line in rest.splitlines()
 47        if line.strip() and not line.strip().startswith("--")
 48    ]
 49    if code_lines:
 50        statements.append(rest.strip())
 51    return statements
 52
 53
 54def display_width(text: str) -> int:
 55    """端末に表示したときの幅を返します（全角文字は 2 として数えます）."""
 56    return sum(
 57        2 if unicodedata.east_asian_width(ch) in ("W", "F") else 1
 58        for ch in text
 59    )
 60
 61
 62def pad(text: str, width: int) -> str:
 63    """表示幅が width になるように右側を空白で埋めます."""
 64    return text + " " * (width - display_width(text))
 65
 66
 67def to_text(value: Any) -> str:
 68    """セルの値を表示用の文字列に変換します（None は NULL と表示します）."""
 69    return "NULL" if value is None else str(value)
 70
 71
 72def format_table(columns: Sequence[str], rows: Sequence[Sequence[Any]]) -> str:
 73    """列名と行のリストから、罫線付きの表の文字列を作ります."""
 74    cells = [[to_text(v) for v in row] for row in rows]
 75    widths = [display_width(c) for c in columns]
 76    for row in cells:
 77        widths = [max(w, display_width(c)) for w, c in zip(widths, row)]
 78
 79    lines = [" | ".join(pad(c, w) for c, w in zip(columns, widths))]
 80    lines.append("-+-".join("-" * w for w in widths))
 81    for row in cells:
 82        lines.append(" | ".join(pad(c, w) for c, w in zip(row, widths)))
 83    lines.append(f"({len(rows)} 行)")
 84    return "\n".join(lines)
 85
 86
 87def run_statement(conn: sqlite3.Connection, statement: str) -> str:
 88    """1 文を実行し、結果を表示用の文字列で返します."""
 89    try:
 90        cursor = conn.execute(statement)
 91        rows = cursor.fetchall()
 92    except sqlite3.Error as error:
 93        return f"エラー: {type(error).__name__}: {error}"
 94
 95    if cursor.description is not None:
 96        columns = [d[0] for d in cursor.description]
 97        return format_table(columns, rows)
 98    if cursor.rowcount >= 0:
 99        return f"({cursor.rowcount} 行を変更しました)"
100    return "(実行しました)"
101
102
103def open_database(db_path: Path, save: bool) -> sqlite3.Connection:
104    """データベースを開きます.
105
106    save が False のときは、メモリ上に複製したデータベースを返します。
107    autocommit=True を指定すると、Python 側で暗黙にトランザクションを
108    開始しなくなり、BEGIN や COMMIT を SQL として書いたとおりに実行できます。
109    """
110    if save:
111        conn = sqlite3.connect(db_path, autocommit=True)
112    else:
113        source = sqlite3.connect(db_path)
114        conn = sqlite3.connect(":memory:", autocommit=True)
115        try:
116            source.backup(conn)
117        finally:
118            source.close()
119    conn.execute("PRAGMA foreign_keys = ON")
120    return conn
121
122
123def main() -> int:
124    """コマンドライン引数を解釈して SQL を実行します."""
125    parser = argparse.ArgumentParser(description="SQL を実行して結果を表示します。")
126    group = parser.add_mutually_exclusive_group(required=True)
127    group.add_argument("sql_file", nargs="?", type=Path, help="SQL ファイル")
128    group.add_argument("-e", "--execute", help="実行する SQL 文")
129    parser.add_argument(
130        "--db", type=Path, default=DEFAULT_DB, help="データベースファイル"
131    )
132    parser.add_argument(
133        "--save", action="store_true", help="変更をデータベースファイルに保存する"
134    )
135    args = parser.parse_args()
136
137    if not args.db.exists():
138        print(f"{args.db} がありません。先に create_shop_db.py を実行してください。")
139        return 1
140
141    if args.execute is not None:
142        sql: str = args.execute
143    else:
144        sql = args.sql_file.read_text(encoding="utf-8")
145
146    conn = open_database(args.db, args.save)
147    try:
148        for statement in split_statements(sql):
149            print(f">>> {statement}")
150            print(run_statement(conn, statement))
151            print()
152    finally:
153        conn.close()
154    return 0
155
156
157if __name__ == "__main__":
158    sys.exit(main())
